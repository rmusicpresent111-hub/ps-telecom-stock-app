     1→     1→/**
     2→     2→ * Local-First Service Layer — 100% offline, NO cloud database.
     3→     3→ *
     4→     4→ * READ STRATEGY:  Always served from local IndexedDB (instant, works with zero internet).
     5→     5→ * WRITE STRATEGY: Written to local IndexedDB immediately (atomic where it matters).
     6→     6→ * DELETE STRATEGY: Deleted from local IndexedDB (with cascades where required).
     7→     7→ *
     8→     8→ * Stock transactions are applied atomically (products + transactions stores in a
     9→     9→ * single IndexedDB transaction) so rapid taps can never create negative stock.
    10→    10→ * Profit/report math uses the purchase-price SNAPSHOT captured on each transaction,
    11→    11→ * so later price edits never rewrite historical profit.
    12→    12→ */

    31→    31→// ============ DATE HELPERS (local timezone — never use UTC for business days) ============
    32→    32→
    33→    33→/**
    34→    34→ * Returns YYYY-MM-DD for a Date in LOCAL time.
    35→    35→ * toISOString() uses UTC — for IST (UTC+5:30) sales between 00:00–05:30 local
    36→    36→ * would land on the previous day. This helper fixes that class of bugs.
    37→    37→ */
    38→    38→export function localDateStr(d: Date = new Date()): string {
    39→    39→  const y = d.getFullYear();
    40→    40→  const m = String(d.getMonth() + 1).padStart(2, '0');
    41→    41→  const day = String(d.getDate()).padStart(2, '0');
    42→    42→  return `${y}-${m}-${day}`;
    43→    43→}

    45→    45→/**
    46→    46→ * Resolves a reporting period to a [from, to] date range (local time).
    47→    47→ */
    48→    48→export function resolvePeriodRange(options?: { period?: string; date?: string; from?: string; to?: string }): { from: string; to: string } {
    49→    49→  const now = new Date();
    50→    50→  if (options?.from && options?.to) {
    51→    51→    // Guard against inverted custom ranges (from > to) — silently swap.
    52→    52→    return options.from <= options.to
    53→    53→      ? { from: options.from, to: options.to }
    54→    54→      : { from: options.to, to: options.from };
    55→    55→  }
    56→    56→  if (options?.date) return { from: options.date, to: options.date };
    57→    57→
    58→    58→  const todayStr = localDateStr(now);
    59→    59→  const period = options?.period || 'today';
    60→    60→  if (period === 'week') {
    61→    61→    const weekStart = new Date(now);
    62→    62→    const day = weekStart.getDay();
    63→    63→    const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1); // Monday start
    64→    64→    weekStart.setDate(diff);
    65→    65→    return { from: localDateStr(weekStart), to: todayStr };
    66→    66→  }
    67→    67→  if (period === 'month') {
    68→    68→    return { from: `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`, to: todayStr };
    69→    69→  }
    70→    70→  // today (default) or any unknown period
    71→    71→  return { from: todayStr, to: todayStr };
    72→    72→}

    76→    76→function stripSync<T>(row: Record<string, unknown>): T {
    77→    77→  const { _synced: _s, _dirty: _d, ...rest } = row;
    78→    78→  return rest as T;
    79→    79→}

    81→    81→function sortTxnsDesc(rows: OfflineDBSchema['transactions']['value'][]): OfflineDBSchema['transactions']['value'][] {
    82→    82→  return [...rows].sort((a, b) => {
    83→    83→    if (a.date !== b.date) return b.date.localeCompare(a.date);
    84→    84→    return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    85→    85→  });
    86→    86→}
    87→    87→
    88→    88→function localTxnsToTransactions(localTxns: OfflineDBSchema['transactions']['value'][]): Transaction[] {
    89→    89→  return sortTxnsDesc(localTxns).map(t => stripSync<Transaction>(t));
    90→    90→}

    92→    92→/** Cost basis for a transaction: the SNAPSHOT taken at sale time, falling back to the current product. */
    93→    93→function txnCostBasis(t: Transaction, products: Product[]): number {
    94→    94→  return t.product?.purchasePrice ?? products.find(p => p.id === t.productId)?.purchasePrice ?? 0;
    95→    95→}

   250→   250→// ============ TRANSACTIONS (Local, ATOMIC stock updates) ============
   251→   251→
   252→   252→/**
   253→   253→ * Creates a transaction AND applies the stock change in ONE atomic IndexedDB
   254→   254→ * transaction. Two rapid sells can never both pass the same stock check.
   255→   255→ *
   256→   256→ * NOTE: idb v8's db.transaction() has NO callback parameter — the correct
   257→   257→ * atomic pattern is: open tx → await requests → `await tx.done`.
   258→   258→ */
   259→   259→export async function createTransactionOffline(transactionData: {
   260→   260→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   261→   261→  productId: string;
   262→   262→  quantity: number;
   263→   263→  unitPrice?: number;
   264→   264→  totalAmount?: number;
   265→   265→  date?: string;
   266→   266→  userId: string;
   267→   267→}): Promise<{ transaction: Transaction }> {
   268→   268→  const quantity = Math.floor(Number(transactionData.quantity));
   269→   269→  if (!Number.isFinite(quantity) || quantity <= 0) throw new Error('Quantity must be a positive whole number');
   270→   270→
   271→   271→  const unitPrice = Number(transactionData.unitPrice ?? 0);
   272→   272→  const totalAmount = Number(transactionData.totalAmount ?? 0);
   273→   273→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   274→   274→  if (!Number.isFinite(totalAmount) || totalAmount < 0) throw new Error('Total amount cannot be negative');
   275→   275→
   276→   276→  const db = await getOfflineDB();
   277→   277→  const now = new Date().toISOString();
   278→   278→  const id = generateRecordId();
   279→   279→
   280→   280→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   281→   281→  const productsStore = tx.objectStore('products');
   282→   282→  const transactionsStore = tx.objectStore('transactions');
   283→   283→
   284→   284→  // Validate BEFORE writing anything — a readwrite tx with no writes commits empty.
   285→   285→  const product = await productsStore.get(transactionData.productId);
   286→   286→  if (!product) {
   287→   287→    throw new Error('Product not found');
   288→   288→  }
   289→   289→
   290→   290→  const isOut = transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL';
   291→   291→  if (isOut && product.quantity < quantity) {
   292→   292→    throw new Error(`Insufficient stock — only ${product.quantity} left`);
   293→   293→  }
   294→   294→
   295→   295→  const quantityChange = transactionData.type === 'STOCK_IN' ? quantity : -quantity;
   296→   296→
   297→   297→  const row: OfflineDBSchema['transactions']['value'] = {
   298→   298→    id,
   299→   299→    productId: transactionData.productId,
   300→   300→    type: transactionData.type,
   301→   301→    quantity,
   302→   302→    unitPrice,
   303→   303→    totalAmount,
   304→   304→    date: transactionData.date || localDateStr(),
   305→   305→    userId: transactionData.userId,
   306→   306→    createdAt: now,
   307→   307→    // Snapshot the product prices at transaction time — historical profit must
   308→   308→    // never change when the user later edits the product's prices.
   309→   309→    product: {
   310→   310→      id: product.id,
   311→   311→      name: product.name,
   312→   312→      purchasePrice: product.purchasePrice,
   313→   313→      sellingPrice: product.sellingPrice,
   314→   314→    },
   315→   315→    _synced: Date.now(),
   316→   316→    _dirty: 0,
   317→   317→  };
   318→   318→
   319→   319→  await transactionsStore.put(row);
   320→   320→  await productsStore.put({
   321→   321→    ...product,
   322→   322→    quantity: product.quantity + quantityChange,
   323→   323→    updatedAt: now,
   324→   324→  });
   325→   325→
   326→   326→  // Commit — rejects (and rolls back both stores) if anything failed.
   327→   327→  await tx.done;
   328→   328→
   329→   329→  return { transaction: stripSync<Transaction>(row) };
   330→   330→}

   332→   332→export async function deleteTransactionOffline(id: string, userId: string): Promise<{ success: boolean }> {
   333→   333→  const db = await getOfflineDB();
   334→   334→  const now = new Date().toISOString();
   335→   335→
   336→   336→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   337→   337→  const productsStore = tx.objectStore('products');
   338→   338→  const transactionsStore = tx.objectStore('transactions');
   339→   339→
   340→   340→  const txn = await transactionsStore.get(id);
   341→   341→  if (!txn || txn.userId !== userId) {
   342→   342→    // Nothing to delete — no writes were queued, committing empty is harmless.
   343→   343→    await tx.done;
   344→   344→    return { success: true };
   345→   345→  }
   346→   346→
   347→   347→  // Reverse the stock change caused by this transaction (same atomic tx)
   348→   348→  const product = await productsStore.get(txn.productId);
   349→   349→  if (product) {
   350→   350→    const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   351→   351→    await productsStore.put({
   352→   352→      ...product,
   353→   353→      quantity: Math.max(0, product.quantity + quantityChange),
   354→   354→      updatedAt: now,
   355→   355→    });
   356→   356→  }
   357→   357→
   358→   358→  await transactionsStore.delete(id);
   359→   359→  await tx.done;
   360→   360→
   361→   361→  return { success: true };
   362→   362→}

   364→   364→export async function getTransactionsOffline(userId: string, options?: { type?: string; productId?: string; from?: string; to?: string }): Promise<Transaction[]> {
   365→   365→  let localTxns = await offlineTransactions.getAll(userId);
   366→   366→
   367→   367→  if (options?.type) localTxns = localTxns.filter(t => t.type === options.type);
   368→   368→  if (options?.productId) localTxns = localTxns.filter(t => t.productId === options.productId);
   369→   369→  if (options?.from) localTxns = localTxns.filter(t => t.date >= options.from!);
   370→   370→  if (options?.to) localTxns = localTxns.filter(t => t.date <= options.to!);
   371→   371→
   372→   372→  return localTxnsToTransactions(localTxns);
   373→   373→}

   377→   377→export async function getDashboardOffline(userId: string) {
   378→   378→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   379→   379→    offlineProducts.getAll(userId),
   380→   380→    offlineCategories.getAll(userId),
   381→   381→    offlineTransactions.getAll(userId),
   382→   382→  ]);
   383→   383→
   384→   384→  const products = localProducts.map(p => stripSync<Product>(p));
   385→   385→  const categories = localCategories.map(c => stripSync<Category>(c));
   386→   386→  const transactions = localTxnsToTransactions(localTransactions);
   387→   387→
   388→   388→  const today = localDateStr();
   389→   389→  const todayTxns = transactions.filter(t => t.date === today);
   390→   390→
   391→   391→  const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   392→   392→  const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   393→   393→  const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   394→   394→  const recentTransactions = transactions.slice(0, 5);
   395→   395→
   396→   396→  const productCountMap: Record<string, number> = {};
   397→   397→  for (const p of products) {
   398→   398→    productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   399→   399→  }
   400→   400→  const catsWithCount = categories.map(c => ({
   401→   401→    ...c,
   402→   402→    _count: { products: productCountMap[c.id] || 0 },
   403→   403→  }));
   404→   404→
   405→   405→  const sevenDaysAgo = new Date();
   406→   406→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   407→   407→  const sevenDaysAgoStr = localDateStr(sevenDaysAgo);
   408→   408→
   409→   409→  const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   410→   410→  const saleMap = new Map<string, { sales: number; quantity: number }>();
   411→   411→  for (const t of saleTxns) {
   412→   412→    const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   413→   413→    existing.sales += t.totalAmount;
   414→   414→    existing.quantity += t.quantity;
   415→   415→    saleMap.set(t.date, existing);
   416→   416→  }
   417→   417→
   418→   418→  const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   419→   419→  for (let i = 6; i >= 0; i--) {
   420→   420→    const d = new Date();
   421→   421→    d.setDate(d.getDate() - i);
   422→   422→    const dateStr = localDateStr(d);
   423→   423→    const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   424→   424→    const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   425→   425→    saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   426→   426→  }
   427→   427→
   428→   428→  const stockByCategory = new Map<string, { quantity: number; value: number }>();
   429→   429→  for (const p of products) {
   430→   430→    const cat = catsWithCount.find(c => c.id === p.categoryId);
   431→   431→    const catName = cat?.name || 'Unknown';
   432→   432→    const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   433→   433→    existing.quantity += p.quantity;
   434→   434→    existing.value += p.quantity * p.purchasePrice;
   435→   435→    stockByCategory.set(catName, existing);
   436→   436→  }
   437→   437→
   438→   438→  const stockOverview = Array.from(stockByCategory.entries())
   439→   439→    .filter(([_, data]) => data.quantity > 0)
   440→   440→    .map(([category, data]) => ({ category, ...data }));
   441→   441→
   442→   442→  return {
   443→   443→    stats: {
   444→   444→      totalItems: products.length,
   445→   445→      lowItems,
   446→   446→      todayTransactions: todayTxns.length,
   447→   447→      stockValue,
   448→   448→    },
   449→   449→    lowStockProducts,
   450→   450→    recentTransactions,
   451→   451→    categories: catsWithCount,
   452→   452→    saleOverview,
   453→   453→    stockOverview,
   454→   454→  };
   455→   455→}

   666→   666→export async function getReportsOffline(userId: string, type: string, options?: { from?: string; to?: string }) {
   667→   667→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   668→   668→    offlineProducts.getAll(userId),
   669→   669→    offlineCategories.getAll(userId),
   670→   670→    offlineTransactions.getAll(userId),
   671→   671→  ]);
   672→   672→
   673→   673→  const products = localProducts.map(p => stripSync<Product>(p));
   674→   674→  const categories = localCategories.map(c => stripSync<Category>(c));
   675→   675→  const transactions = localTxnsToTransactions(localTransactions);
   676→   676→
   677→   677→  // Apply date filters
   678→   678→  let filteredTxns = transactions;
   679→   679→  if (options?.from) filteredTxns = filteredTxns.filter(t => t.date >= options.from!);
   680→   680→  if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   681→   681→
   682→   682→  if (type === 'stock-value') {
   683→   683→    const stockValueData = categories.map(cat => {
   684→   684→      const catProducts = products.filter(p => p.categoryId === cat.id);
   685→   685→      const totalQty = catProducts.reduce((sum, p) => sum + p.quantity, 0);
   686→   686→      const totalPurchaseValue = catProducts.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   687→   687→      const totalSellingValue = catProducts.reduce((sum, p) => sum + p.quantity * p.sellingPrice, 0);
   688→   688→      const totalProfit = totalSellingValue - totalPurchaseValue;
   689→   689→      const lowStockCount = catProducts.filter(p => p.quantity <= p.lowStockThreshold).length;
   690→   690→
   691→   691→      return {
   692→   692→        categoryId: cat.id,
   693→   693→        categoryName: cat.name,
   694→   694→        categoryImage: cat.image,
   695→   695→        productCount: catProducts.length,
   696→   696→        totalQty,
   697→   697→        totalPurchaseValue,
   698→   698→        totalSellingValue,
   699→   699→        totalProfit,
   700→   700→        lowStockCount,
   701→   701→        products: catProducts.map(p => ({
   702→   702→          id: p.id,
   703→   703→          name: p.name,
   704→   704→          quantity: p.quantity,
   705→   705→          purchasePrice: p.purchasePrice,
   706→   706→          sellingPrice: p.sellingPrice,
   707→   707→          stockValue: p.quantity * p.purchasePrice,
   708→   708→          purchaseValue: p.quantity * p.purchasePrice,
   709→   709→          lowStock: p.quantity <= p.lowStockThreshold,
   710→   710→        })),
   711→   711→      };
   712→   712→    }).filter(item => item.productCount > 0);
   713→   713→
   714→   714→    const grandTotal = {
   715→   715→      totalProducts: stockValueData.reduce((s, d) => s + d.productCount, 0),
   716→   716→      totalQty: stockValueData.reduce((s, d) => s + d.totalQty, 0),
   717→   717→      totalPurchaseValue: stockValueData.reduce((s, d) => s + d.totalPurchaseValue, 0),
   718→   718→      totalSellingValue: stockValueData.reduce((s, d) => s + d.totalSellingValue, 0),
   719→   719→      totalProfit: stockValueData.reduce((s, d) => s + d.totalProfit, 0),
   720→   720→      totalLowStock: stockValueData.reduce((s, d) => s + d.lowStockCount, 0),
   721→   721→    };
   722→   722→
   723→   723→    return { type: 'stock-value', data: stockValueData, grandTotal };
   724→   724→  }
   725→   725→
   726→   726→  // Daily report
   727→   727→  if (type === 'daily') {
   728→   728→    const dailyMap = new Map<string, {
   729→   729→      date: string; revenue: number; cost: number; profit: number;
   730→   730→      stockIn: number; stockOut: number; sell: number;
   731→   731→    }>();

   742→   742→        entry.cost += (t.quantity || 0) * costBasis;
   743→   743→        entry.profit += (t.totalAmount || 0) - (t.quantity || 0) * costBasis;
   744→   744→        entry.sell += t.quantity || 0;
   745→   745→      } else if (t.type === 'STOCK_IN') {
   746→   746→        entry.cost += (t.quantity || 0) * (t.unitPrice || 0);
   747→   747→        entry.stockIn += t.quantity || 0;
   748→   748→      } else if (t.type === 'STOCK_OUT') {
   749→   749→        entry.stockOut += t.quantity || 0;

   978→   978→  const num = Number(value);

   984→   984→async function nextBillNumber(userId: string, prefix: string): Promise<string> {
   985→   985→  const bills = await offlineBills.getAll(userId);
   986→   986→  const now = new Date();
   987→   987→  const yy = String(now.getFullYear()).slice(-2);
   988→   988→  const mm = String(now.getMonth() + 1).padStart(2, '0');
   989→   989→  const monthPrefix = `${prefix}-${yy}${mm}-`;
   990→   990→  let maxSeq = 0;
   991→   991→  for (const b of bills) {
   992→   992→    if (typeof b.billNumber === 'string' && b.billNumber.startsWith(monthPrefix)) {
   993→   993→      const seq = parseInt(b.billNumber.slice(monthPrefix.length), 10);
   994→   994→      if (Number.isFinite(seq) && seq > maxSeq) maxSeq = seq;
   995→   995→    }
   996→   996→  }
   997→   997→  return `${monthPrefix}${String(maxSeq + 1).padStart(4, '0')}`;
   998→   998→}

  1000→  1000→export async function createBillOffline(data: {
  1001→  1001→  userId: string;
  1002→  1002→  transactionId: string;
  1003→  1003→  productId: string;
  1004→  1004→  productName: string;
  1005→  1005→  quantity: number;
  1006→  1006→  unitPrice: number;
  1007→  1007→  customerName?: string;
  1008→  1008→  customerMobile?: string;
  1009→  1009→  discountType?: 'amount' | 'percent';
  1010→  1010→  discountValue?: number;
  1011→  1011→  paymentMethod?: PaymentMethod;
  1012→  1012→  paidAmount?: number;
  1013→  1013→  note?: string;
  1014→  1014→  date?: string;
  1015→  1015→}): Promise<{ bill: Bill }> {
  1016→  1016→  const qty = Math.floor(Number(data.quantity));
  1017→  1017→  if (!Number.isFinite(qty) || qty <= 0) throw new Error('Quantity must be a positive whole number');
  1018→  1018→  const unitPrice = Number(data.unitPrice);
  1019→  1019→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
  1020→  1020→
  1021→  1021→  const settings = await getBillingSettingsOffline(data.userId);
  1022→  1022→
  1023→  1023→  const discountType = data.discountType === 'percent' ? 'percent' : 'amount';
  1024→  1024→  let discountValue = Math.max(0, Number(data.discountValue ?? 0) || 0);
  1025→  1025→  const subtotal = round2(qty * unitPrice);
  1026→  1026→
  1027→  1027→  let discountAmount = 0;
  1028→  1028→  if (discountType === 'percent') {
  1029→  1029→    discountValue = clampNumber(discountValue, 0, 100);
  1030→  1030→    discountAmount = round2((subtotal * discountValue) / 100);
  1031→  1031→  } else {
  1032→  1032→    discountAmount = round2(Math.min(discountValue, subtotal));
  1033→  1033→  }
  1034→  1034→
  1035→  1035→  const gstEnabled = !!settings.gstEnabled && !!settings.gstNumber.trim() && settings.gstRate > 0;
  1036→  1036→  const gstAmount = gstEnabled ? round2(((subtotal - discountAmount) * settings.gstRate) / 100) : 0;
  1037→  1037→  const total = round2(subtotal - discountAmount + gstAmount);
  1038→  1038→
  1039→  1039→  const paymentMethod: PaymentMethod = data.paymentMethod ?? 'cash';
  1040→  1040→  let paidAmount = data.paidAmount === undefined ? total : Math.max(0, Number(data.paidAmount) || 0);
  1041→  1041→  if (paymentMethod === 'due') paidAmount = 0;
  1042→  1042→  paidAmount = round2(Math.min(paidAmount, total));
  1043→  1043→  const dueAmount = round2(total - paidAmount);
  1044→  1044→
  1045→  1045→  // Guard: one bill per stock-out/sell transaction — prevents accidental duplicates
  1046→  1046→  const existingBills = await offlineBills.getAll(data.userId);
  1047→  1047→  const dupe = existingBills.find(b => b.transactionId === data.transactionId);
  1048→  1048→  if (dupe) throw new Error('A bill already exists for this transaction');
  1049→  1049→
  1050→  1050→  const now = new Date().toISOString();
  1051→  1051→  const billNumber = await nextBillNumber(data.userId, settings.billPrefix.trim() || 'PS');
  1052→  1052→
  1053→  1053→  const bill: OfflineDBSchema['bills']['value'] = {
  1054→  1054→    id: generateRecordId(),
  1055→  1055→    userId: data.userId,
  1056→  1056→    billNumber,
  1057→  1057→    customerName: (data.customerName ?? '').trim() || 'Walk-in Customer',
  1058→  1058→    customerMobile: (data.customerMobile ?? '').replace(/[^\d+]/g, ''),
  1059→  1059→    items: [
  1060→  1060→      {
  1061→  1061→        productId: data.productId,
  1062→  1062→        name: data.productName,
  1063→  1063→        quantity: qty,
  1064→  1064→        unitPrice,
  1065→  1065→        total: subtotal,
  1066→  1066→      },
  1067→  1067→    ],
  1068→  1068→    subtotal,
  1069→  1069→    discountValue,
  1070→  1070→    discountType,
  1071→  1071→    discountAmount,
  1072→  1072→    gstEnabled,
  1073→  1073→    gstRate: gstEnabled ? settings.gstRate : 0,
  1074→  1074→    gstAmount,
  1075→  1075→    total,
  1076→  1076→    paymentMethod,
  1077→  1077→    paidAmount,
  1078→  1078→    dueAmount,
  1079→  1079→    note: (data.note ?? '').trim(),
  1080→  1080→    transactionId: data.transactionId,
  1081→  1081→    shopSnapshot: {
  1082→  1082→      name: settings.shopName,
  1083→  1083→      proprietorName: settings.proprietorName,
  1084→  1084→      address: settings.shopAddress,
  1085→  1085→      phone: settings.shopPhone,
  1086→  1086→      gstNumber: settings.gstNumber,
  1087→  1087→    },
  1088→  1088→    date: data.date || localDateStr(),
  1089→  1089→    createdAt: now,
  1090→  1090→    updatedAt: now,
  1091→  1091→  };
  1092→  1092→
  1093→  1093→  await offlineBills.put(bill);
  1094→  1094→  return { bill: bill as unknown as Bill };
  1095→  1095→}
