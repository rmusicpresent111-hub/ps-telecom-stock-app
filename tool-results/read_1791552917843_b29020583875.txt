   207→  const [stats, setStats] = useState<DashboardStats | null>(null);
   208→  const [lowStockProducts, setLowStockProducts] = useState<Product[]>([]);
   209→  const [saleOverview, setSaleOverview] = useState<SaleOverviewItem[]>([]);
   210→  const [stockOverview, setStockOverview] = useState<StockOverviewItem[]>([]);
   211→  const [todayProfit, setTodayProfit] = useState(0);
   212→  const [loading, setLoading] = useState(true);
   213→  const isFetchingRef = useRef(false);
   214→  const fetchDashboard = useCallback(async (isBackground = false) => {
   215→    if (!user?.id || isFetchingRef.current) return;
   216→
   217→    isFetchingRef.current = true;
   218→    try {
   219→      if (!isBackground) setLoading(true);
   220→      const data = await getDashboard(user.id) as Record<string, unknown>;
   221→      setStats(data.stats as DashboardStats | null);
   222→      setLowStockProducts((data.lowStockProducts as Product[]) || []);
   223→      setSaleOverview((data.saleOverview as SaleOverviewItem[]) || []);
   224→      setStockOverview((data.stockOverview as StockOverviewItem[]) || []);
   225→      if ((data.categories as Category[] | undefined)?.length) {
   226→        setCategories(data.categories as Category[]);
   227→      } else {
   228→        // Empty DB (fresh account or after Reset All Data / restore) must also
   229→        // reach the store — otherwise stale categories keep rendering.
   230→        setCategories([]);
   231→      }
   232→
   233→      // Calculate today's profit from transactions
   234→      // Profit = totalAmount - (quantity × purchase price SNAPSHOT at sale time)
   235→      const today = localDateStr();
   236→      const { getTransactionsOffline: getTxns } = await import('@/lib/offline-service');
   237→      const todayTxns = await getTxns(user.id, { from: today, to: today });
   238→      const sellTxns = (todayTxns || []).filter((tx) => tx.type === 'SELL');
   239→      let profit = 0;
   240→      for (const tx of sellTxns) {
   241→        // Use the snapshot captured on the transaction — current product price edits
   242→        // must not rewrite today's profit.
   243→        const cost = (tx.quantity || 0) * (tx.product?.purchasePrice ?? 0);
   244→        profit += (tx.totalAmount || 0) - cost;
   245→      }
   246→      setTodayProfit(profit);
   247→    } catch {
   248→      if (!isBackground) toast.error(t('error', language));
   249→    } finally {
   250→      setLoading(false);
   251→      isFetchingRef.current = false;
   252→    }
   253→  }, [user?.id]);
   254→
   255→  // Initial load
   256→  useEffect(() => {
   257→    fetchDashboard();
   258→  }, [fetchDashboard]);
   259→
   260→  // ✅ Background refresh on window focus (stale-while-revalidate)
   261→  useEffect(() => {
   262→    const handleFocus = () => {
   263→      fetchDashboard(true);
   264→    };
   265→    window.addEventListener('focus', handleFocus);
   266→    return () => window.removeEventListener('focus', handleFocus);
   267→  }, [fetchDashboard]);

   287→  const totalStockQty = useMemo(() => stockOverview.reduce((sum, d) => sum + d.quantity, 0), [stockOverview]);
   288→  const totalStockVal = useMemo(() => stockOverview.reduce((sum, d) => sum + d.value, 0), [stockOverview]);
