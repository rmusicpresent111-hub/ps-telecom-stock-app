    11→async function hashPassword(password: string): Promise<string> {
    12→  const encoder = new TextEncoder();
    13→  const data = encoder.encode(password);
    14→  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    15→  return Array.from(new Uint8Array(hashBuffer)).map(b => b.toString(16).padStart(2, '0')).join('');
    16→}

    18→export async function login(email: string, password: string) {
    19→  if (!email || !password) {
    20→    throw new Error('Email and password are required');
    21→  }
    22→
    23→  const { data, error } = await supabase
    24→    .from('users')
    25→    .select('*')
    26→    .eq('email', email)
    27→    .single();
    28→
    29→  if (error || !data) {
    30→    throw new Error('Invalid email or password');
    31→  }
    32→
    33→  const user = toCamelCase(data);
    34→  const hashedInput = await hashPassword(password);
    35→
    36→  if (hashedInput !== (user.password as string)) {
    37→    throw new Error('Invalid email or password');
    38→  }
    39→
    40→  const { password: _, ...userWithoutPassword } = user;
    41→  return { user: userWithoutPassword };
    42→}

    44→export async function signup(name: string, email: string, password: string, shopName: string) {
    45→  if (!name || !email || !password) {
    46→    throw new Error('Name, email, and password are required');
    47→  }
    48→
    49→  const { data: existingData } = await supabase
    50→    .from('users')
    51→    .select('id')
    52→    .eq('email', email)
    53→    .single();
    54→
    55→  if (existingData) {
    56→    throw new Error('Email already exists');
    57→  }
    58→
    59→  const hashedPassword = await hashPassword(password);
    60→  const now = new Date().toISOString();
    61→
    62→  const snakeCaseData = toSnakeCase({
    63→    id: generateId(),
    64→    name,
    65→    email,
    66→    password: hashedPassword,
    67→    shopName: shopName || 'PS TELECOM',
    68→    createdAt: now,
    69→    updatedAt: now,
    70→  });
    71→
    72→  const { data, error } = await supabase
    73→    .from('users')
    74→    .insert(snakeCaseData)
    75→    .select('*')
    76→    .single();
    77→
    78→  if (error) {
    79→    throw new Error('Internal server error');
    80→  }
    81→
    82→  const user = toCamelCase(data);
    83→  const { password: _, ...userWithoutPassword } = user;
    84→  return { user: userWithoutPassword };
    85→}

   169→export async function getProducts(userId: string, options?: { categoryId?: string; search?: string }) {
   170→  const optsKey = options ? `${options.categoryId || ''}:${options.search || ''}` : 'all';
   171→  const key = cacheKeys.products(userId, optsKey);
   172→  const cached = getCached<{ products: unknown[] }>(key);
   173→  if (cached) return cached;
   174→
   175→  let query = supabase
   176→    .from('products')
   177→    .select('*, category:categories(*)')
   178→    .eq('user_id', userId)
   179→    .order('created_at', { ascending: false });
   180→
   181→  if (options?.categoryId) {
   182→    query = query.eq('category_id', options.categoryId);
   183→  }
   184→
   185→  if (options?.search) {
   186→    query = query.ilike('name', `%${options.search}%`);
   187→  }
   188→
   189→  const { data, error } = await query;
   190→
   191→  if (error) {
   192→    throw new Error('Internal server error');
   193→  }
   194→
   195→  const products = (data || []).map((product: Record<string, unknown>) => {
   196→    const { category, ...productFields } = product as Record<string, unknown>;
   197→    const camelProduct = toCamelCase(productFields as Record<string, unknown>);
   198→    if (category && typeof category === 'object') {
   199→      camelProduct.category = toCamelCase(category as Record<string, unknown>);
   200→    }
   201→    return camelProduct;
   202→  });
   203→
   204→  const result = { products };
   205→  setCache(key, result, cacheTTL.products);
   206→  return result;
   207→}

   345→export async function deleteTransaction(id: string) {
   346→  // First get the transaction to reverse stock change
   347→  const { data: txn } = await supabase
   348→    .from('transactions')
   349→    .select('product_id, type, quantity')
   350→    .eq('id', id)
   351→    .single();
   352→
   353→  if (txn) {
   354→    // Reverse the stock change on the product
   355→    const { data: product } = await supabase
   356→      .from('products')
   357→      .select('quantity, user_id')
   358→      .eq('id', txn.product_id)
   359→      .single();
   360→
   361→    if (product) {
   362→      const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   363→      await supabase
   364→        .from('products')
   365→        .update({ quantity: Math.max(0, product.quantity + quantityChange) })
   366→        .eq('id', txn.product_id);
   367→
   368→      const userId = product.user_id as string;
   369→      invalidateCache(cacheKeys.products(userId));
   370→      invalidateCache(cacheKeys.dashboard(userId));
   371→    }
   372→  }
   373→
   374→  const { error } = await supabase.from('transactions').delete().eq('id', id);
   375→  if (error) {
   376→    throw new Error('Failed to delete transaction');
   377→  }
   378→
   379→  return { success: true };
   380→}

   384→export async function getCategories(userId: string) {
   385→  const key = cacheKeys.categories(userId);
   386→  const cached = getCached<{ categories: unknown[] }>(key);
   387→  if (cached) return cached;
   388→
   389→  // Run both queries in parallel
   390→  const [categoriesResult, productCountsResult] = await Promise.all([
   391→    supabase
   392→      .from('categories')
   393→      .select('*')
   394→      .eq('user_id', userId)
   395→      .order('created_at', { ascending: false }),
   396→    supabase
   397→      .from('products')
   398→      .select('category_id')
   399→      .eq('user_id', userId),
   400→  ]);
   401→
   402→  if (categoriesResult.error) {
   403→    throw new Error('Internal server error');
   404→  }
   405→
   406→  if (productCountsResult.error) {
   407→    throw new Error('Internal server error');
   408→  }
   409→
   410→  const countMap: Record<string, number> = {};
   411→  for (const row of productCountsResult.data ?? []) {
   412→    const catId = row.category_id;
   413→    countMap[catId] = (countMap[catId] ?? 0) + 1;
   414→  }
   415→
   416→  const result = {
   417→    categories: (categoriesResult.data ?? []).map((cat) => ({
   418→      ...toCamelCase(cat),
   419→      _count: { products: countMap[cat.id] ?? 0 },
   420→    })),
   421→  };
   422→
   423→  setCache(key, result, cacheTTL.categories);
   424→  return result;
   425→}

   504→export async function getTransactions(userId: string, options?: { type?: string; productId?: string; date?: string; from?: string; to?: string }) {
   505→  const optsKey = `${options?.type || ''}:${options?.productId || ''}:${options?.date || ''}:${options?.from || ''}:${options?.to || ''}`;
   506→  const key = cacheKeys.transactions(userId, optsKey);
   507→  const cached = getCached<{ transactions: unknown[] }>(key);
   508→  if (cached) return cached;
   509→
   510→  let query = supabase
   511→    .from('transactions')
   512→    .select('*, product:products(*)')
   513→    .eq('user_id', userId)
   514→    .order('created_at', { ascending: false });
   515→
   516→  if (options?.type) {
   517→    query = query.eq('type', options.type);
   518→  }
   519→
   520→  if (options?.productId) {
   521→    query = query.eq('product_id', options.productId);
   522→  }
   523→
   524→  if (options?.date) {
   525→    query = query.eq('date', options.date);
   526→  }
   527→
   528→  if (options?.from) {
   529→    query = query.gte('date', options.from);
   530→  }
   531→
   532→  if (options?.to) {
   533→    query = query.lte('date', options.to);
   534→  }
   535→
   536→  const { data, error } = await query;
   537→
   538→  if (error) {
   539→    throw new Error('Failed to fetch transactions');
   540→  }
   541→
   542→  const transactions = data.map((row: Record<string, unknown>) => {
   543→    const { product, ...transactionFields } = row as Record<string, unknown>;
   544→    const camelTransaction = toCamelCase(transactionFields);
   545→    if (product && typeof product === 'object') {
   546→      camelTransaction.product = toCamelCase(product as Record<string, unknown>);
   547→    }
   548→    return camelTransaction;
   549→  });
   550→
   551→  const result = { transactions };
   552→  setCache(key, result, cacheTTL.transactions);
   553→  return result;
   554→}

   556→export async function createTransaction(transactionData: {
   557→  type: string;
   558→  productId: string;
   559→  quantity: number;
   560→  unitPrice?: number;
   561→  totalAmount?: number;
   562→  date?: string;
   563→  userId: string;
   564→}) {
   565→  const { type, productId, quantity, unitPrice, totalAmount, date, userId } = transactionData;
   566→
   567→  if (!type || !productId || !quantity || !userId) {
   568→    throw new Error('Type, productId, quantity, and userId are required');
   569→  }
   570→
   571→  if (!['STOCK_IN', 'STOCK_OUT', 'SELL'].includes(type)) {
   572→    throw new Error('Type must be STOCK_IN, STOCK_OUT, or SELL');
   573→  }
   574→
   575→  // Step 1: Fetch the product
   576→  const { data: product, error: productError } = await supabase
   577→    .from('products')
   578→    .select('*')
   579→    .eq('id', productId)
   580→    .single();
   581→
   582→  if (productError || !product) {
   583→    throw new Error('Product not found');
   584→  }
   585→
   586→  // Step 2: Check stock if STOCK_OUT or SELL
   587→  if ((type === 'STOCK_OUT' || type === 'SELL') && product.quantity < quantity) {
   588→    throw new Error('Insufficient stock');
   589→  }
   590→
   591→  // Step 3: Insert the transaction
   592→  const transactionId = generateId();
   593→  const quantityChange = type === 'STOCK_IN' ? quantity : -quantity;
   594→
   595→  const { data: newTransaction, error: insertError } = await supabase
   596→    .from('transactions')
   597→    .insert(
   598→      toSnakeCase({
   599→        id: transactionId,
   600→        type,
   601→        productId,
   602→        quantity,
   603→        unitPrice: unitPrice ?? 0,
   604→        totalAmount: totalAmount ?? 0,
   605→        date: date || new Date().toISOString().split('T')[0],
   606→        userId,
   607→      })
   608→    )
   609→    .select('*, product:products(*)')
   610→    .single();
   611→
   612→  if (insertError) {
   613→    throw new Error('Failed to create transaction');
   614→  }
   615→
   616→  // Step 4: Update the product quantity
   617→  const { error: updateError } = await supabase
   618→    .from('products')
   619→    .update({ quantity: product.quantity + quantityChange })
   620→    .eq('id', productId);
   621→
   622→  if (updateError) {
   623→    await supabase.from('transactions').delete().eq('id', transactionId);
   624→    throw new Error('Failed to update product stock');
   625→  }
   626→
   627→  const camelTransaction = toCamelCase(newTransaction as Record<string, unknown>);
   628→  const { product: productData, ...transactionFields } = camelTransaction as Record<string, unknown>;
   629→  if (productData && typeof productData === 'object') {
   630→    transactionFields.product = toCamelCase(productData as Record<string, unknown>);
   631→  }
   632→
   633→  // Invalidate all relevant caches
   634→  invalidateCache(cacheKeys.transactions(userId));
   635→  invalidateCache(cacheKeys.products(userId));
   636→  invalidateCache(cacheKeys.dashboard(userId));
   637→
   638→  return { transaction: transactionFields };
   639→}

   643→function getDateRange(period: string, date?: string, from?: string, to?: string): { startDate: string; endDate: string } {
   644→  const now = new Date();
   645→  let startDate: string;
   646→  let endDate: string;
   647→
   648→  if (from && to) {
   649→    startDate = from;
   650→    endDate = to;
   651→  } else if (date) {
   652→    startDate = date;
   653→    endDate = date;
   654→  } else {
   655→    const todayStr = now.toISOString().split('T')[0];
   656→    if (period === 'today') {
   657→      startDate = todayStr;
   658→      endDate = todayStr;
   659→    } else if (period === 'week') {
   660→      const weekStart = new Date(now);
   661→      const day = weekStart.getDay();
   662→      const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1);
   663→      weekStart.setDate(diff);
   664→      startDate = weekStart.toISOString().split('T')[0];
   665→      endDate = todayStr;
   666→    } else if (period === 'month') {
   667→      startDate = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`;
   668→      endDate = todayStr;
   669→    } else {
   670→      startDate = todayStr;
   671→      endDate = todayStr;
   672→    }
   673→  }
   674→
   675→  return { startDate, endDate };
   676→}

   678→export async function getExpenses(userId: string, options?: { period?: string; date?: string; from?: string; to?: string; category?: string }) {
   679→  const period = options?.period || 'today';
   680→  const optsKey = `${period}:${options?.date || ''}:${options?.from || ''}:${options?.to || ''}:${options?.category || ''}`;
   681→  const key = cacheKeys.expenses(userId, optsKey);
   682→  const cached = getCached<unknown>(key);
   683→  if (cached) return cached;
   684→
   685→  const { startDate, endDate } = getDateRange(period, options?.date, options?.from, options?.to);
   686→
   687→  let query = supabase
   688→    .from('expenses')
   689→    .select('*')
   690→    .eq('user_id', userId)
   691→    .gte('date', startDate)
   692→    .lte('date', endDate)
   693→    .order('date', { ascending: false });
   694→
   695→  if (options?.category) {
   696→    query = query.eq('category', options.category);
   697→  }
   698→
   699→  const { data, error } = await query;
   700→
   701→  if (error) {
   702→    throw new Error('Failed to fetch expenses');
   703→  }
   704→
   705→  const expenses = (data || []).map(toCamelCase) as Record<string, unknown>[];
   706→
   707→  const totalExpense = expenses.reduce((sum: number, e) => sum + ((e.amount as number) || 0), 0);
   708→
   709→  const byCategory = expenses.reduce((acc: Record<string, number>, e: Record<string, unknown>) => {
   710→    const cat = (e.category as string) || 'other';
   711→    if (!acc[cat]) acc[cat] = 0;
   712→    acc[cat] += (e.amount as number) || 0;
   713→    return acc;

   715→
   716→  const result = {
   717→    expenses,
   718→    summary: {
   719→      totalExpense,
   720→      byCategory,
   721→      count: expenses.length,
   722→    },
   723→  };

  1313→function deepCamelCase(obj: unknown): unknown {
  1314→  if (obj === null || obj === undefined) return obj;
  1315→  if (Array.isArray(obj)) return obj.map(deepCamelCase);
  1316→  if (typeof obj === 'object') {
  1317→    const result: Record<string, unknown> = {};
  1318→    for (const [key, value] of Object.entries(obj as Record<string, unknown>)) {
  1319→      const camelKey = key.replace(/_([a-z])/g, (_, letter) => letter.toUpperCase());
  1320→      result[camelKey] = deepCamelCase(value);
  1321→    }
  1322→    return result;
  1323→  }
  1324→  return obj;
  1325→}
