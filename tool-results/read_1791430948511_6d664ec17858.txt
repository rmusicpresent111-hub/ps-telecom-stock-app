   214→  const [isOffline, setIsOffline] = useState(!checkOnline());
   215→  const isFetchingRef = useRef(false);
   216→
   217→  // Listen for online/offline status changes
   218→  const fetchDashboard = useCallback(async (isBackground = false) => {
   219→    if (!user?.id || isFetchingRef.current) return;
   220→
   221→    isFetchingRef.current = true;
   222→    try {
   223→      if (!isBackground) setLoading(true);
   224→      const data = await getDashboard(user.id) as Record<string, unknown>;
   225→      setStats(data.stats as DashboardStats | null);
   226→      setLowStockProducts((data.lowStockProducts as Product[]) || []);
   227→      setSaleOverview((data.saleOverview as SaleOverviewItem[]) || []);
   228→      setStockOverview((data.stockOverview as StockOverviewItem[]) || []);
   229→      if ((data.categories as Category[] | undefined)?.length) {
   230→        setCategories(data.categories as Category[]);
   231→      }
   232→
   233→      // Calculate today's profit from transactions
   234→      // Profit = sellingPrice - purchasePrice per unit sold
   235→      const today = new Date().toISOString().split('T')[0];
   236→      const { getTransactionsOffline: getTxns, getProductsOffline: getProds } = await import('@/lib/offline-service');
   237→      const [todayTxns, allProducts] = await Promise.all([
   238→        getTxns(user.id, { from: today, to: today }),
   239→        getProds(user.id),
   240→      ]);
   241→      const sellTxns = (todayTxns || []).filter((tx: { type: string }) => tx.type === 'SELL');
   242→      const prodMap = new Map((allProducts || []).map((p: Product) => [p.id, p]));
   243→      let profit = 0;
   244→      for (const tx of sellTxns) {
   245→        const prod = prodMap.get(tx.productId);
   246→        const cost = (tx.quantity || 0) * (prod?.purchasePrice ?? 0);
   247→        profit += (tx.totalAmount || 0) - cost;
   248→      }
   249→      setTodayProfit(profit);
   250→    } catch {
   251→      if (!isBackground) toast.error(t('error', language));
   252→    } finally {
   253→      setLoading(false);
   254→      isFetchingRef.current = false;
   255→    }
   256→  }, [user?.id]);
   257→
   258→  // Initial load
   259→  useEffect(() => {
   260→    fetchDashboard();
   261→  }, [fetchDashboard]);
   262→
   263→  // ✅ Listen for online/offline changes
   264→  useEffect(() => {
   265→    const handleOnlineStatus = (e: Event) => {
   266→      const online = e instanceof CustomEvent ? e.detail : navigator.onLine;
   267→      setIsOffline(!online);
   268→      if (online) {
   269→        // Back online - refresh data
   270→        invalidateCache(cacheKeys.dashboard(user?.id || ''));
   271→        fetchDashboard(true);
   272→      }
   273→    };
   274→    window.addEventListener('app:online-status', handleOnlineStatus);
   275→    window.addEventListener('online', handleOnlineStatus);
   276→    window.addEventListener('offline', handleOnlineStatus);
   277→    return () => {
   278→      window.removeEventListener('app:online-status', handleOnlineStatus);
   279→      window.removeEventListener('online', handleOnlineStatus);
   280→      window.removeEventListener('offline', handleOnlineStatus);
   281→    };
   282→  }, [user?.id, fetchDashboard]);

   284→  // ✅ Background refresh on window focus (stale-while-revalidate)
