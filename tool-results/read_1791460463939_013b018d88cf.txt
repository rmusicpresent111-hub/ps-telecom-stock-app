    35→export function formatMoney(n: number): string {
    36→  const num = Number(n) || 0;
    37→  return num.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    38→}
    39→
    40→export function formatDate(iso: string): string {
    41→  const d = new Date(`${iso.slice(0, 10)}T00:00:00`);
    42→  if (Number.isNaN(d.getTime())) return iso;
    43→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
    44→}

    46→/**
    47→ * Normalize a mobile number for wa.me: digits only, ensure country code.
    48→ * 10-digit Indian numbers get 91 prefixed.
    49→ */
    50→export function normalizeMobile(mobile: string): string {
    51→  let digits = (mobile || '').replace(/\D/g, '');
    52→  if (digits.length === 11 && digits.startsWith('0')) digits = digits.slice(1);
    53→  if (digits.length === 10) digits = `91${digits}`;
    54→  return digits;
    55→}

    57→/** Indian-format number to words, e.g. 1250.50 → "One Thousand Two Hundred Fifty Rupees and 50 Paise" */
    58→export function amountInWords(amount: number): string {
    59→  const rupees = Math.floor(amount);
    60→  const paise = Math.round((amount - rupees) * 100);
    61→  const rupeeWords = rupees === 0 ? 'Zero' : numberToWordsIndian(rupees);
    62→  let out = `${rupeeWords} Rupees`;
    63→  if (paise > 0) out += ` and ${numberToWordsIndian(paise)} Paise`;
    64→  return `${out} Only`;
    65→}
    66→
    67→function numberToWordsIndian(n: number): string {
    68→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten',
    69→    'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen'];
    70→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety'];
    71→
    72→  const twoDigits = (num: number): string => {
    73→    if (num < 20) return ones[num];
    74→    return `${tens[Math.floor(num / 10)]}${num % 10 ? ` ${ones[num % 10]}` : ''}`;
    75→  };
    76→  const threeDigits = (num: number): string => {
    77→    const hundreds = Math.floor(num / 100);
    78→    const rest = num % 100;
    79→    return `${hundreds ? `${ones[hundreds]} Hundred${rest ? ' ' : ''}` : ''}${rest ? twoDigits(rest) : ''}`;
    80→  };
    81→
    82→  if (n === 0) return 'Zero';
    83→  const parts: string[] = [];
    84→  const crore = Math.floor(n / 10000000);
    85→  const lakh = Math.floor((n % 10000000) / 100000);
    86→  const thousand = Math.floor((n % 100000) / 1000);
    87→  const rest = n % 1000;
    88→  if (crore) parts.push(`${numberToWordsIndian(crore)} Crore`);
    89→  if (lakh) parts.push(`${twoDigits(lakh)} Lakh`);
    90→  if (thousand) parts.push(`${twoDigits(thousand)} Thousand`);
    91→  if (rest) parts.push(threeDigits(rest));
    92→  return parts.join(' ');
    93→}

    95→/** First letters of up to 2 words — monogram initials, e.g. "PS TELECOM" → "PS". */
    96→export function shopInitials(name: string): string {
    97→  const words = (name || '').trim().split(/\s+/).filter(Boolean);
    98→  if (!words.length) return 'PS';
    99→  if (words.length === 1) return words[0].slice(0, 2).toUpperCase();
   100→  return ((words[0][0] || '') + (words[1][0] || '')).toUpperCase();
   101→}
   102→
   103→const lerp = (a: number, b: number, t: number) => Math.round(a + (b - a) * t);
   104→
   105→/** Paints a vertical gold gradient band by stacking thin interpolated strips. */
   106→function goldGradient(doc: jsPDF, x: number, y: number, w: number, h: number): void {
   107→  const steps = Math.max(10, Math.min(64, Math.round(h)));
   108→  const stepH = h / steps;
   109→  for (let i = 0; i < steps; i++) {
   110→    const t = i / (steps - 1);
   111→    doc.setFillColor(
   112→      lerp(GOLD_LIGHT[0], GOLD_DEEP[0], t),
   113→      lerp(GOLD_LIGHT[1], GOLD_DEEP[1], t),
   114→      lerp(GOLD_LIGHT[2], GOLD_DEEP[2], t),
   115→    );
   116→    doc.rect(x, y + stepH * i, w, stepH + 0.4, 'F');
   117→  }
   118→}
   119→
   120→/** Small gold diamond ornament (two triangles) centered at (cx, cy). */
   121→function diamond(doc: jsPDF, cx: number, cy: number, r: number): void {
   122→  doc.setFillColor(...GOLD);
   123→  doc.triangle(cx - r, cy, cx, cy - r, cx, cy, 'F');
   124→  doc.triangle(cx, cy - r, cx + r, cy, cx, cy, 'F');
   125→  doc.triangle(cx - r, cy, cx, cy + r, cx, cy, 'F');
   126→  doc.triangle(cx, cy + r, cx + r, cy, cx, cy, 'F');
   127→}

   145→/**
   146→ * Draws a premium A5 invoice PDF for the bill.
   147→ * shopSnapshot (frozen at bill time) drives the header; signature/QR come
   148→ * from current billing settings.
   149→ */
   150→export async function generateBillPdf(
   151→  bill: Bill,
   152→  images: BillPdfImages = {},
   153→): Promise<GeneratedPdf> {
   154→  const doc = new jsPDF({ unit: 'pt', format: 'a5', orientation: 'portrait' });
   155→  const W = doc.internal.pageSize.getWidth();   // 419.5 pt
   156→  const H = doc.internal.pageSize.getHeight();  // 595.3 pt
   157→  const M = 26; // margin
   158→
   159→  const shop = bill.shopSnapshot || { name: 'PS TELECOM', address: '', phone: '', gstNumber: '' };
   160→  const shopName = shop.name || 'PS TELECOM';
   161→  const proprietor = shop.proprietorName ?? images.proprietorName ?? PROPRIETOR_FALLBACK;
   162→
   163→  // ---- Page frame (double gold certificate border) ----
   164→  doc.setDrawColor(...GOLD);
   165→  doc.setLineWidth(1.1);
   166→  doc.rect(8, 8, W - 16, H - 16, 'S');
   167→  doc.setDrawColor(...GOLD_LIGHT);
   168→  doc.setLineWidth(0.4);
   169→  doc.rect(12, 12, W - 24, H - 24, 'S');
   170→
   171→  // ---- Corner ornaments ----
   172→  const corner = (cx: number, cy: number, sx: number, sy: number) => {
   173→    doc.setFillColor(...GOLD);
   174→    doc.circle(cx, cy, 1.6, 'F');
   175→    doc.setDrawColor(...GOLD);
   176→    doc.setLineWidth(0.7);
   177→    doc.line(cx + sx * 5, cy, cx + sx * 14, cy);
   178→    doc.line(cx, cy + sy * 5, cx, cy + sy * 14);
   179→  };
   180→  corner(18, 18, 1, 1);
   181→  corner(W - 18, 18, -1, 1);
   182→  corner(18, H - 18, 1, -1);
   183→  corner(W - 18, H - 18, -1, -1);
   184→
   185→  // ---- Watermark (behind everything on the body) ----
   186→  doc.setFont('times', 'bold');
   187→  doc.setFontSize(42);
   188→  doc.setTextColor(...WATERMARK);
   189→  doc.text(shopName, W / 2, H * 0.58, { align: 'center', angle: 28 });
   190→
   191→  // ---- Header band ----
   192→  const HEAD_H = 116;
   193→  doc.setFillColor(...DARK);
   194→  doc.rect(0, 0, W, HEAD_H, 'F');
   195→  // subtle darker vignette strip
   196→  doc.setFillColor(...DARK2);
   197→  doc.rect(0, 0, W, 22, 'F');
   198→  goldGradient(doc, 0, HEAD_H, W, 4);
   199→  doc.setFillColor(...GOLD_DEEP);
   200→  doc.rect(0, HEAD_H + 4, W, 0.8, 'F');
   201→
   202→  // Monogram medallion
   203→  const mcx = M + 17, mcy = 32, mr = 16;
   204→  doc.setFillColor(...GOLD);
   205→  doc.circle(mcx, mcy, mr, 'F');
   206→  doc.setDrawColor(...GOLD_LIGHT);
   207→  doc.setLineWidth(0.9);
   208→  doc.circle(mcx, mcy, mr - 2.4, 'S');
   209→  doc.setTextColor(...DARK);
   210→  doc.setFont('times', 'bold');
   211→  doc.setFontSize(13);
   212→  doc.text(shopInitials(shopName), mcx, mcy + 4.5, { align: 'center' });
   213→
   214→  // Shop name — serif, letter-spaced, auto-shrink to fit
   215→  const nameX = M + 44;
   216→  const nameMaxW = W - nameX - M - 84;
   217→  let nameSize = 19;
   218→  doc.setFont('times', 'bold');
   219→  doc.setTextColor(255, 255, 255);
   220→  do {
   221→    doc.setFontSize(nameSize);
   222→    doc.setCharSpace(1.1);
   223→    if (doc.getTextWidth(shopName) <= nameMaxW) break;
   224→    doc.setCharSpace(0);
   225→    nameSize -= 0.5;
   226→  } while (nameSize > 11);
   227→  doc.text(shopName, nameX, 36);
   228→  doc.setCharSpace(0);
   229→
   230→  // Proprietor line
   231→  let headerY = 50;
   232→  if (proprietor) {
   233→    doc.setFont('times', 'italic');
   234→    doc.setFontSize(8.8);
   235→    doc.setTextColor(238, 208, 140);
   236→    doc.text(`Proprietor: ${proprietor}`, nameX, headerY);
   237→    headerY += 12;
   238→  }
   239→
   240→  // Address / contact
   241→  doc.setFont('helvetica', 'normal');
   242→  doc.setFontSize(7.8);
   243→  doc.setTextColor(214, 216, 226);
   244→  if (shop.address) {
   245→    const addressLines = doc.splitTextToSize(shop.address, W - nameX - M - 84) as string[];
   246→    doc.text(addressLines, nameX, headerY);
   247→    headerY += addressLines.length * 9.5;
   248→  }
   249→  const contactBits: string[] = [];
   250→  if (shop.phone) contactBits.push(`Ph: ${shop.phone}`);
   251→  if (shop.gstNumber) contactBits.push(`GSTIN: ${shop.gstNumber}`);
   252→  if (contactBits.length) doc.text(contactBits.join('  |  '), nameX, headerY);
   253→
   254→  // INVOICE plate (right, gold gradient with engraved border)
   255→  const plateW = 78, plateH = 20, plateX = W - M - plateW, plateY = 16;
   256→  goldGradient(doc, plateX, plateY, plateW, plateH);
   257→  doc.setDrawColor(...DARK);
   258→  doc.setLineWidth(0.9);
   259→  doc.rect(plateX + 2, plateY + 2, plateW - 4, plateH - 4, 'S');
   260→  doc.setTextColor(...DARK);
   261→  doc.setFont('helvetica', 'bold');
   262→  doc.setFontSize(8.6);
   263→  doc.text(bill.gstEnabled ? 'TAX INVOICE' : 'INVOICE', plateX + plateW / 2, plateY + 13.5, { align: 'center' });
   264→
   265→  // Bill meta (right column)
   266→  doc.setTextColor(226, 228, 238);
   267→  doc.setFont('helvetica', 'normal');
   268→  doc.setFontSize(8.4);
   269→  doc.text(`Bill No: ${bill.billNumber}`, W - M, 52, { align: 'right' });
   270→  doc.text(`Date: ${formatDate(bill.date)}`, W - M, 63.5, { align: 'right' });
   271→
   272→  // ---- Bill To (with gold accent bar) ----
   273→  let y = 142;
   274→  doc.setFillColor(...GOLD);
   275→  doc.roundedRect(M, y - 14, 2.6, 44, 1.2, 1.2, 'F');
   276→  doc.setDrawColor(226, 219, 200);
   277→  doc.setLineWidth(0.7);
   278→  doc.roundedRect(M + 6, y - 14, W - M * 2 - 6, 44, 6, 6, 'S');
   279→  doc.setTextColor(...GRAY);
   280→  doc.setFontSize(7.4);
   281→  doc.setFont('helvetica', 'bold');
   282→  doc.setCharSpace(1);
   283→  doc.text('BILL TO', M + 16, y - 1);
   284→  doc.setCharSpace(0);
   285→  doc.setTextColor(...DARK);
   286→  doc.setFont('times', 'bold');
   287→  doc.setFontSize(12);
   288→  doc.text(bill.customerName || 'Walk-in Customer', M + 16, y + 14);
   289→  if (bill.customerMobile) {
   290→    doc.setFont('helvetica', 'normal');
   291→    doc.setFontSize(9);
   292→    doc.setTextColor(84, 88, 100);
   293→    doc.text(`Mob: ${bill.customerMobile}`, W - M - 14, y + 14, { align: 'right' });
   294→  }
   295→
   296→  y += 50;
   297→
   298→  // ---- Items table ----
   299→  const colItem = M + 10;
   300→  const colQty = W * 0.52;
   301→  const colRate = W * 0.66;
   302→  const colAmount = W - M - 12;
   303→  const tableTop = y;
   304→
   305→  doc.setFillColor(...DARK);
   306→  doc.rect(M, tableTop, W - M * 2, 23, 'F');
   307→  doc.setFillColor(...GOLD);
   308→  doc.rect(M, tableTop + 23, W - M * 2, 1.4, 'F');
   309→  doc.setTextColor(...GOLD_LIGHT);
   310→  doc.setFontSize(8.2);
   311→  doc.setFont('helvetica', 'bold');
   312→  doc.setCharSpace(0.6);
   313→  doc.text('ITEM', colItem, tableTop + 15.5);
   314→  doc.text('QTY', colQty, tableTop + 15.5, { align: 'center' });
   315→  doc.text('RATE', colRate, tableTop + 15.5, { align: 'right' });
   316→  doc.text('AMOUNT', colAmount, tableTop + 15.5, { align: 'right' });
   317→  doc.setCharSpace(0);
   318→
   319→  let rowY = tableTop + 24.4;
   320→  doc.setFont('helvetica', 'normal');
   321→  doc.setTextColor(...DARK);
   322→  for (const item of bill.items) {
   323→    const nameLines = doc.splitTextToSize(item.name, colQty - colItem - 14) as string[];
   324→    const rowHeight = Math.max(24, nameLines.length * 11 + 10);
   325→
   326→    if (bill.items.indexOf(item) % 2 === 1) {
   327→      doc.setFillColor(...IVORY);
   328→      doc.rect(M, rowY, W - M * 2, rowHeight, 'F');
   329→    }
   330→
   331→    doc.setFontSize(9.4);
   332→    doc.text(nameLines, colItem, rowY + 15);
   333→    doc.text(String(item.quantity), colQty, rowY + 15, { align: 'center' });
   334→    doc.text(`Rs. ${formatMoney(item.unitPrice)}`, colRate, rowY + 15, { align: 'right' });
   335→    doc.setFont('times', 'bold');
   336→    doc.setFontSize(10);
   337→    doc.text(`Rs. ${formatMoney(item.total)}`, colAmount, rowY + 15, { align: 'right' });
   338→    doc.setFont('helvetica', 'normal');
   339→    doc.setFontSize(9.4);
   340→
   341→    rowY += rowHeight;
   342→    doc.setDrawColor(228, 224, 214);
   343→    doc.setLineWidth(0.5);
   344→    doc.line(M, rowY, W - M, rowY);
   345→  }
   346→
   347→  // ---- Totals ----
   348→  y = rowY + 14;
   349→  const totalsX = W * 0.5;
   350→  const totalsLabelX = totalsX + 12;
   351→  const totalsValueX = W - M - 12;
   352→  const line = (label: string, value: string, opts?: { bold?: boolean; color?: RGB }) => {
   353→    doc.setFont('helvetica', opts?.bold ? 'bold' : 'normal');
   354→    doc.setFontSize(opts?.bold ? 10.5 : 9);
   355→    if (opts?.color) doc.setTextColor(...opts.color); else doc.setTextColor(70, 74, 86);
   356→    doc.text(label, totalsLabelX, y);
   357→    doc.text(value, totalsValueX, y, { align: 'right' });
   358→    y += opts?.bold ? 18 : 15;
   359→  };
   360→
   361→  line('Subtotal', `Rs. ${formatMoney(bill.subtotal)}`);
   362→  if (bill.discountAmount > 0) {
   363→    const pct = bill.discountType === 'percent' ? ` (${bill.discountValue}%)` : '';
   364→    line(`Discount${pct}`, `- Rs. ${formatMoney(bill.discountAmount)}`, { color: GREEN });
   365→  }
   366→  if (bill.gstEnabled && bill.gstAmount > 0) {
   367→    line(`GST (${bill.gstRate}%)`, `Rs. ${formatMoney(bill.gstAmount)}`);
   368→  }
   369→
   370→  // Grand total — engraved gold plate
   371→  y += 2;
   372→  const bandH = 28;
   373→  goldGradient(doc, totalsX, y, W - M - totalsX, bandH);
   374→  doc.setDrawColor(...DARK);
   375→  doc.setLineWidth(0.9);
   376→  doc.rect(totalsX + 2, y + 2, W - M - totalsX - 4, bandH - 4, 'S');
   377→  doc.setTextColor(...DARK);
   378→  doc.setFont('times', 'bold');
   379→  doc.setFontSize(11);
   380→  doc.text('GRAND TOTAL', totalsLabelX, y + 18.5);
   381→  doc.setFontSize(13.5);
   382→  doc.text(`Rs. ${formatMoney(bill.total)}`, totalsValueX, y + 18.5, { align: 'right' });
   383→  y += bandH + 8;
   384→
   385→  if (bill.paidAmount > 0 || bill.dueAmount > 0) {
   386→    line('Paid', `Rs. ${formatMoney(bill.paidAmount)}`, { color: GREEN });
   387→    if (bill.dueAmount > 0) {
   388→      line('Balance Due', `Rs. ${formatMoney(bill.dueAmount)}`, { color: RED, bold: true });
   389→    }
   390→  }
   391→  const pmLabel: Record<string, string> = { cash: 'Cash', upi: 'UPI', card: 'Card', due: 'Due (Credit)' };
   392→  line('Payment Method', pmLabel[bill.paymentMethod] || bill.paymentMethod);
   393→
   394→  // Amount in words (left column, next to totals)
   395→  const wordsY = tableTop + 22 + Math.max(bill.items.length * 26, 30) + 12;
   396→  doc.setFont('times', 'bolditalic');
   397→  doc.setFontSize(8.4);
   398→  doc.setTextColor(...GRAY);
   399→  const wordsLines = doc.splitTextToSize(`In Words: ${amountInWords(bill.total)}`, W * 0.46) as string[];
   400→  doc.text(wordsLines, M + 2, Math.min(wordsY, rowY + 24));
   401→
   402→  if (bill.note) {
   403→    doc.setFont('helvetica', 'italic');
   404→    doc.setFontSize(8.5);
   405→    doc.setTextColor(90, 94, 106);
   406→    const noteLines = doc.splitTextToSize(`Note: ${bill.note}`, W - M * 2) as string[];
   407→    doc.text(noteLines, M, y + 6);
   408→    y += noteLines.length * 11 + 8;
   409→  }
   410→
   411→  // ---- QR + Signature row (pinned near bottom) ----
   412→  const bottomZone = H - 104;
   413→  if (y < bottomZone - 8) y = bottomZone;
   414→
   415→  if (images.qrCodeDataUrl) {
   416→    try {
   417→      const qrSize = 62;
   418→      doc.setFillColor(255, 255, 255);
   419→      doc.setDrawColor(...GOLD);
   420→      doc.setLineWidth(0.8);
   421→      doc.roundedRect(M - 2, y - 2, qrSize + 4, qrSize + 4, 4, 4, 'FD');
   422→      doc.addImage(images.qrCodeDataUrl, 'PNG', M, y, qrSize, qrSize);
   423→      doc.setFont('helvetica', 'normal');
   424→      doc.setFontSize(7);
   425→      doc.setTextColor(...GRAY);
   426→      doc.text(images.upiId ? `Scan to pay  |  UPI: ${images.upiId}` : 'Scan to pay', M + qrSize / 2, y + qrSize + 12, { align: 'center' });
   427→    } catch {
   428→      // invalid image data — skip QR silently
   429→    }
   430→  }
   431→
   432→  // Signature block
   433→  if (images.signatureDataUrl) {
   434→    try {
   435→      doc.addImage(images.signatureDataUrl, 'PNG', W - M - 92, y - 14, 92, 32);
   436→    } catch {
   437→      // invalid image data — skip signature silently
   438→    }
   439→  }
   440→  doc.setDrawColor(140, 144, 156);
   441→  doc.setLineWidth(0.6);
   442→  doc.line(W - M - 102, y + 26, W - M, y + 26);
   443→  doc.setFont('helvetica', 'normal');
   444→  doc.setFontSize(8);
   445→  doc.setTextColor(...GRAY);
   446→  doc.text(`For ${shopName}`, W - M, y + 12, { align: 'right' });
   447→  if (proprietor) {
   448→    doc.setFont('times', 'bold');
   449→    doc.setFontSize(9.5);
   450→    doc.setTextColor(...DARK);
   451→    doc.text(proprietor, W - M, y + 38, { align: 'right' });
   452→    doc.setFont('helvetica', 'normal');
   453→    doc.setFontSize(7.2);
   454→    doc.setTextColor(...GRAY);
   455→    doc.text('Proprietor', W - M, y + 47, { align: 'right' });
   456→  }
   457→
   458→  // ---- Footer: ornament divider + thank you + terms ----
   459→  const divY = H - 44;
   460→  doc.setDrawColor(...GOLD);
   461→  doc.setLineWidth(0.6);
   462→  doc.line(M + 4, divY, W / 2 - 9, divY);
   463→  doc.line(W / 2 + 9, divY, W - M - 4, divY);
   464→  diamond(doc, W / 2, divY, 3);
   465→
   466→  doc.setFont('times', 'bolditalic');
   467→  doc.setFontSize(10);
   468→  doc.setTextColor(...DARK);
   469→  const thanks = images.thankYouNote || 'Thank you for your business!';
   470→  doc.text(thanks, W / 2, H - 31, { align: 'center' });
   471→
   472→  doc.setFont('helvetica', 'normal');
   473→  doc.setFontSize(6.6);
   474→  doc.setTextColor(150, 154, 166);
   475→  let footer = `Generated on ${new Date().toLocaleString('en-GB')} — PS TELECOM App`;
   476→  if (images.termsText) {
   477→    footer = `${images.termsText}  |  ${footer}`;
   478→  }
   479→  doc.text(doc.splitTextToSize(footer, W - M * 2 - 20) as string[], W / 2, H - 19.5, { align: 'center' });
   480→
   481→  const blob = doc.output('blob');
   482→  const fileName = `${shopName.replace(/[^a-zA-Z0-9]+/g, '-')}-${bill.billNumber}.pdf`;
   483→  return { blob, fileName };
   484→}

   488→export function downloadBlob(blob: Blob, fileName: string): void {
   489→  const url = URL.createObjectURL(blob);
   490→  const a = document.createElement('a');

   492→  a.download = fileName;
   493→  document.body.appendChild(a);
   494→  a.click();
   495→  a.remove();
   496→  setTimeout(() => URL.revokeObjectURL(url), 4000);

   510→/**
   511→ * Sync native-app detection. @capacitor/core registers a global
   512→ * `window.Capacitor` in BOTH the APK WebView and the web build (it's loaded
   513→ * by capacitor-init), so the tell is `isNativePlatform()`: false in normal
   514→ * browsers, true inside the APK. Deciding this WITHOUT awaiting anything
   515→ * matters — waiting would burn the tap's transient user activation and
   516→ * mobile browsers would reject navigator.share with NotAllowedError → the
   517→ * PDF silently never reaches WhatsApp.
   518→ */
   519→function isNativeApp(): boolean {
   520→  if (typeof window === 'undefined') return false;
   521→  const cap = (window as { Capacitor?: { isNativePlatform?: () => boolean } }).Capacitor;
   522→  return !!cap?.isNativePlatform?.();
   523→}

   525→/**
   526→ * Blob → base64 (no data-url prefix) — needed by Capacitor Filesystem.writeFile.
   527→ */
   528→function blobToBase64(blob: Blob): Promise<string> {
   529→  return new Promise((resolve, reject) => {
   530→    const reader = new FileReader();
   531→    reader.onload = () => resolve(String(reader.result).split(',')[1] || '');
   532→    reader.onerror = () => reject(new Error('Could not read the PDF file'));
   533→    reader.readAsDataURL(blob);
   534→  });
   535→}

   537→export type FileShareResult =
   538→  | 'shared'      // the real PDF file was handed to the OS share sheet
   539→  | 'cancelled'   // user closed the share sheet without sending
   540→  | 'unsupported' // this browser cannot share files (share sheet unavailable)
   541→  | 'blocked';    // share rejected (NotAllowedError / SecurityError / …)
   542→
   543→/**
   544→ * Hand the ACTUAL PDF file to the OS share sheet, choosing the right
   545→ * transport without ever delaying the first share attempt:
   546→ *
   547→ *  1. Mobile/desktop browser — `navigator.share({ files })` is the FIRST
   548→ *     thing that runs, before ANY await, so the call happens while the
   549→ *     user's tap activation is still fresh. This is what makes the PDF
   550→ *     actually attach in WhatsApp on phones.
   551→ *  2. Capacitor APK — Android WebView has no navigator.share; we write the
   552→ *     PDF to the app cache and hand the real file to the Android share
   553→ *     sheet via @capacitor/share.
   554→ */
   555→export async function sharePdfFile(blob: Blob, fileName: string, text: string): Promise<FileShareResult> {
   556→  const file = new File([blob], fileName, { type: 'application/pdf' });
   557→  const nav = navigator as Navigator;
   558→
   559→  // ---- 1. Browser: Web Share API Level 2 (files) — called instantly ----
   560→  if (!isNativeApp() && typeof nav.share === 'function') {
   561→    try {
   562→      await nav.share({ files: [file], title: fileName, text });
   563→      return 'shared';
   564→    } catch (err) {
   565→      const name = (err as Error)?.name || '';
   566→      if (name === 'AbortError') return 'cancelled'; // user closed the sheet
   567→      // TypeError/NotSupportedError → this browser cannot share FILES
   568→      if (name === 'TypeError' || name === 'NotSupportedError') return 'unsupported';
   569→      return 'blocked'; // NotAllowedError / SecurityError (e.g. framed page)
   570→    }
   571→  }
   572→
   573→  // ---- 2. Native APK (WebView: no navigator.share) ----
   574→  if (isNativeApp()) {
   575→    try {
   576→      const [{ Directory, Filesystem }, { Share }] = await Promise.all([
   577→        import('@capacitor/filesystem'),
   578→        import('@capacitor/share'),
   579→      ]);
   580→      const base64Data = await blobToBase64(blob);
   581→      const written = await Filesystem.writeFile({
   582→        path: fileName,
   583→        data: base64Data,
   584→        directory: Directory.Cache,
   585→        recursive: true,
   586→      });
   587→      await Share.share({
   588→        title: fileName,
   589→        text,
   590→        files: [written.uri],
   591→        dialogTitle: 'Send bill via…',
   592→      });
   593→      return 'shared';
   594→    } catch (err) {
   595→      const msg = String((err as Error)?.message || '').toLowerCase();
   596→      if (msg.includes('cancel')) return 'cancelled';
   597→      return 'blocked';
   598→    }
   599→  }
   600→
   601→  return 'unsupported';
   602→}

   604→/** Build the WhatsApp text summary for a bill. */
   605→export function buildBillMessage(bill: Bill, opts?: { withPdfNote?: boolean }): string {
   606→  const shop = bill.shopSnapshot || { name: 'PS TELECOM', address: '', phone: '', gstNumber: '' };
   607→  const shopName = shop.name || 'PS TELECOM';
   608→  const proprietor = shop.proprietorName ?? PROPRIETOR_FALLBACK;
   609→  const lines: string[] = [];
   610→
   611→  lines.push(`*${shopName}*`);
   612→  if (proprietor) lines.push(`_${proprietor}_`);
   613→  if (shop.address) lines.push(shop.address.replace(/\n+/g, ', '));
   614→  const contact: string[] = [];
   615→  if (shop.phone) contact.push(`Ph: ${shop.phone}`);
   616→  if (shop.gstNumber) contact.push(`GSTIN: ${shop.gstNumber}`);
   617→  if (contact.length) lines.push(contact.join(' | '));
   618→
   619→  lines.push('━━━━━━━━━━━━━━━━━━');
   620→  // NOTE: no astral-plane (4-byte UTF-8) emoji here — some network proxies
   621→  // mangle them into U+FFFD. BMP symbols (━ • — ₹ ⚠) travel safely.
   622→  lines.push(`*${bill.gstEnabled ? 'TAX INVOICE' : 'INVOICE'}*`);
   623→  lines.push(`Bill No: ${bill.billNumber}`);
   624→  lines.push(`Date: ${formatDate(bill.date)}`);
   625→  lines.push(`Customer: ${bill.customerName || 'Walk-in Customer'}${bill.customerMobile ? ` (${bill.customerMobile})` : ''}`);
   626→  lines.push('━━━━━━━━━━━━━━━━━━');
   627→  for (const item of bill.items) {
   628→    lines.push(`• ${item.name} — ${item.quantity} × ₹${formatMoney(item.unitPrice)} = *₹${formatMoney(item.total)}*`);
   629→  }
   630→  lines.push('━━━━━━━━━━━━━━━━━━');
   631→  lines.push(`Subtotal: ₹${formatMoney(bill.subtotal)}`);
   632→  if (bill.discountAmount > 0) {
   633→    const pct = bill.discountType === 'percent' ? ` (${bill.discountValue}%)` : '';
   634→    lines.push(`Discount${pct}: -₹${formatMoney(bill.discountAmount)}`);
   635→  }
   636→  if (bill.gstEnabled && bill.gstAmount > 0) lines.push(`GST (${bill.gstRate}%): ₹${formatMoney(bill.gstAmount)}`);
   637→  lines.push(`*GRAND TOTAL: ₹${formatMoney(bill.total)}*`);
   638→  if (bill.paidAmount > 0) lines.push(`Paid: ₹${formatMoney(bill.paidAmount)}`);
   639→  if (bill.dueAmount > 0) lines.push(`⚠ Balance Due: ₹${formatMoney(bill.dueAmount)}`);
   640→  const pmLabel: Record<string, string> = { cash: 'Cash', upi: 'UPI', card: 'Card', due: 'Due (Credit)' };
   641→  lines.push(`Payment: ${pmLabel[bill.paymentMethod] || bill.paymentMethod}`);
   642→  lines.push('━━━━━━━━━━━━━━━━━━');
   643→  lines.push('Thank you for shopping with us!');
   644→  if (opts?.withPdfNote) lines.push('— PDF bill attached —');
   645→  return lines.join('\n');
   646→}

   665→export type WhatsappSendResult =
   666→  | 'shared'          // PDF handed to the OS share sheet → user picks WhatsApp
   667→  | 'cancelled'       // user closed the share sheet
   668→  | 'iframe-blocked'  // blocked because the app runs inside an iframe (preview)
   669→  | 'fallback'        // PDF downloaded + wa.me chat opened → attach the file manually
   670→  | 'no-mobile';      // PDF downloaded, but customer has no WhatsApp number
   671→
   672→/**
   673→ * Fallback when the file share is impossible: download the PDF and open the
   674→ * customer's WhatsApp chat with the bill summary, so the user only has to
   675→ * attach the downloaded file with the 📎 button.
   676→ */
   677→export async function fallbackDownloadAndChat(
   678→  bill: Bill,
   679→  pdf: GeneratedPdf,
   680→): Promise<'fallback' | 'no-mobile'> {
   681→  downloadBlob(pdf.blob, pdf.fileName);
   682→  if (bill.customerMobile) {
   683→    openWhatsappChat(bill.customerMobile, buildBillMessage(bill));
   684→    return 'fallback';
   685→  }
   686→  return 'no-mobile';
   687→}

   689→/**
   690→ * One-tap WhatsApp flow WITH the PDF attached:
   691→ *  1. Mobile browser (HTTPS): Web Share API Level 2 — the share sheet opens
   692→ *     carrying the REAL PDF file + pre-filled bill message; the user taps
   693→ *     WhatsApp, picks the customer chat, and the PDF travels attached.
   694→ *  2. Native APK: Android share sheet carrying the real PDF file.
   695→ *  3. Fallback (desktop / files unsupported / share blocked): downloads the
   696→ *     PDF and opens the customer's wa.me chat with the bill summary so the
   697→ *     user attaches the downloaded file manually.
   698→ */
   699→export async function sendBillViaWhatsapp(
   700→  bill: Bill,
   701→  pdf: GeneratedPdf,
   702→  opts?: { pdfNote?: boolean },
   703→): Promise<WhatsappSendResult> {
   704→  const message = buildBillMessage(bill, { withPdfNote: opts?.pdfNote !== false });
   705→  const res = await sharePdfFile(pdf.blob, pdf.fileName, message);
   706→  if (res === 'shared' || res === 'cancelled') return res;
   707→  if (res === 'blocked' && isEmbeddedFrame()) return 'iframe-blocked';
   708→  return fallbackDownloadAndChat(bill, pdf);
   709→}
