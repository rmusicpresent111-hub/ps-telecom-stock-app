     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef } from 'react';
     4→import { useAppStore } from '@/store/appStore';
     5→import { t } from '@/lib/i18n';
     6→import { Product, Transaction } from '@/lib/types';
     7→import { getProductsOffline, createTransactionOffline } from '@/lib/offline-service';
     8→import { playStockOutSound, playSellSound, playStockInSound } from '@/lib/sound-service';
     9→import { ArrowLeft, Search, X, TrendingUp, TrendingDown, ShoppingBag, ReceiptText } from 'lucide-react';
    10→import { toast } from 'sonner';
    11→import { motion, AnimatePresence } from 'framer-motion';
    12→
    13→const stockOpMeta: Record<string, { icon: React.ElementType; confirmLabel: string }> = {
    14→  STOCK_IN: { icon: TrendingUp, confirmLabel: 'Stock In' },
    15→  STOCK_OUT: { icon: TrendingDown, confirmLabel: 'Stock Out' },
    16→  SELL: { icon: ShoppingBag, confirmLabel: 'Sell' },
    17→};
    18→
    19→export default function StockOperationScreen() {
    20→  const user = useAppStore(s => s.user);
    21→  const language = useAppStore(s => s.language);
    22→  const goBack = useAppStore(s => s.goBack);
    23→  const navigateTo = useAppStore(s => s.navigateTo);
    24→  const setPendingBillData = useAppStore(s => s.setPendingBillData);
    25→  const selectedProductId = useAppStore(s => s.selectedProductId);
    26→  const stockOperationType = useAppStore(s => s.stockOperationType);
    27→
    28→  const isSell = stockOperationType === 'SELL';
    29→
    30→  const [product, setProduct] = useState<Product | null>(null);
    31→  const [productName, setProductName] = useState('');
    32→  const [allProducts, setAllProducts] = useState<Product[]>([]);
    33→  const [showSuggestions, setShowSuggestions] = useState(false);
    34→  const [quantity, setQuantity] = useState('');
    35→  const [price, setPrice] = useState('');
    36→  const [loading, setLoading] = useState(false);
    37→  const [fetching, setFetching] = useState(true);
    38→  const [showConfirmDialog, setShowConfirmDialog] = useState(false);
    39→  const [billPrompt, setBillPrompt] = useState<Transaction | null>(null);
    40→  const suggestionsRef = useRef<HTMLDivElement>(null);
    41→  const lastMatchedIdRef = useRef<string | null>(null);
    42→  const preselectedRef = useRef(false);
    43→
    44→  // Always search across ALL products — a stale global category filter used to
    45→  // make stock ops fail with "Product not found" for products of other categories.
    46→  const fetchProducts = useCallback(async () => {
    47→    if (!user?.id) return;
    48→    try {
    49→      setFetching(true);
    50→      const products = await getProductsOffline(user.id);
    51→      setAllProducts(products || []);
    52→      setProductName('');
    53→    } catch {
    54→      toast.error(t('error', language));
    55→    } finally {
    56→      setFetching(false);
    57→    }
    58→  }, [user?.id, language]);
    59→
    60→  useEffect(() => {
    61→    fetchProducts();
    62→  }, [fetchProducts]);
    63→
    64→  // Preselect the product when arriving from Product/Category detail
    65→  useEffect(() => {
    66→    if (preselectedRef.current || fetching) return;
    67→    if (!selectedProductId || allProducts.length === 0) return;
    68→    const pre = allProducts.find(p => p.id === selectedProductId);
    69→    if (pre) {
    70→      setProduct(pre);
    71→      setProductName(pre.name);
    72→      setPrice(stockOperationType === 'STOCK_IN' ? String(pre.purchasePrice) : String(pre.sellingPrice));
    73→      lastMatchedIdRef.current = pre.id;
    74→    }
    75→    preselectedRef.current = true;
    76→  }, [allProducts, fetching, selectedProductId, stockOperationType]);
    77→
    78→  // Close suggestions on outside click
    79→  useEffect(() => {
    80→    const handleClickOutside = (e: MouseEvent) => {
    81→      if (suggestionsRef.current && !suggestionsRef.current.contains(e.target as Node)) {
    82→        setShowSuggestions(false);
    83→      }
    84→    };
    85→    document.addEventListener('mousedown', handleClickOutside);
    86→    return () => document.removeEventListener('mousedown', handleClickOutside);
    87→  }, []);
    88→
    89→  // For non-SELL: auto-match product by exact name
    90→  useEffect(() => {
    91→    if (isSell) return;
    92→    if (!productName.trim()) {
    93→      setProduct(null);
    94→      lastMatchedIdRef.current = null;
    95→      return;
    96→    }
    97→    const matched = allProducts.find(
    98→      p => p.name.toLowerCase() === productName.trim().toLowerCase()
    99→    );
   100→    if (matched) {
   101→      setProduct(matched);
   102→      // Only auto-fill the price when the matched product CHANGES —
   103→      // re-running per keystroke used to clobber the user's manual edits.
   104→      if (lastMatchedIdRef.current !== matched.id) {
   105→        lastMatchedIdRef.current = matched.id;
   106→        if (stockOperationType === 'STOCK_IN') {
   107→          setPrice(String(matched.purchasePrice));
   108→        }
   109→      }
   110→    } else {
   111→      setProduct(null);
   112→      lastMatchedIdRef.current = null;
   113→    }
   114→  }, [productName, allProducts, stockOperationType, isSell]);
   115→
   116→  // SELL: filtered suggestions
   117→  const sellSuggestions = isSell && productName.trim()
   118→    ? allProducts.filter(p =>
   119→        p.name.toLowerCase().includes(productName.toLowerCase())
   120→      )
   121→    : isSell ? allProducts : [];
   122→
   123→  const handleSelectSuggestion = (p: Product) => {
   124→    setProduct(p);
   125→    setProductName('');
   126→    setShowSuggestions(false);
   127→    setPrice(String(p.sellingPrice));
   128→    setQuantity('');
   129→  };
   130→
   131→  const handleProductNameChange = (val: string) => {
   132→    setProductName(val);
   133→    if (isSell) {
   134→      setShowSuggestions(true);
   135→      if (product) {
   136→        setProduct(null);
   137→        setPrice('');
   138→        setQuantity('');
   139→      }
   140→    }
   141→  };
   142→
   143→  const getTitle = () => {
   144→    switch (stockOperationType) {
   145→      case 'STOCK_IN': return t('stockInTitle', language);
   146→      case 'STOCK_OUT': return t('stockOutTitle', language);
   147→      case 'SELL': return t('instantSellTitle', language);
   148→      default: return '';
   149→    }
   150→  };
   151→
   152→  const getPriceLabel = () => {
   153→    switch (stockOperationType) {
   154→      case 'SELL': return t('sellingPrice', language);
   155→      case 'STOCK_IN': return t('purchasePrice', language);
   156→      case 'STOCK_OUT': return t('price', language);
   157→      default: return t('price', language);
   158→    }
   159→  };
   160→
   161→  const getAccentColor = () => {
   162→    switch (stockOperationType) {
   163→      case 'STOCK_IN': return '#F5DEB3';
   164→      case 'STOCK_OUT': return '#ff6b00';
   165→      case 'SELL': return '#A07C3E';
   166→      default: return '#D4A853';
   167→    }
   168→  };
   169→
   170→  const handleConfirmClick = () => {
   171→    if (!product?.id || !user?.id || !stockOperationType || !quantity) {
   172→      toast.error(t('error', language));
   173→      return;
   174→    }
   175→    const qty = Number(quantity);
   176→    if (!Number.isInteger(qty) || qty <= 0) {
   177→      toast.error(t('enterQuantity', language));
   178→      return;
   179→    }
   180→    const unitPrice = Number(price);
   181→    if (!Number.isFinite(unitPrice) || unitPrice < 0) {
   182→      toast.error(t('error', language), { description: 'Price cannot be negative' });
   183→      return;
   184→    }
   185→    if ((stockOperationType === 'SELL' || stockOperationType === 'STOCK_IN') && unitPrice <= 0) {
   186→      toast.error(t('error', language), {
   187→        description: stockOperationType === 'SELL'
   188→          ? 'Selling price must be greater than 0'
   189→          : 'Purchase price must be greater than 0',
   190→      });
   191→      return;
   192→    }
   193→    setShowConfirmDialog(true);
   194→  };
   195→
   196→  const handleConfirmYes = async () => {
   197→    setShowConfirmDialog(false);
   198→    if (!product?.id || !user?.id || !stockOperationType) return;
   199→
   200→    const qty = Number(quantity);
   201→    const unitPrice = Number(price);
   202→    const totalAmount = qty * unitPrice;
   203→
   204→    setLoading(true);
   205→    try {
   206→      const { transaction } = await createTransactionOffline({
   207→        type: stockOperationType,
   208→        productId: product.id,
   209→        quantity: qty,
   210→        unitPrice,
   211→        totalAmount,
   212→        userId: user.id,
   213→      });
   214→
   215→      // Play success sound based on operation type
   216→      if (stockOperationType === 'SELL') {
   217→        playSellSound();
   218→      } else if (stockOperationType === 'STOCK_OUT') {
   219→        playStockOutSound();
   220→      } else if (stockOperationType === 'STOCK_IN') {
   221→        playStockInSound();
   222→      }
   223→
   224→      toast.success(t('success', language));
   225→
   226→      if (stockOperationType === 'STOCK_OUT' || stockOperationType === 'SELL') {
   227→        // Ask for e-Bill BEFORE leaving the screen (user can skip)
   228→        setBillPrompt(transaction);
   229→      } else {
   230→        goBack();
   231→      }
   232→    } catch (error) {
   233→      toast.error((error as Error).message || t('error', language));
   234→    } finally {
   235→      setLoading(false);
   236→    }
   237→  };
   238→
   239→  const handleBillSkip = () => {
   240→    setBillPrompt(null);
   241→    goBack();
   242→  };
   243→
   244→  const handleBillCreate = () => {
   245→    if (!billPrompt || !product) return;
   246→    setPendingBillData({
   247→      transactionId: billPrompt.id,
   248→      productId: billPrompt.productId,
   249→      productName: billPrompt.product?.name || product.name,
   250→      quantity: billPrompt.quantity,
   251→      unitPrice: billPrompt.unitPrice,
   252→      totalAmount: billPrompt.totalAmount,
   253→      date: billPrompt.date,
   254→    });
   255→    setBillPrompt(null);
   256→    goBack();          // leave the stock screen (stack: [..., prev])
   257→    navigateTo('invoice'); // push e-Bill screen (stack: [..., prev, invoice])
   258→  };
   259→
   260→  const handleConfirmCancel = () => {
   261→    setShowConfirmDialog(false);
   262→  };
   263→
   264→  const accent = getAccentColor();
   265→  const meta = stockOperationType ? stockOpMeta[stockOperationType] : null;
   266→  const ConfirmIcon = meta?.icon;
   267→
   268→  const totalAmount = quantity && price ? (Number(quantity) * Number(price)) : 0;
   269→
   270→  return (
   271→    <div className="animated-bg min-h-screen pb-24">
   272→      <div className="max-w-md mx-auto px-4 pt-4">
   273→        {/* Header */}
   274→        <div className="flex items-center justify-between mb-6">
   275→          <button onClick={goBack} className="p-2 rounded-full glass-card" aria-label="Back">
   276→            <ArrowLeft size={20} className="text-emerald-400" />
   277→          </button>
   278→          <h1 className="text-lg font-bold" style={{ color: accent }}>{getTitle()}</h1>
   279→          <div className="w-10" />
   280→        </div>
   281→
   282→        {fetching ? (
   283→          <div className="text-center py-12 text-white/40">{t('loading', language)}</div>
   284→        ) : allProducts.length === 0 ? (
   285→          <div className="text-center py-12 text-white/40">{t('noData', language)}</div>
   286→        ) : isSell ? (
   287→          /* ==================== INSTANT SELL LAYOUT ==================== */
   288→          <>
   289→            <div className="glass-card-strong p-6 space-y-4">
   290→              {/* Step 1: Product name with suggestion dropdown */}
   291→              <div className="relative" ref={suggestionsRef}>
   292→                <label className="text-xs text-white/60 mb-1 block">{t('productName', language)}</label>
   293→                <div className="relative">
   294→                  <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-white/40" />
   295→                  <input
   296→                    type="text"
   297→                    value={productName}
   298→                    onChange={(e) => handleProductNameChange(e.target.value)}
   299→                    onFocus={() => setShowSuggestions(true)}
   300→                    placeholder={t('searchProducts', language)}
   301→                    className="glass-input w-full pl-10 pr-4 py-3 text-sm"
   302→                    autoFocus
   303→                  />
   304→                </div>
   305→
   306→                {/* Suggestion dropdown */}
   307→                {showSuggestions && sellSuggestions.length > 0 && !product && (
   308→                  <div className="absolute z-50 left-0 right-0 top-full mt-1 glass-card-strong max-h-60 overflow-y-auto rounded-xl">
   309→                    {sellSuggestions.map((p) => (
   310→                      <button
   311→                        key={p.id}
   312→                        onClick={() => handleSelectSuggestion(p)}
   313→                        className="w-full px-4 py-3 text-left text-sm flex items-center justify-between hover:bg-white/10 transition-colors border-b border-white/5 last:border-b-0"
   314→                      >
   315→                        <div className="flex-1">
   316→                          <span className="text-white/90">{p.name}</span>
   317→                          <div className="flex items-center gap-2 mt-0.5">
   318→                            <span className={`text-[10px] ${p.quantity <= p.lowStockThreshold ? 'text-orange-400' : 'text-green-400'}`}>
   319→                              Stock: {p.quantity}
   320→                            </span>
   321→                            <span className="text-[10px] text-emerald-400/60">₹{p.sellingPrice}</span>
   322→                          </div>
   323→                        </div>
   324→                      </button>
   325→                    ))}
   326→                  </div>
   327→                )}
   328→
   329→                {/* No results */}
   330→                {showSuggestions && productName.trim() && sellSuggestions.length === 0 && !product && (
   331→                  <div className="absolute z-50 left-0 right-0 top-full mt-1 glass-card-strong rounded-xl p-4 text-center">
   332→                    <span className="text-xs text-white/40">{t('noData', language)}</span>
   333→                  </div>
   334→                )}
   335→              </div>
   336→
   337→              {/* Step 2: After product selected */}
   338→              {product && (
   339→                <div className="space-y-4">
   340→                  {/* Selected product info */}
   341→                  <div className="glass-card p-3 flex items-center justify-between">
   342→                    <div>
   343→                      <p className="text-sm font-semibold text-white/90">{product.name}</p>
   344→                      <p className={`text-[11px] ${product.quantity <= product.lowStockThreshold ? 'text-orange-400' : 'text-green-400'}`}>
   345→                        Stock: {product.quantity}
   346→                      </p>
   347→                    </div>
   348→                    <button
   349→                      onClick={() => {
   350→                        setProduct(null);
   351→                        setProductName('');
   352→                        setPrice('');
   353→                        setQuantity('');
   354→                        setShowSuggestions(false);
   355→                      }}
   356→                      className="text-xs text-white/40 hover:text-white/70 px-2 py-1"
   357→                    >
   358→                      ✕ Change
   359→                    </button>
   360→                  </div>
   361→
   362→                  {/* Quantity */}
   363→                  <div>
   364→                    <label className="text-xs text-white/60 mb-1 block">{t('quantity', language)}</label>
   365→                    <input
   366→                      type="number"
   367→                      value={quantity}
   368→                      onChange={(e) => setQuantity(e.target.value)}
   369→                      placeholder="0"
   370→                      className="glass-input w-full px-4 py-3 text-sm"
   371→                      autoFocus
   372→                    />
   373→                  </div>
   374→
   375→                  {/* Selling Price */}
   376→                  <div>
   377→                    <label className="text-xs text-white/60 mb-1 block">{t('sellingPrice', language)}</label>
   378→                    <input
   379→                      type="number"
   380→                      value={price}
   381→                      onChange={(e) => setPrice(e.target.value)}
   382→                      placeholder="₹0"
   383→                      className="glass-input w-full px-4 py-3 text-sm"
   384→                    />
   385→                  </div>
   386→
   387→                  {/* Total preview */}
   388→                  {quantity && price && (
   389→                    <div className="pt-2 border-t border-white/10">
   390→                      <div className="flex justify-between items-center">
   391→                        <span className="text-xs text-white/60">Total Amount</span>
   392→                        <span className="text-lg font-bold" style={{ color: accent }}>
   393→                          ₹{totalAmount.toLocaleString()}
   394→                        </span>
   395→                      </div>
   396→                    </div>
   397→                  )}
   398→                </div>
   399→              )}
   400→            </div>
   401→
   402→            {/* Confirm button */}
   403→            <div className="mt-6">
   404→              <button
   405→                onClick={handleConfirmClick}
   406→                disabled={loading || !quantity || !product}
   407→                className="neon-btn-solid w-full py-3 font-semibold text-sm disabled:opacity-50"
   408→                style={{
   409→                  background: `linear-gradient(135deg, ${accent}44, ${accent}88)`,
   410→                  color: 'white',
   411→                }}
   412→              >
   413→                {loading ? t('loading', language) : t('confirm', language)}
   414→              </button>
   415→            </div>
   416→          </>
   417→        ) : (
   418→          /* ==================== STOCK IN / STOCK OUT LAYOUT ==================== */
   419→          <>
   420→            <div className="glass-card-strong p-6 space-y-4">
   421→              {/* Product name */}
   422→              <div>
   423→                <label className="text-xs text-white/60 mb-1 block">{t('productName', language)}</label>
   424→                <input
   425→                  type="text"
   426→                  value={productName}
   427→                  onChange={(e) => setProductName(e.target.value)}
   428→                  placeholder={t('productName', language)}
   429→                  className="glass-input w-full px-4 py-3 text-sm"
   430→                  autoFocus
   431→                />
   432→                {productName.trim() && (
   433→                  product ? (
   434→                    <p className="text-[11px] text-green-400 mt-1">✓ {product.name} — Stock: {product.quantity}</p>
   435→                  ) : (
   436→                    <p className="text-[11px] text-orange-400/60 mt-1">Product not found</p>
   437→                  )
   438→                )}
   439→              </div>
   440→
   441→              {product && (
   442→                <>
   443→                  <div>
   444→                    <label className="text-xs text-white/60 mb-1 block">{t('quantity', language)}</label>
   445→                    <input
   446→                      type="number"
   447→                      value={quantity}
   448→                      onChange={(e) => setQuantity(e.target.value)}
   449→                      placeholder="0"
   450→                      className="glass-input w-full px-4 py-3 text-sm"
   451→                    />
   452→                  </div>
   453→
   454→                  <div>
   455→                    <label className="text-xs text-white/60 mb-1 block">{getPriceLabel()}</label>
   456→                    <input
   457→                      type="number"
   458→                      value={price}
   459→                      onChange={(e) => setPrice(e.target.value)}
   460→                      placeholder="₹0"
   461→                      className="glass-input w-full px-4 py-3 text-sm"
   462→                    />
   463→                  </div>
   464→
   465→                  {quantity && price && (
   466→                    <div className="pt-2 border-t border-white/10">
   467→                      <div className="flex justify-between items-center">
   468→                        <span className="text-xs text-white/60">Total Amount</span>
   469→                        <span className="text-lg font-bold" style={{ color: accent }}>
   470→                          ₹{totalAmount.toLocaleString()}
   471→                        </span>
   472→                      </div>
   473→                    </div>
   474→                  )}
   475→                </>
   476→              )}
   477→            </div>
   478→
   479→            <div className="mt-6">
   480→              <button
   481→                onClick={handleConfirmClick}
   482→                disabled={loading || !quantity || !product}
   483→                className="neon-btn-solid w-full py-3 font-semibold text-sm disabled:opacity-50"
   484→                style={accent !== '#D4A853' ? {
   485→                  background: `linear-gradient(135deg, ${accent}44, ${accent}88)`,
   486→                  color: 'white',
   487→                } : undefined}
   488→              >
   489→                {loading ? t('loading', language) : t('confirm', language)}
   490→              </button>
   491→            </div>
   492→          </>
   493→        )}
   494→      </div>
   495→
   496→      {/* Confirmation Dialog */}
   497→      <AnimatePresence>
   498→        {showConfirmDialog && product && meta && ConfirmIcon && (
   499→          <motion.div
   500→            initial={{ opacity: 0 }}
   501→            animate={{ opacity: 1 }}
   502→            exit={{ opacity: 0 }}
   503→            className="fixed inset-0 z-[100] flex items-center justify-center p-6"
   504→            onClick={handleConfirmCancel}
   505→          >
   506→            <div className="absolute inset-0 bg-black/60" />
   507→
   508→            <motion.div
   509→              initial={{ scale: 0.9, opacity: 0 }}
   510→              animate={{ scale: 1, opacity: 1 }}
   511→              exit={{ scale: 0.9, opacity: 0 }}
   512→              transition={{ type: 'spring', damping: 25, stiffness: 300 }}
   513→              onClick={(e) => e.stopPropagation()}
   514→              className="glass-card-strong p-6 w-full max-w-sm relative z-10"
   515→            >
   516→              <button
   517→                onClick={handleConfirmCancel}
   518→                className="absolute top-3 right-3 p-1.5 rounded-full hover:bg-white/10 transition-colors"
   519→              >
   520→                <X size={16} className="text-white/40" />
   521→              </button>
   522→
   523→              <div
   524→                className="w-14 h-14 rounded-full flex items-center justify-center mx-auto mb-4"
   525→                style={{ backgroundColor: `${accent}20`, border: `1px solid ${accent}66` }}
   526→              >
   527→                <ConfirmIcon size={24} style={{ color: accent }} />
   528→              </div>
   529→
   530→              <h3 className="text-lg font-bold text-center mb-3" style={{ color: accent }}>
   531→                {meta.confirmLabel}?
   532→              </h3>
   533→
   534→              <div className="space-y-2 mb-5">
   535→                <div className="flex justify-between items-center py-2 px-3 rounded-lg" style={{ background: 'rgba(255,255,255,0.04)' }}>
   536→                  <span className="text-xs text-white/50">Product</span>
   537→                  <span className="text-sm font-medium text-white/90">{product.name}</span>
   538→                </div>
   539→                <div className="flex justify-between items-center py-2 px-3 rounded-lg" style={{ background: 'rgba(255,255,255,0.04)' }}>
   540→                  <span className="text-xs text-white/50">{t('quantity', language)}</span>
   541→                  <span className="text-sm font-medium text-white/90">{quantity} pcs</span>
   542→                </div>
   543→                <div className="flex justify-between items-center py-2 px-3 rounded-lg" style={{ background: 'rgba(255,255,255,0.04)' }}>
   544→                  <span className="text-xs text-white/50">{getPriceLabel()}</span>
   545→                  <span className="text-sm font-medium text-white/90">₹{parseFloat(price || '0').toLocaleString()}</span>
   546→                </div>
   547→                {totalAmount > 0 && (
