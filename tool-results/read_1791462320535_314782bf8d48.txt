     1→/**
     2→ * Local-First Service Layer — 100% offline, NO cloud database.
     3→ *
     4→ * READ STRATEGY:  Always served from local IndexedDB (instant, works with zero internet).
     5→ * WRITE STRATEGY: Written to local IndexedDB immediately (atomic where it matters).
     6→ * DELETE STRATEGY: Deleted from local IndexedDB (with cascades where required).
     7→ *
     8→ * Stock transactions are applied atomically (products + transactions stores in a
     9→ * single IndexedDB transaction) so rapid taps can never create negative stock.
    10→ * Profit/report math uses the purchase-price SNAPSHOT captured on each transaction,
    11→ * so later price edits never rewrite historical profit.
    12→ */

    31→// ============ DATE HELPERS (local timezone — never use UTC for business days) ============
    32→
    33→/**
    34→ * Returns YYYY-MM-DD for a Date in LOCAL time.
    35→ * toISOString() uses UTC — for IST (UTC+5:30) sales between 00:00–05:30 local
    36→ * would land on the previous day. This helper fixes that class of bugs.
    37→ */
    38→export function localDateStr(d: Date = new Date()): string {
    39→  const y = d.getFullYear();
    40→  const m = String(d.getMonth() + 1).padStart(2, '0');
    41→  const day = String(d.getDate()).padStart(2, '0');
    42→  return `${y}-${m}-${day}`;
    43→}

    45→/**
    46→ * Resolves a reporting period to a [from, to] date range (local time).
    47→ */
    48→export function resolvePeriodRange(options?: { period?: string; date?: string; from?: string; to?: string }): { from: string; to: string } {
    49→  const now = new Date();
    50→  if (options?.from && options?.to) {
    51→    // Guard against inverted custom ranges (from > to) — silently swap.
    52→    return options.from <= options.to
    53→      ? { from: options.from, to: options.to }
    54→      : { from: options.to, to: options.from };
    55→  }
    56→  if (options?.date) return { from: options.date, to: options.date };
    57→
    58→  const todayStr = localDateStr(now);
    59→  const period = options?.period || 'today';
    60→  if (period === 'week') {
    61→    const weekStart = new Date(now);
    62→    const day = weekStart.getDay();
    63→    const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1); // Monday start
    64→    weekStart.setDate(diff);
    65→    return { from: localDateStr(weekStart), to: todayStr };
    66→  }
    67→  if (period === 'month') {
    68→    return { from: `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`, to: todayStr };
    69→  }
    70→  // today (default) or any unknown period
    71→  return { from: todayStr, to: todayStr };
    72→}

    74→// ============ SHARED STRIP HELPERS ============

    76→function stripSync<T>(row: Record<string, unknown>): T {
    77→  const { _synced: _s, _dirty: _d, ...rest } = row;
    78→  return rest as T;
    79→}

    81→function sortTxnsDesc(rows: OfflineDBSchema['transactions']['value'][]): OfflineDBSchema['transactions']['value'][] {
    82→  return [...rows].sort((a, b) => {
    83→    if (a.date !== b.date) return b.date.localeCompare(a.date);
    84→    return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    85→  });
    86→}
    87→
    88→function localTxnsToTransactions(localTxns: OfflineDBSchema['transactions']['value'][]): Transaction[] {
    89→  return sortTxnsDesc(localTxns).map(t => stripSync<Transaction>(t));
    90→}

    92→/** Cost basis for a transaction: the SNAPSHOT taken at sale time, falling back to the current product. */
    93→function txnCostBasis(t: Transaction, products: Product[]): number {
    94→  return t.product?.purchasePrice ?? products.find(p => p.id === t.productId)?.purchasePrice ?? 0;
    95→}

   250→// ============ TRANSACTIONS (Local, ATOMIC stock updates) ============
   251→
   252→/**
   253→ * Creates a transaction AND applies the stock change in ONE atomic IndexedDB
   254→ * transaction. Two rapid sells can never both pass the same stock check.
   255→ *
   256→ * NOTE: idb v8's db.transaction() has NO callback parameter — the correct
   257→ * atomic pattern is: open tx → await requests → `await tx.done`.
   258→ */
   259→export async function createTransactionOffline(transactionData: {
   260→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   261→  productId: string;
   262→  quantity: number;
   263→  unitPrice?: number;
   264→  totalAmount?: number;
   265→  date?: string;
   266→  userId: string;
   267→}): Promise<{ transaction: Transaction }> {
   268→  const quantity = Math.floor(Number(transactionData.quantity));
   269→  if (!Number.isFinite(quantity) || quantity <= 0) throw new Error('Quantity must be a positive whole number');
   270→
   271→  const unitPrice = Number(transactionData.unitPrice ?? 0);
   272→  const totalAmount = Number(transactionData.totalAmount ?? 0);
   273→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   274→  if (!Number.isFinite(totalAmount) || totalAmount < 0) throw new Error('Total amount cannot be negative');
   275→
   276→  const db = await getOfflineDB();
   277→  const now = new Date().toISOString();
   278→  const id = generateRecordId();
   279→
   280→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   281→  const productsStore = tx.objectStore('products');
   282→  const transactionsStore = tx.objectStore('transactions');
   283→
   284→  // Validate BEFORE writing anything — a readwrite tx with no writes commits empty.
   285→  const product = await productsStore.get(transactionData.productId);
   286→  if (!product) {
   287→    throw new Error('Product not found');
   288→  }
   289→
   290→  const isOut = transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL';
   291→  if (isOut && product.quantity < quantity) {
   292→    throw new Error(`Insufficient stock — only ${product.quantity} left`);
   293→  }
   294→
   295→  const quantityChange = transactionData.type === 'STOCK_IN' ? quantity : -quantity;
   296→
   297→  const row: OfflineDBSchema['transactions']['value'] = {
   298→    id,
   299→    productId: transactionData.productId,
   300→    type: transactionData.type,
   301→    quantity,
   302→    unitPrice,
   303→    totalAmount,
   304→    date: transactionData.date || localDateStr(),
   305→    userId: transactionData.userId,
   306→    createdAt: now,
   307→    // Snapshot the product prices at transaction time — historical profit must
   308→    // never change when the user later edits the product's prices.
   309→    product: {
   310→      id: product.id,
   311→      name: product.name,
   312→      purchasePrice: product.purchasePrice,
   313→      sellingPrice: product.sellingPrice,
   314→    },
   315→    _synced: Date.now(),
   316→    _dirty: 0,
   317→  };
   318→
   319→  await transactionsStore.put(row);
   320→  await productsStore.put({
   321→    ...product,
   322→    quantity: product.quantity + quantityChange,
   323→    updatedAt: now,
   324→  });
   325→
   326→  // Commit — rejects (and rolls back both stores) if anything failed.
   327→  await tx.done;
   328→
   329→  return { transaction: stripSync<Transaction>(row) };
   330→}

   332→export async function deleteTransactionOffline(id: string, userId: string): Promise<{ success: boolean }> {
   333→  const db = await getOfflineDB();
   334→  const now = new Date().toISOString();
   335→
   336→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   337→  const productsStore = tx.objectStore('products');
   338→  const transactionsStore = tx.objectStore('transactions');
   339→
   340→  const txn = await transactionsStore.get(id);
   341→  if (!txn || txn.userId !== userId) {
   342→    // Nothing to delete — no writes were queued, committing empty is harmless.
   343→    await tx.done;
   344→    return { success: true };
   345→  }
   346→
   347→  // Reverse the stock change caused by this transaction (same atomic tx)
   348→  const product = await productsStore.get(txn.productId);
   349→  if (product) {
   350→    const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   351→    await productsStore.put({
   352→      ...product,
   353→      quantity: Math.max(0, product.quantity + quantityChange),
   354→      updatedAt: now,
   355→    });
   356→  }
   357→
   358→  await transactionsStore.delete(id);
   359→  await tx.done;
   360→
   361→  return { success: true };
   362→}

   364→export async function getTransactionsOffline(userId: string, options?: { type?: string; productId?: string; from?: string; to?: string }): Promise<Transaction[]> {
   365→  let localTxns = await offlineTransactions.getAll(userId);

   368→  if (options?.productId) localTxns = localTxns.filter(t => t.productId === options.productId);
   369→  if (options?.from) localTxns = localTxns.filter(t => t.date >= options.from!);
   370→  if (options?.to) localTxns = localTxns.filter(t => t.date <= options.to!);

   372→  return localTxnsToTransactions(localTxns);

   377→export async function getDashboardOffline(userId: string) {
   378→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   379→    offlineProducts.getAll(userId),
   380→    offlineCategories.getAll(userId),
   381→    offlineTransactions.getAll(userId),
   382→  ]);
   383→
   384→  const products = localProducts.map(p => stripSync<Product>(p));
   385→  const categories = localCategories.map(c => stripSync<Category>(c));
   386→  const transactions = localTxnsToTransactions(localTransactions);
   387→
   388→  const today = localDateStr();
   389→  const todayTxns = transactions.filter(t => t.date === today);
   390→
   391→  const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   392→  const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   393→  const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   394→  const recentTransactions = transactions.slice(0, 5);
   395→
   396→  const productCountMap: Record<string, number> = {};
   397→  for (const p of products) {
   398→    productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   399→  }
   400→  const catsWithCount = categories.map(c => ({
   401→    ...c,
   402→    _count: { products: productCountMap[c.id] || 0 },
   403→  }));
   404→
   405→  const sevenDaysAgo = new Date();
   406→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   407→  const sevenDaysAgoStr = localDateStr(sevenDaysAgo);
   408→
   409→  const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   410→  const saleMap = new Map<string, { sales: number; quantity: number }>();
   411→  for (const t of saleTxns) {
   412→    const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   413→    existing.sales += t.totalAmount;
   414→    existing.quantity += t.quantity;
   415→    saleMap.set(t.date, existing);
   416→  }
   417→
   418→  const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   419→  for (let i = 6; i >= 0; i--) {
   420→    const d = new Date();
   421→    d.setDate(d.getDate() - i);
   422→    const dateStr = localDateStr(d);
   423→    const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   424→    const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   425→    saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   426→  }
   427→
   428→  const stockByCategory = new Map<string, { quantity: number; value: number }>();
   429→  for (const p of products) {
   430→    const cat = catsWithCount.find(c => c.id === p.categoryId);
   431→    const catName = cat?.name || 'Unknown';
   432→    const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   433→    existing.quantity += p.quantity;
   434→    existing.value += p.quantity * p.purchasePrice;
   435→    stockByCategory.set(catName, existing);
   436→  }
   437→
   438→  const stockOverview = Array.from(stockByCategory.entries())
   439→    .filter(([_, data]) => data.quantity > 0)
   440→    .map(([category, data]) => ({ category, ...data }));
   441→
   442→  return {
   443→    stats: {
   444→      totalItems: products.length,
   445→      lowItems,
   446→      todayTransactions: todayTxns.length,
   447→      stockValue,
   448→    },
   449→    lowStockProducts,
   450→    recentTransactions,
   451→    categories: catsWithCount,
   452→    saleOverview,
   453→    stockOverview,
   454→  };
   455→}

   462→export async function getExpensesOffline(userId: string, options?: { period?: string; date?: string; from?: string; to?: string; category?: string }) {
   463→  let localExpenses = await offlineExpenses.getAll(userId);
   464→
   465→  // Honor the reporting period exactly — otherwise "Today" would show all-time totals.
   466→  const { from, to } = resolvePeriodRange(options);
   467→  localExpenses = localExpenses.filter(e => e.date >= from && e.date <= to);
   468→  if (options?.category) localExpenses = localExpenses.filter(e => e.category === options.category);
   469→
   470→  const expenses = localExpenses
   471→    .map(e => stripSync<Expense>(e))
   472→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   473→  const totalExpense = expenses.reduce((sum, e) => sum + (e.amount || 0), 0);
   474→  const byCategory = expenses.reduce((acc, e) => {
   475→    const cat = e.category || 'other';
   476→    acc[cat] = (acc[cat] || 0) + (e.amount || 0);
   477→    return acc;
   478→  }, {} as Record<string, number>);
   479→
   480→  return {
   481→    expenses,
   482→    summary: { totalExpense, byCategory, count: expenses.length },
   483→  };
   484→}

   552→  let localEntries = await offlineCashEntries.getAll(userId);
   553→
   554→  // Honor the reporting period.

   556→  localEntries = localEntries.filter(e => e.date >= from && e.date <= to);

   558→  const entries = localEntries
   559→    .map(e => stripSync<CashEntry>(e))
   560→    .sort((a, b) => b.date.localeCompare(a.date));
   561→  // Balance semantics: the LATEST entry in the filtered period holds the running balance.
   562→  const latest = entries[0];
   563→  const totalHandCash = latest?.handCash || 0;
   564→  const totalLiquidCash = latest?.liquidCash || 0;
   565→
   566→  return {
   567→    entries,
   568→    summary: { totalHandCash, totalLiquidCash, totalCash: totalHandCash + totalLiquidCash },

   572→export async function upsertCashEntryOffline(entryData: {
   573→  userId: string;
   574→  date: string;
   575→  handCash?: number;
   576→  liquidCash?: number;
   577→  note?: string;

   581→      const num = Number(value);

   584→  }

   586→  const now = new Date().toISOString();

   603→      _synced: Date.now(),
   604→      _dirty: 0,
   605→    });

   607→    return { entry: stripSync<CashEntry>(updated) };
   608→  }

   610→  const newEntry = {
   611→    id: generateRecordId(),
   612→    userId: entryData.userId,
   613→    date: entryData.date,
   614→    handCash: Number(entryData.handCash ?? 0),
   615→    liquidCash: Number(entryData.liquidCash ?? 0),

   622→    ...newEntry,
   623→    _synced: Date.now(),
   624→    _dirty: 0,

   627→  return { entry: newEntry };

   666→export async function getReportsOffline(userId: string, type: string, options?: { from?: string; to?: string }) {
   667→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   668→    offlineProducts.getAll(userId),
   669→    offlineCategories.getAll(userId),
   670→    offlineTransactions.getAll(userId),
   671→  ]);
   672→
   673→  const products = localProducts.map(p => stripSync<Product>(p));
   674→  const categories = localCategories.map(c => stripSync<Category>(c));
   675→  const transactions = localTxnsToTransactions(localTransactions);
   676→
   677→  // Apply date filters
   678→  let filteredTxns = transactions;
   679→  if (options?.from) filteredTxns = filteredTxns.filter(t => t.date >= options.from!);
   680→  if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   681→
   682→  if (type === 'stock-value') {
   683→    const stockValueData = categories.map(cat => {
   684→      const catProducts = products.filter(p => p.categoryId === cat.id);
   685→      const totalQty = catProducts.reduce((sum, p) => sum + p.quantity, 0);
   686→      const totalPurchaseValue = catProducts.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   687→      const totalSellingValue = catProducts.reduce((sum, p) => sum + p.quantity * p.sellingPrice, 0);
   688→      const totalProfit = totalSellingValue - totalPurchaseValue;
   689→      const lowStockCount = catProducts.filter(p => p.quantity <= p.lowStockThreshold).length;
   690→
   691→      return {
   692→        categoryId: cat.id,
   693→        categoryName: cat.name,
   694→        categoryImage: cat.image,
   695→        productCount: catProducts.length,
   696→        totalQty,
   697→        totalPurchaseValue,
   698→        totalSellingValue,
   699→        totalProfit,
   700→        lowStockCount,
   701→        products: catProducts.map(p => ({
   702→          id: p.id,
   703→          name: p.name,
   704→          quantity: p.quantity,
   705→          purchasePrice: p.purchasePrice,
   706→          sellingPrice: p.sellingPrice,
   707→          stockValue: p.quantity * p.purchasePrice,
   708→          purchaseValue: p.quantity * p.purchasePrice,
   709→          lowStock: p.quantity <= p.lowStockThreshold,
   710→        })),
   711→      };
   712→    }).filter(item => item.productCount > 0);
   713→
   714→    const grandTotal = {
   715→      totalProducts: stockValueData.reduce((s, d) => s + d.productCount, 0),
   716→      totalQty: stockValueData.reduce((s, d) => s + d.totalQty, 0),
   717→      totalPurchaseValue: stockValueData.reduce((s, d) => s + d.totalPurchaseValue, 0),
   718→      totalSellingValue: stockValueData.reduce((s, d) => s + d.totalSellingValue, 0),

   720→      totalLowStock: stockValueData.reduce((s, d) => s + d.lowStockCount, 0),
   721→    };
   722→
   723→    return { type: 'stock-value', data: stockValueData, grandTotal };
   724→  }
   725→
   726→  // Daily report
   727→  if (type === 'daily') {
   728→    const dailyMap = new Map<string, {
   729→      date: string; revenue: number; cost: number; profit: number;
   730→      stockIn: number; stockOut: number; sell: number;
   731→    }>();
   732→
   733→    for (const t of filteredTxns) {
   734→      if (!dailyMap.has(t.date)) {
   735→        dailyMap.set(t.date, { date: t.date, revenue: 0, cost: 0, profit: 0, stockIn: 0, stockOut: 0, sell: 0 });
   736→      }
   737→      const entry = dailyMap.get(t.date)!;

   739→
   740→      if (t.type === 'SELL') {
   741→        entry.revenue += t.totalAmount || 0;
   742→        entry.cost += (t.quantity || 0) * costBasis;
   743→        entry.profit += (t.totalAmount || 0) - (t.quantity || 0) * costBasis;
   744→        entry.sell += t.quantity || 0;
   745→      } else if (t.type === 'STOCK_IN') {
   746→        entry.cost += (t.quantity || 0) * (t.unitPrice || 0);
   747→        entry.stockIn += t.quantity || 0;
   748→      } else if (t.type === 'STOCK_OUT') {
   749→        entry.stockOut += t.quantity || 0;
   750→      }
   751→    }
   752→
   753→    const values = Array.from(dailyMap.values());
   754→    return {
   755→      type: 'daily',
   756→      data: values.sort((a, b) => b.date.localeCompare(a.date)),

   758→        totalRevenue: values.reduce((s, d) => s + d.revenue, 0),
   759→        totalCost: values.reduce((s, d) => s + d.cost, 0),
   760→        totalProfit: values.reduce((s, d) => s + d.profit, 0),
   761→        totalStockIn: values.reduce((s, d) => s + d.stockIn, 0),
   762→        totalStockOut: values.reduce((s, d) => s + d.stockOut, 0),
   763→        totalSell: values.reduce((s, d) => s + d.sell, 0),
   764→      },
   765→    };
   766→  }
   767→
   768→  // Monthly report
   769→  if (type === 'monthly') {
   770→    const monthlyMap = new Map<string, {
   771→      month: string; revenue: number; cost: number; profit: number;
   772→      stockIn: number; stockOut: number; sell: number;
   773→    }>();
   774→
   775→    for (const t of filteredTxns) {
   776→      const month = t.date.substring(0, 7); // YYYY-MM
   777→      if (!monthlyMap.has(month)) {
   778→        monthlyMap.set(month, { month, revenue: 0, cost: 0, profit: 0, stockIn: 0, stockOut: 0, sell: 0 });
   779→      }
   780→      const entry = monthlyMap.get(month)!;
   781→      const costBasis = txnCostBasis(t, products);
   782→
   783→      if (t.type === 'SELL') {
   784→        entry.revenue += t.totalAmount || 0;
   785→        entry.cost += (t.quantity || 0) * costBasis;
   786→        entry.profit += (t.totalAmount || 0) - (t.quantity || 0) * costBasis;
   787→        entry.sell += t.quantity || 0;
   788→      } else if (t.type === 'STOCK_IN') {
   789→        entry.cost += (t.quantity || 0) * (t.unitPrice || 0);
   790→        entry.stockIn += t.quantity || 0;
   791→      } else if (t.type === 'STOCK_OUT') {
   792→        entry.stockOut += t.quantity || 0;
   793→      }
   794→    }
   795→
   796→    const values = Array.from(monthlyMap.values());
   797→    return {
   798→      type: 'monthly',
   799→      data: values.sort((a, b) => b.month.localeCompare(a.month)),
   800→      grandTotal: {
   801→        totalRevenue: values.reduce((s, d) => s + d.revenue, 0),
   802→        totalCost: values.reduce((s, d) => s + d.cost, 0),
   803→        totalProfit: values.reduce((s, d) => s + d.profit, 0),
   804→        totalStockIn: values.reduce((s, d) => s + d.stockIn, 0),

   806→        totalSell: values.reduce((s, d) => s + d.sell, 0),
   807→      },
   808→    };
   809→  }
   810→
   811→  // Category report
   812→  if (type === 'category') {
   813→    const categoryMap = new Map<string, {
   814→      categoryId: string; categoryName: string;
   815→      revenue: number; cost: number; profit: number;
   816→      stockIn: number; stockOut: number; sell: number; quantity: number;
   817→    }>();
   818→
   819→    for (const t of filteredTxns) {
   820→      const product = products.find(p => p.id === t.productId) ?? (t.product ? ({ ...t.product, categoryId: '', createdAt: '', updatedAt: '', boxNumber: '', lowStockThreshold: 0, quantity: 0, userId: '' } as Product) : undefined);
   821→      const cat = categories.find(c => c.id === product?.categoryId);
   822→      const catId = cat?.id || 'unknown';
   823→      const catName = cat?.name || 'Unknown';
   824→
   825→      if (!categoryMap.has(catId)) {
   826→        categoryMap.set(catId, { categoryId: catId, categoryName: catName, revenue: 0, cost: 0, profit: 0, stockIn: 0, stockOut: 0, sell: 0, quantity: 0 });
   827→      }
   828→      const entry = categoryMap.get(catId)!;
   829→      const costBasis = txnCostBasis(t, products);
   830→
   831→      if (t.type === 'SELL') {
   832→        entry.revenue += t.totalAmount || 0;
   833→        entry.cost += (t.quantity || 0) * costBasis;
   834→        entry.profit += (t.totalAmount || 0) - (t.quantity || 0) * costBasis;

   838→        entry.cost += (t.quantity || 0) * (t.unitPrice || 0);

   846→
   847→    const values = Array.from(categoryMap.values());
   848→    return {
   849→      type: 'category',
   850→      data: values.sort((a, b) => b.revenue - a.revenue),
   851→      grandTotal: {
   852→        totalRevenue: values.reduce((s, d) => s + d.revenue, 0),
   853→        totalCost: values.reduce((s, d) => s + d.cost, 0),
   854→        totalProfit: values.reduce((s, d) => s + d.profit, 0),
   855→        totalStockIn: values.reduce((s, d) => s + d.stockIn, 0),
   856→        totalStockOut: values.reduce((s, d) => s + d.stockOut, 0),
   857→        totalSell: values.reduce((s, d) => s + d.sell, 0),
   858→        totalQuantity: values.reduce((s, d) => s + d.quantity, 0),
   859→      },
   860→    };
   861→  }
   862→
   863→  return { type, data: [], grandTotal: {} };
   864→}

   983→/** Generates the next sequential bill number: PREFIX-YYMM-0001 */
   984→async function nextBillNumber(userId: string, prefix: string): Promise<string> {
   985→  const bills = await offlineBills.getAll(userId);
   986→  const now = new Date();
   987→  const yy = String(now.getFullYear()).slice(-2);
   988→  const mm = String(now.getMonth() + 1).padStart(2, '0');
   989→  const monthPrefix = `${prefix}-${yy}${mm}-`;
   990→  let maxSeq = 0;
   991→  for (const b of bills) {
   992→    if (typeof b.billNumber === 'string' && b.billNumber.startsWith(monthPrefix)) {
   993→      const seq = parseInt(b.billNumber.slice(monthPrefix.length), 10);
   994→      if (Number.isFinite(seq) && seq > maxSeq) maxSeq = seq;
   995→    }
   996→  }
   997→  return `${monthPrefix}${String(maxSeq + 1).padStart(4, '0')}`;

  1000→export async function createBillOffline(data: {
  1001→  userId: string;
  1002→  transactionId: string;
  1003→  productId: string;
  1004→  productName: string;
  1005→  quantity: number;
  1006→  unitPrice: number;
  1007→  customerName?: string;
  1008→  customerMobile?: string;
  1009→  discountType?: 'amount' | 'percent';
  1010→  discountValue?: number;
  1011→  paymentMethod?: PaymentMethod;
  1012→  paidAmount?: number;
  1013→  note?: string;
  1014→  date?: string;
  1015→}): Promise<{ bill: Bill }> {
  1016→  const qty = Math.floor(Number(data.quantity));
  1017→  if (!Number.isFinite(qty) || qty <= 0) throw new Error('Quantity must be a positive whole number');
  1018→  const unitPrice = Number(data.unitPrice);
  1019→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
  1020→
  1021→  const settings = await getBillingSettingsOffline(data.userId);
  1022→
  1023→  const discountType = data.discountType === 'percent' ? 'percent' : 'amount';
  1024→  let discountValue = Math.max(0, Number(data.discountValue ?? 0) || 0);
  1025→  const subtotal = round2(qty * unitPrice);
  1026→
  1027→  let discountAmount = 0;
  1028→  if (discountType === 'percent') {
  1029→    discountValue = clampNumber(discountValue, 0, 100);
  1030→    discountAmount = round2((subtotal * discountValue) / 100);
  1031→  } else {
  1032→    discountAmount = round2(Math.min(discountValue, subtotal));
  1033→  }
  1034→
  1035→  const gstEnabled = !!settings.gstEnabled && !!settings.gstNumber.trim() && settings.gstRate > 0;
  1036→  const gstAmount = gstEnabled ? round2(((subtotal - discountAmount) * settings.gstRate) / 100) : 0;
  1037→  const total = round2(subtotal - discountAmount + gstAmount);
  1038→
  1039→  const paymentMethod: PaymentMethod = data.paymentMethod ?? 'cash';
  1040→  let paidAmount = data.paidAmount === undefined ? total : Math.max(0, Number(data.paidAmount) || 0);
  1041→  if (paymentMethod === 'due') paidAmount = 0;
  1042→  paidAmount = round2(Math.min(paidAmount, total));
  1043→  const dueAmount = round2(total - paidAmount);
  1044→
  1045→  // Guard: one bill per stock-out/sell transaction — prevents accidental duplicates
  1046→  const existingBills = await offlineBills.getAll(data.userId);
  1047→  const dupe = existingBills.find(b => b.transactionId === data.transactionId);
  1048→  if (dupe) throw new Error('A bill already exists for this transaction');
  1049→

  1101→export async function getBillsOffline(userId: string, options?: { from?: string; to?: string; search?: string }): Promise<Bill[]> {
  1102→  let rows = await offlineBills.getAll(userId);
  1103→  if (options?.from) rows = rows.filter(b => b.date >= options.from!);
  1104→  if (options?.to) rows = rows.filter(b => b.date <= options.to!);
  1105→  if (options?.search) {
  1106→    const q = options.search.toLowerCase();
  1107→    rows = rows.filter(b =>
  1108→      b.customerName.toLowerCase().includes(q) ||
  1109→      b.customerMobile.includes(q) ||
  1110→      b.billNumber.toLowerCase().includes(q) ||
  1111→      b.items.some(i => i.name.toLowerCase().includes(q))
  1112→    );
  1113→  }
  1114→  return rows
  1115→    .sort((a, b) => {
  1116→      if (a.date !== b.date) return b.date.localeCompare(a.date);
  1117→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
  1118→    })
  1119→    .map(b => b as unknown as Bill);
  1120→}

  1165→export async function importBackupOffline(userId: string, backupData: {
  1166→  categories: unknown[];
  1167→  products: unknown[];
  1168→  transactions: unknown[];
  1169→  expenses?: unknown[];
  1170→  cashEntries?: unknown[];
  1171→  serviceTransactions?: unknown[];
  1172→  bills?: unknown[];
  1173→  billingSettings?: unknown;
  1174→}) {
  1175→  const now = Date.now();
  1176→
  1177→  // Validate rows: every row MUST have an id — malformed backups must never
  1178→  // be written into IndexedDB.
  1179→  const asRows = (rows: unknown[]): Record<string, unknown>[] => {
  1180→    if (!Array.isArray(rows)) return [];
  1181→    return rows.filter((r): r is Record<string, unknown> =>
  1182→      !!r && typeof r === 'object' && typeof (r as Record<string, unknown>).id === 'string' && (r as Record<string, unknown>).id !== ''
  1183→    );
  1184→  };
  1185→
  1186→  const categories = asRows(backupData.categories || []);
  1187→  const products = asRows(backupData.products || []);
  1188→  const transactions = asRows(backupData.transactions || []);
  1189→  const expenses = asRows(backupData.expenses || []);
  1190→  const cashEntries = asRows(backupData.cashEntries || []);
  1191→  const serviceTransactions = asRows(backupData.serviceTransactions || []);
  1192→  const bills = asRows(backupData.bills || []);
  1193→  const billingSettings = (backupData.billingSettings && typeof backupData.billingSettings === 'object')
  1194→    ? backupData.billingSettings as Record<string, unknown>
  1195→    : null;
  1196→
  1197→  if (categories.length === 0 && products.length === 0 && transactions.length === 0 && expenses.length === 0 && cashEntries.length === 0 && serviceTransactions.length === 0 && bills.length === 0) {
  1198→    throw new Error('Backup file contains no valid data');
  1199→  }
  1200→
  1201→  // REPLACE semantics: clear existing stores first so stale rows can't survive a restore.
  1202→  const [allLocalCats, allLocalProducts, allLocalTxns, allLocalExpenses, allLocalCash, allLocalSvc, allLocalBills] = await Promise.all([
  1203→    offlineCategories.getAll(userId),
  1204→    offlineProducts.getAll(userId),
  1205→    offlineTransactions.getAll(userId),
  1206→    offlineExpenses.getAll(userId),
  1207→    offlineCashEntries.getAll(userId),
  1208→    offlineServiceTransactions.getAll(userId),
  1209→    offlineBills.getAll(userId),
  1210→  ]);
  1211→  await Promise.all([
  1212→    ...allLocalCats.map(c => offlineCategories.delete(c.id)),
  1213→    ...allLocalProducts.map(p => offlineProducts.delete(p.id)),
  1214→    ...allLocalTxns.map(t => offlineTransactions.delete(t.id)),
  1215→    ...allLocalExpenses.map(e => offlineExpenses.delete(e.id)),
  1216→    ...allLocalCash.map(e => offlineCashEntries.delete(e.id)),
  1217→    ...allLocalSvc.map(s => offlineServiceTransactions.delete(s.id)),
  1218→    ...allLocalBills.map(b => offlineBills.delete(b.id)),
  1219→  ]);
  1220→
  1221→  if (categories.length) {
  1222→    await offlineCategories.putBulk(
  1223→      categories.map((cat) => ({ ...cat, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['categories']['value'][]
  1224→    );
  1225→  }
  1226→  if (products.length) {
  1227→    await offlineProducts.putBulk(
  1228→      products.map((prod) => ({ ...prod, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['products']['value'][]
  1229→    );
  1230→  }
  1231→  if (transactions.length) {
  1232→    await offlineTransactions.putBulk(
  1233→      transactions.map((txn) => ({ ...txn, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['transactions']['value'][]
  1234→    );
  1235→  }
  1236→  if (expenses.length) {
  1237→    await offlineExpenses.putBulk(
  1238→      expenses.map((exp) => ({ ...exp, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['expenses']['value'][]
  1239→    );
  1240→  }
  1241→  if (cashEntries.length) {
  1242→    await offlineCashEntries.putBulk(
  1243→      cashEntries.map((entry) => ({ ...entry, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['cashEntries']['value'][]
  1244→    );
  1245→  }
  1246→  if (serviceTransactions.length) {
  1247→    await offlineServiceTransactions.putBulk(
  1248→      serviceTransactions.map((svc) => ({ ...svc, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['serviceTransactions']['value'][]
  1249→    );
  1250→  }
  1251→  if (bills.length) {
  1252→    await offlineBills.putBulk(
  1253→      bills.map((bill) => ({ ...bill, userId, createdAt: bill.createdAt || new Date().toISOString(), updatedAt: bill.updatedAt || new Date().toISOString() })) as OfflineDBSchema['bills']['value'][]
  1254→    );
  1255→  }
  1256→  if (billingSettings) {
  1257→    const db = await getOfflineDB();
  1258→    await db.put('billingSettings', {
  1259→      ...DEFAULT_BILLING_SETTINGS,
  1260→      ...billingSettings,
  1261→      userId,
  1262→      updatedAt: new Date().toISOString(),
  1263→    } as OfflineDBSchema['billingSettings']['value']);
  1264→  }
  1265→
  1266→  return { message: 'Backup imported successfully' };
  1267→}
