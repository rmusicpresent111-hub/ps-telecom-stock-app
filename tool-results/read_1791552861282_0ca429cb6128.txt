     1→/**
     2→ * Local-First Service Layer — 100% offline, NO cloud database.
     3→ *
     4→ * READ STRATEGY:  Always served from local IndexedDB (instant, works with zero internet).
     5→ * WRITE STRATEGY: Written to local IndexedDB immediately (atomic where it matters).
     6→ * DELETE STRATEGY: Deleted from local IndexedDB (with cascades where required).
     7→ *
     8→ * Stock transactions are applied atomically (products + transactions stores in a
     9→ * single IndexedDB transaction) so rapid taps can never create negative stock.
    10→ * Profit/report math uses the purchase-price SNAPSHOT captured on each transaction,
    11→ * so later price edits never rewrite historical profit.
    12→ */

    31→// ============ DATE HELPERS (local timezone — never use UTC for business days) ============
    32→
    33→/**
    34→ * Returns YYYY-MM-DD for a Date in LOCAL time.
    35→ * toISOString() uses UTC — for IST (UTC+5:30) sales between 00:00–05:30 local
    36→ * would land on the previous day. This helper fixes that class of bugs.
    37→ */
    38→export function localDateStr(d: Date = new Date()): string {
    39→  const y = d.getFullYear();
    40→  const m = String(d.getMonth() + 1).padStart(2, '0');
    41→  const day = String(d.getDate()).padStart(2, '0');
    42→  return `${y}-${m}-${day}`;
    43→}

    45→/**
    46→ * Resolves a reporting period to a [from, to] date range (local time).
    47→ */
    48→function resolvePeriodRange(options?: { period?: string; date?: string; from?: string; to?: string }): { from: string; to: string } {
    49→  const now = new Date();
    50→  if (options?.from && options?.to) {
    51→    // Guard against inverted custom ranges (from > to) — silently swap.
    52→    return options.from <= options.to
    53→      ? { from: options.from, to: options.to }
    54→      : { from: options.to, to: options.from };
    55→  }
    56→  if (options?.date) return { from: options.date, to: options.date };
    57→
    58→  const todayStr = localDateStr(now);
    59→  const period = options?.period || 'today';
    60→  if (period === 'week') {
    61→    const weekStart = new Date(now);
    62→    const day = weekStart.getDay();
    63→    const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1); // Monday start
    64→    weekStart.setDate(diff);
    65→    return { from: localDateStr(weekStart), to: todayStr };
    66→  }
    67→  if (period === 'month') {
    68→    return { from: `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`, to: todayStr };
    69→  }
    70→  // today (default) or any unknown period
    71→  return { from: todayStr, to: todayStr };
    72→}

    76→function stripSync<T>(row: Record<string, unknown>): T {
    77→  const { _synced: _s, _dirty: _d, ...rest } = row;
    78→  return rest as T;
    79→}

    81→function sortTxnsDesc(rows: OfflineDBSchema['transactions']['value'][]): OfflineDBSchema['transactions']['value'][] {
    82→  return [...rows].sort((a, b) => {
    83→    if (a.date !== b.date) return b.date.localeCompare(a.date);
    84→    return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    85→  });
    86→}
    87→
    88→function localTxnsToTransactions(localTxns: OfflineDBSchema['transactions']['value'][]): Transaction[] {
    89→  return sortTxnsDesc(localTxns).map(t => stripSync<Transaction>(t));
    90→}

    92→/** Cost basis for a transaction: the SNAPSHOT taken at sale time, falling back to the current product. */
    93→function txnCostBasis(t: Transaction, products: Product[]): number {
    94→  return t.product?.purchasePrice ?? products.find(p => p.id === t.productId)?.purchasePrice ?? 0;
    95→}

   188→/** Only these keys may be changed by an update — never trust raw spreads. */
   189→function applyProductUpdates(
   190→  product: OfflineDBSchema['products']['value'],
   191→  updates: Partial<Product>
   192→): OfflineDBSchema['products']['value'] {
   193→  const next = { ...product };
   194→  if (updates.name !== undefined) {
   195→    const name = String(updates.name).trim();
   196→    if (name) next.name = name;
   197→  }
   198→  if (updates.categoryId !== undefined) next.categoryId = updates.categoryId;
   199→  if (updates.boxNumber !== undefined) next.boxNumber = updates.boxNumber;
   200→  if (updates.quantity !== undefined) {
   201→    next.quantity = Math.max(0, Math.floor(Number(updates.quantity)) || 0);
   202→  }
   203→  if (updates.purchasePrice !== undefined) {
   204→    next.purchasePrice = Math.max(0, Number(updates.purchasePrice)) || 0;
   205→  }
   206→  if (updates.sellingPrice !== undefined) {
   207→    next.sellingPrice = Math.max(0, Number(updates.sellingPrice)) || 0;
   208→  }
   209→  if (updates.lowStockThreshold !== undefined) {
   210→    next.lowStockThreshold = Math.max(0, Math.floor(Number(updates.lowStockThreshold)) || 0);
   211→  }
   212→  return next;
   213→}

   237→export async function deleteProductOffline(id: string, userId: string): Promise<{ message: string }> {
   238→  // Ownership check — a product must never be deleted by a different account
   239→  // on a shared device. The cascade (related transactions + the product) runs
   240→  // in ONE IndexedDB transaction so a crash can never leave the product deleted
   241→  // but its transactions orphaned, or the other way around.
   242→  const db = await getOfflineDB();
   243→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   244→  const product = await tx.objectStore('products').get(id);
   245→  if (!product || (userId && product.userId !== userId)) throw new Error('Not found');
   246→
   247→  // Delete related transactions first (cascade)
   248→  const localTxns = await tx.objectStore('transactions').index('by-userId').getAll(userId);
   249→  for (const txn of localTxns) {
   250→    if (txn.productId === id) void tx.objectStore('transactions').delete(txn.id);
   251→  }
   252→
   253→  void tx.objectStore('products').delete(id);
   254→  await tx.done;
   255→
   256→  // Cloud: drop the product AND its cascaded transactions.
   257→  markDirty(userId, {
   258→    table: 'ps_products',
   259→    deleted: [id],
   260→    changed: [],
   261→  });
   262→  const removedTxnIds = localTxns.filter(t => t.productId === id).map(t => t.id);
   263→  if (removedTxnIds.length > 0) {
   264→    markDirty(userId, { table: 'ps_transactions', deleted: removedTxnIds });
   265→  }
   266→
   267→  return { message: 'Product deleted successfully' };
   268→}

   270→// ============ TRANSACTIONS (Local, ATOMIC stock updates) ============
   271→
   272→/**
   273→ * Creates a transaction AND applies the stock change in ONE atomic IndexedDB
   274→ * transaction. Two rapid sells can never both pass the same stock check.
   275→ *
   276→ * NOTE: idb v8's db.transaction() has NO callback parameter — the correct
   277→ * atomic pattern is: open tx → await requests → `await tx.done`.
   278→ */
   279→export async function createTransactionOffline(transactionData: {
   280→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   281→  productId: string;
   282→  quantity: number;
   283→  unitPrice?: number;
   284→  totalAmount?: number;
   285→  date?: string;
   286→  userId: string;
   287→}): Promise<{ transaction: Transaction }> {
   288→  const quantity = Math.floor(Number(transactionData.quantity));
   289→  if (!Number.isFinite(quantity) || quantity <= 0) throw new Error('Quantity must be a positive whole number');
   290→
   291→  const unitPrice = Number(transactionData.unitPrice ?? 0);
   292→  const totalAmount = Number(transactionData.totalAmount ?? 0);
   293→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   294→  if (!Number.isFinite(totalAmount) || totalAmount < 0) throw new Error('Total amount cannot be negative');
   295→
   296→  const db = await getOfflineDB();
   297→  const now = new Date().toISOString();
   298→  const id = generateRecordId();
   299→
   300→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   301→  const productsStore = tx.objectStore('products');
   302→  const transactionsStore = tx.objectStore('transactions');
   303→
   304→  // Validate BEFORE writing anything — a readwrite tx with no writes commits empty.
   305→  const product = await productsStore.get(transactionData.productId);
   306→  if (!product) {
   307→    throw new Error('Product not found');
   308→  }
   309→  // Ownership check — a second account on a shared device must never be able
   310→  // to move stock on (or write transactions against) someone else's product.
   311→  // Generic message, same as the not-found case (no enumeration).
   312→  if (product.userId !== transactionData.userId) {
   313→    throw new Error('Product not found');
   314→  }
   315→
   316→  const isOut = transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL';
   317→  if (isOut && product.quantity < quantity) {
   318→    throw new Error(`Insufficient stock — only ${product.quantity} left`);
   319→  }
   320→
   321→  const quantityChange = transactionData.type === 'STOCK_IN' ? quantity : -quantity;
   322→
   323→  const row: OfflineDBSchema['transactions']['value'] = {
   324→    id,
   325→    productId: transactionData.productId,
   326→    type: transactionData.type,
   327→    quantity,
   328→    unitPrice,
   329→    totalAmount,
   330→    date: transactionData.date || localDateStr(),
   331→    userId: transactionData.userId,
   332→    createdAt: now,
   333→    // Snapshot the product prices at transaction time — historical profit must
   334→    // never change when the user later edits the product's prices.
   335→    product: {
   336→      id: product.id,
   337→      name: product.name,
   338→      purchasePrice: product.purchasePrice,
   339→      sellingPrice: product.sellingPrice,
   340→    },
   341→    _synced: Date.now(),
   342→    _dirty: 0,
   343→  };
   344→
   345→  await transactionsStore.put(row);
   346→  await productsStore.put({
   347→    ...product,
   348→    quantity: product.quantity + quantityChange,
   349→    updatedAt: now,
   350→  });
   351→
   352→  // Commit — rejects (and rolls back both stores) if anything failed.
   353→  await tx.done;
   354→
   355→  // Cloud auto-sync: the stock change AND the transaction itself.
   356→  markDirty(transactionData.userId, {
   357→    table: 'ps_products',
   358→    changed: [transactionData.productId],
   359→  });
   360→  markDirty(transactionData.userId, { table: 'ps_transactions', changed: [id] });
   361→
   362→  return { transaction: stripSync<Transaction>(row) };
   363→}

   365→export async function getTransactionsOffline(userId: string, options?: { type?: string; productId?: string; from?: string; to?: string }): Promise<Transaction[]> {
   366→  let localTxns = await offlineTransactions.getAll(userId);
   367→
   368→  if (options?.type) localTxns = localTxns.filter(t => t.type === options.type);
   369→  if (options?.productId) localTxns = localTxns.filter(t => t.productId === options.productId);
   370→  if (options?.from) localTxns = localTxns.filter(t => t.date >= options.from!);
   371→  if (options?.to) localTxns = localTxns.filter(t => t.date <= options.to!);
   372→
   373→  return localTxnsToTransactions(localTxns);
   374→}

   378→export async function getDashboardOffline(userId: string) {
   379→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   380→    offlineProducts.getAll(userId),
   381→    offlineCategories.getAll(userId),
   382→    offlineTransactions.getAll(userId),
   383→  ]);
   384→
   385→  const products = localProducts.map(p => stripSync<Product>(p));
   386→  const categories = localCategories.map(c => stripSync<Category>(c));
   387→  const transactions = localTxnsToTransactions(localTransactions);
   388→
   389→  const today = localDateStr();
   390→  const todayTxns = transactions.filter(t => t.date === today);
   391→
   392→  const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   393→  const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   394→  const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   395→  const recentTransactions = transactions.slice(0, 5);
   396→
   397→  const productCountMap: Record<string, number> = {};
   398→  for (const p of products) {
   399→    productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   400→  }
   401→  const catsWithCount = categories.map(c => ({
   402→    ...c,
   403→    _count: { products: productCountMap[c.id] || 0 },
   404→  }));
   405→
   406→  const sevenDaysAgo = new Date();
   407→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   408→  const sevenDaysAgoStr = localDateStr(sevenDaysAgo);
   409→
   410→  const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   411→  const saleMap = new Map<string, { sales: number; quantity: number }>();
   412→  for (const t of saleTxns) {
   413→    const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   414→    existing.sales += t.totalAmount;
   415→    existing.quantity += t.quantity;
   416→    saleMap.set(t.date, existing);
   417→  }
   418→
   419→  const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   420→  for (let i = 6; i >= 0; i--) {
   421→    const d = new Date();
   422→    d.setDate(d.getDate() - i);
   423→    const dateStr = localDateStr(d);
   424→    const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   425→    const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   426→    saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   427→  }
   428→
   429→  const stockByCategory = new Map<string, { quantity: number; value: number }>();
   430→  for (const p of products) {
   431→    const cat = catsWithCount.find(c => c.id === p.categoryId);
   432→    const catName = cat?.name || 'Unknown';
   433→    const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   434→    existing.quantity += p.quantity;
   435→    existing.value += p.quantity * p.purchasePrice;
   436→    stockByCategory.set(catName, existing);
   437→  }
   438→
   439→  const stockOverview = Array.from(stockByCategory.entries())
   440→    .filter(([_, data]) => data.quantity > 0)
   441→    .map(([category, data]) => ({ category, ...data }));
   442→
   443→  return {
   444→    stats: {
   445→      totalItems: products.length,
   446→      lowItems,
   447→      todayTransactions: todayTxns.length,
   448→      stockValue,
   449→    },
   450→    lowStockProducts,
   451→    recentTransactions,
   452→    categories: catsWithCount,
   453→    saleOverview,
   454→    stockOverview,
   455→  };
   456→}

   463→export async function getExpensesOffline(userId: string, options?: { period?: string; date?: string; from?: string; to?: string; category?: string }) {
   464→  let localExpenses = await offlineExpenses.getAll(userId);
   465→
   466→  // Honor the reporting period exactly — otherwise "Today" would show all-time totals.
   467→  const { from, to } = resolvePeriodRange(options);
   468→  localExpenses = localExpenses.filter(e => e.date >= from && e.date <= to);
   469→  if (options?.category) localExpenses = localExpenses.filter(e => e.category === options.category);
   470→
   471→  const expenses = localExpenses
   472→    .map(e => stripSync<Expense>(e))
   473→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   474→  const totalExpense = expenses.reduce((sum, e) => sum + (e.amount || 0), 0);
   475→  const byCategory = expenses.reduce((acc, e) => {
   476→    const cat = e.category || 'other';
   477→    acc[cat] = (acc[cat] || 0) + (e.amount || 0);
   478→    return acc;
   479→  }, {} as Record<string, number>);
   480→
   481→  return {
   482→    expenses,
   483→    summary: { totalExpense, byCategory, count: expenses.length },
   484→  };
   485→}

   520→export async function updateExpenseOffline(id: string, updates: { amount?: number; category?: string; description?: string }, userId: string): Promise<{ expense: Expense }> {
   521→  const allExpenses = await offlineExpenses.getAll(userId);
   522→  const existing = allExpenses.find(e => e.id === id);
   523→  if (!existing) throw new Error('Expense not found');
   524→
   525→  if (updates.amount !== undefined) {
   526→    const amount = Number(updates.amount);
   527→    if (!Number.isFinite(amount) || amount <= 0) throw new Error('Expense amount must be greater than 0');
   528→  }
   529→
   530→  const updatedExpense = {
   531→    ...existing,
   532→    ...(updates.amount !== undefined ? { amount: Number(updates.amount) } : {}),

   701→  if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   702→
   703→  if (type === 'stock-value') {
   704→    const stockValueData = categories.map(cat => {
   705→      const catProducts = products.filter(p => p.categoryId === cat.id);
   706→      const totalQty = catProducts.reduce((sum, p) => sum + p.quantity, 0);
   707→      const totalPurchaseValue = catProducts.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   708→      const totalSellingValue = catProducts.reduce((sum, p) => sum + p.quantity * p.sellingPrice, 0);
   709→      const totalProfit = totalSellingValue - totalPurchaseValue;
   710→      const lowStockCount = catProducts.filter(p => p.quantity <= p.lowStockThreshold).length;
   711→

   713→        categoryId: cat.id,
   714→        categoryName: cat.name,
   715→        categoryImage: cat.image,

   836→      revenue: number; cost: number; profit: number;

   878→        totalSell: values.reduce((s, d) => s + d.sell, 0),
   879→        totalQuantity: values.reduce((s, d) => s + d.quantity, 0),

   881→    };
   882→  }
   883→
   884→  return { type, data: [], grandTotal: {} };

  1006→function clampNumber(value: number, min: number, max: number): number {
  1007→  const num = Number(value);
  1008→  if (!Number.isFinite(num)) return min;
  1009→  return Math.min(max, Math.max(min, num));
  1010→}

  1012→/**
  1013→ * Generates the next sequential bill number: PREFIX-YYMM-0001.
  1014→ * Runs against rows already fetched inside the SAME transaction that will save
  1015→ * the new bill (no read-then-write race), and skips any number that is already
  1016→ * taken (e.g. numbers that arrived via a restore/import) so duplicates are
  1017→ * impossible.
  1018→ */
  1019→function nextBillNumberFromRows(
  1020→  bills: OfflineDBSchema['bills']['value'][],
  1021→  prefix: string
  1022→): string {
  1023→  const now = new Date();
  1024→  const yy = String(now.getFullYear()).slice(-2);
  1025→  const mm = String(now.getMonth() + 1).padStart(2, '0');
  1026→  const monthPrefix = `${prefix}-${yy}${mm}-`;
  1027→  let maxSeq = 0;
  1028→  const taken = new Set<string>();
  1029→  for (const b of bills) {
  1030→    if (typeof b.billNumber === 'string' && b.billNumber.startsWith(monthPrefix)) {
  1031→      taken.add(b.billNumber);
  1032→      const seq = parseInt(b.billNumber.slice(monthPrefix.length), 10);
  1033→      if (Number.isFinite(seq) && seq > maxSeq) maxSeq = seq;
  1034→    }
  1035→  }
  1036→  let seq = maxSeq + 1;
  1037→  let candidate = `${monthPrefix}${String(seq).padStart(4, '0')}`;
  1038→  while (taken.has(candidate)) {
  1039→    seq += 1;
  1040→    candidate = `${monthPrefix}${String(seq).padStart(4, '0')}`;
  1041→  }
  1042→  return candidate;
  1043→}

  1045→export async function createBillOffline(data: {
  1046→  userId: string;
  1047→  transactionId: string;
  1048→  productId: string;
  1049→  productName: string;
  1050→  quantity: number;
  1051→  unitPrice: number;
  1052→  customerName?: string;
  1053→  customerMobile?: string;
  1054→  discountType?: 'amount' | 'percent';
  1055→  discountValue?: number;
  1056→  paymentMethod?: PaymentMethod;
  1057→  paidAmount?: number;
  1058→  note?: string;
  1059→  date?: string;
  1060→}): Promise<{ bill: Bill }> {
  1061→  const qty = Math.floor(Number(data.quantity));
  1062→  if (!Number.isFinite(qty) || qty <= 0) throw new Error('Quantity must be a positive whole number');
  1063→  const unitPrice = Number(data.unitPrice);
  1064→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
  1065→
  1066→  const settings = await getBillingSettingsOffline(data.userId);
  1067→
  1068→  const discountType = data.discountType === 'percent' ? 'percent' : 'amount';
  1069→  let discountValue = Math.max(0, Number(data.discountValue ?? 0) || 0);
  1070→  const subtotal = round2(qty * unitPrice);
  1071→
  1072→  let discountAmount = 0;
  1073→  if (discountType === 'percent') {
  1074→    discountValue = clampNumber(discountValue, 0, 100);
  1075→    discountAmount = round2((subtotal * discountValue) / 100);
  1076→  } else {
  1077→    discountAmount = round2(Math.min(discountValue, subtotal));
  1078→  }
  1079→
  1080→  const gstEnabled = !!settings.gstEnabled && !!settings.gstNumber.trim() && settings.gstRate > 0;
  1081→  const gstAmount = gstEnabled ? round2(((subtotal - discountAmount) * settings.gstRate) / 100) : 0;
  1082→  const total = round2(subtotal - discountAmount + gstAmount);
  1083→
  1084→  const paymentMethod: PaymentMethod = data.paymentMethod ?? 'cash';
  1085→  let paidAmount = data.paidAmount === undefined ? total : Math.max(0, Number(data.paidAmount) || 0);
  1086→  if (paymentMethod === 'due') paidAmount = 0;
  1087→  paidAmount = round2(Math.min(paidAmount, total));
  1088→  const dueAmount = round2(total - paidAmount);
  1089→
  1090→  const now = new Date().toISOString();
  1091→
  1092→  // Atomic bill creation: the duplicate-transaction guard, the bill-number
  1093→  // allocation and the save run inside ONE IndexedDB transaction, so a
  1094→  // double-tap or racing call can never mint two bills for one stock-out
  1095→  // transaction or allocate the same number twice.
  1096→  const db = await getOfflineDB();
  1097→  const tx = db.transaction('bills', 'readwrite');
  1098→  const billsOs = tx.objectStore('bills');
  1099→  const existingBills = await billsOs.index('by-userId').getAll(data.userId);
  1100→
  1101→  const dupe = existingBills.find(b => b.transactionId === data.transactionId);
  1102→  if (dupe) throw new Error('A bill already exists for this transaction');
  1103→
  1104→  const billNumber = nextBillNumberFromRows(existingBills, settings.billPrefix.trim() || 'PS');
  1105→
  1106→  const bill: OfflineDBSchema['bills']['value'] = {
  1107→    id: generateRecordId(),
  1108→    userId: data.userId,
  1109→    billNumber,
  1110→    customerName: (data.customerName ?? '').trim() || 'Walk-in Customer',
  1111→    customerMobile: (data.customerMobile ?? '').replace(/[^\d+]/g, ''),
  1112→    items: [
  1113→      {
  1114→        productId: data.productId,
  1115→        name: data.productName,
  1116→        quantity: qty,
  1117→        unitPrice,
  1118→        total: subtotal,
  1119→      },
  1120→    ],
  1121→    subtotal,
  1122→    discountValue,
  1123→    discountType,
  1124→    discountAmount,
  1125→    gstEnabled,
  1126→    gstRate: gstEnabled ? settings.gstRate : 0,
  1127→    gstAmount,
  1128→    total,
  1129→    paymentMethod,
  1130→    paidAmount,
  1131→    dueAmount,
  1132→    note: (data.note ?? '').trim(),
  1133→    transactionId: data.transactionId,
  1134→    shopSnapshot: {
  1135→      name: settings.shopName,
  1136→      proprietorName: settings.proprietorName,
  1137→      address: settings.shopAddress,
  1138→      phone: settings.shopPhone,
  1139→      gstNumber: settings.gstNumber,
  1140→    },
  1141→    date: data.date || localDateStr(),
  1142→    createdAt: now,
  1143→    updatedAt: now,
  1144→  };
  1145→
  1146→  void billsOs.put(bill);
  1147→  await tx.done;
  1148→
  1149→  markDirty(data.userId, { table: 'ps_bills', changed: [bill.id] });
  1150→
  1151→  return { bill: bill as unknown as Bill };
  1152→}

  1155→  return Math.round(n * 100) / 100;
  1156→}

  1158→export async function getBillsOffline(userId: string, options?: { from?: string; to?: string; search?: string }): Promise<Bill[]> {
  1159→  let rows = await offlineBills.getAll(userId);
  1160→  if (options?.from) rows = rows.filter(b => b.date >= options.from!);
  1161→  if (options?.to) rows = rows.filter(b => b.date <= options.to!);
  1162→  if (options?.search) {
  1163→    const q = options.search.toLowerCase();
  1164→    rows = rows.filter(b =>
  1165→      b.customerName.toLowerCase().includes(q) ||
  1166→      b.customerMobile.includes(q) ||
  1167→      b.billNumber.toLowerCase().includes(q) ||
  1168→      b.items.some(i => i.name.toLowerCase().includes(q))
  1169→    );
  1170→  }
  1171→  return rows
  1172→    .sort((a, b) => {
  1173→      if (a.date !== b.date) return b.date.localeCompare(a.date);
  1174→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
  1175→    })
  1176→    .map(b => b as unknown as Bill);
  1177→}

  1225→export async function importBackupOffline(userId: string, backupData: {
  1226→  categories: unknown[];
  1227→  products: unknown[];
  1228→  transactions: unknown[];
  1229→  expenses?: unknown[];
  1230→  cashEntries?: unknown[];
  1231→  serviceTransactions?: unknown[];
  1232→  bills?: unknown[];
  1233→  billingSettings?: unknown;
  1234→}) {
  1235→  const now = Date.now();
  1236→  const nowIso = new Date().toISOString();
  1237→
  1238→  // ---- Numeric coercion helpers (same clamps the create* functions use) ----
  1239→  const toNum = (v: unknown): number => {
  1240→    const n = Number(v);
  1241→    return Number.isFinite(n) ? n : 0;
  1242→  };
  1243→  /** Non-negative number — mirrors Math.max(0, Number(x)) || 0 in create*. */
  1244→  const numPos = (v: unknown): number => Math.max(0, toNum(v));
  1245→  /** Non-negative whole number — mirrors Math.max(0, Math.floor(Number(x))) || 0. */
  1246→  const intPos = (v: unknown): number => Math.max(0, Math.floor(toNum(v)) || 0);
  1247→  /** Coerces the listed numeric fields (only when present) on a raw backup row. */
  1248→  const coerce = (row: Record<string, unknown>, fields: Record<string, 'int' | 'num'>): Record<string, unknown> => {
  1249→    const out: Record<string, unknown> = { ...row };
  1250→    for (const [key, kind] of Object.entries(fields)) {
  1251→      if (key in out) out[key] = kind === 'int' ? intPos(out[key]) : numPos(out[key]);
  1252→    }
  1253→    return out;
  1254→  };
  1255→  const hasName = (row: Record<string, unknown>): boolean =>
  1256→    typeof row.name === 'string' && row.name.trim() !== '';
  1257→  /** A hand-edited backup with date: 20251012 (number) would crash every
  1258→   *  localeCompare()/date filter later — force date-ish fields to strings. */
  1259→  const isoStr = (v: unknown): string => (typeof v === 'string' ? v : v == null ? '' : String(v));
  1260→  /** Normalizes any date-ish field that is PRESENT in the row to a string. */
  1261→  const fixDates = (row: Record<string, unknown>): Record<string, unknown> => {
  1262→    const out: Record<string, unknown> = { ...row };
  1263→    if (out.date !== undefined) out.date = isoStr(out.date);
  1264→    if (out.createdAt !== undefined) out.createdAt = isoStr(out.createdAt);
  1265→    if (out.updatedAt !== undefined) out.updatedAt = isoStr(out.updatedAt);
  1266→    return out;
  1267→  };
  1268→
  1269→  // Validate rows: every row MUST have an id — malformed backups must never
  1270→  // be written into IndexedDB.
  1271→  const asRows = (rows: unknown[]): Record<string, unknown>[] => {
  1272→    if (!Array.isArray(rows)) return [];
  1273→    return rows.filter((r): r is Record<string, unknown> =>
  1274→      !!r && typeof r === 'object' && typeof (r as Record<string, unknown>).id === 'string' && (r as Record<string, unknown>).id !== ''
  1275→    );
  1276→  };
  1277→
  1278→  const categories = asRows(backupData.categories || []).filter(hasName);
  1279→  const products = asRows(backupData.products || []).filter(hasName);
  1280→  const transactions = asRows(backupData.transactions || []);
  1281→  const expenses = asRows(backupData.expenses || []);
  1282→  const cashEntries = asRows(backupData.cashEntries || []);
  1283→  const serviceTransactions = asRows(backupData.serviceTransactions || []);
  1284→  const bills = asRows(backupData.bills || []);
  1285→  const billingSettings = (backupData.billingSettings && typeof backupData.billingSettings === 'object')
  1286→    ? backupData.billingSettings as Record<string, unknown>
  1287→    : null;
  1288→
  1289→  if (categories.length === 0 && products.length === 0 && transactions.length === 0 && expenses.length === 0 && cashEntries.length === 0 && serviceTransactions.length === 0 && bills.length === 0 && !billingSettings) {
  1290→    // A settings-only backup (fresh shop, no rows yet) is legitimate — only
  1291→    // reject a file that has literally nothing usable.
  1292→    throw new Error('Backup file contains no valid data');
  1293→  }
  1294→
  1295→  // ---- REPLACE semantics, ATOMICALLY ----
  1296→  // The old rows and the imported rows swap inside ONE IndexedDB transaction
  1297→  // spanning every store: a crash or quota error mid-restore can never leave
  1298→  // the device half-wiped or half-imported. (Previously the wipe and the write
  1299→  // ran in separate transactions — a failure in between destroyed local data.)
  1300→
  1301→  // 1. Prepare all validated + coerced rows first (pure data, no I/O) so the
  1302→  //    transaction itself stays short.
  1303→  const catRows: OfflineDBSchema['categories']['value'][] = categories.map(
  1304→    (cat) => ({ ...fixDates(cat), userId, _synced: now, _dirty: 0 }) as OfflineDBSchema['categories']['value']
  1305→  );
  1306→  const prodRows: OfflineDBSchema['products']['value'][] = products.map(
  1307→    (prod) => coerce({ ...fixDates(prod), userId, _synced: now, _dirty: 0 }, {
  1308→      quantity: 'int',
  1309→      purchasePrice: 'num',
  1310→      sellingPrice: 'num',
  1311→      lowStockThreshold: 'int',
  1312→    }) as OfflineDBSchema['products']['value']
  1313→  );
  1314→  const txnRows: OfflineDBSchema['transactions']['value'][] = transactions.map(
  1315→    (txn) => coerce({ ...fixDates(txn), date: isoStr(txn.date), userId, _synced: now, _dirty: 0 }, {
  1316→      quantity: 'int',
  1317→      unitPrice: 'num',
  1318→      totalAmount: 'num',
  1319→    }) as OfflineDBSchema['transactions']['value']
  1320→  );
  1321→  const expRows: OfflineDBSchema['expenses']['value'][] = expenses.map(
  1322→    (exp) => coerce({ ...fixDates(exp), date: isoStr(exp.date), userId, _synced: now, _dirty: 0 }, { amount: 'num' }) as OfflineDBSchema['expenses']['value']
  1323→  );
  1324→  const cashRows: OfflineDBSchema['cashEntries']['value'][] = cashEntries.map(
  1325→    (entry) => coerce({ ...fixDates(entry), date: isoStr(entry.date), userId, _synced: now, _dirty: 0 }, {
  1326→      handCash: 'num',
  1327→      liquidCash: 'num',
  1328→    }) as OfflineDBSchema['cashEntries']['value']
  1329→  );
  1330→  const svcRows: OfflineDBSchema['serviceTransactions']['value'][] = serviceTransactions.map(
  1331→    (svc) => coerce({ ...fixDates(svc), date: isoStr(svc.date), userId, _synced: now, _dirty: 0 }, { amount: 'num' }) as OfflineDBSchema['serviceTransactions']['value']
  1332→  );
  1333→  const billRows: OfflineDBSchema['bills']['value'][] = bills.map(
  1334→    (bill) => coerce({
  1335→      ...fixDates(bill),
  1336→      date: isoStr(bill.date),
  1337→      userId,
  1338→      createdAt: isoStr(bill.createdAt) || nowIso,
  1339→      updatedAt: isoStr(bill.updatedAt) || nowIso,
  1340→    }, {
  1341→      subtotal: 'num',
  1342→      discountValue: 'num',
  1343→      discountAmount: 'num',
  1344→      gstRate: 'num',
  1345→      gstAmount: 'num',
  1346→      total: 'num',
  1347→      paidAmount: 'num',
  1348→      dueAmount: 'num',
  1349→    }) as OfflineDBSchema['bills']['value']
  1350→  );
  1351→
  1352→  // 2. One readwrite transaction across ALL stores: wipe this user's rows,
  1353→  //    write the imported ones, swap billingSettings — all or nothing.
  1354→  const db = await getOfflineDB();
  1355→  const tx = db.transaction(
  1356→    ['categories', 'products', 'transactions', 'expenses', 'cashEntries', 'serviceTransactions', 'bills', 'billingSettings'],
  1357→    'readwrite'
  1358→  );
  1359→
  1360→  for (const store of ['categories', 'products', 'transactions', 'expenses', 'cashEntries', 'serviceTransactions', 'bills'] as const) {
  1361→    const os = tx.objectStore(store);
  1362→    const keys = await os.index('by-userId').getAllKeys(userId);
  1363→    for (const key of keys) void os.delete(key);
  1364→  }
  1365→
  1366→  for (const row of catRows) void tx.objectStore('categories').put(row);
  1367→  for (const row of prodRows) void tx.objectStore('products').put(row);
  1368→  for (const row of txnRows) void tx.objectStore('transactions').put(row);
  1369→  for (const row of expRows) void tx.objectStore('expenses').put(row);
  1370→  for (const row of cashRows) void tx.objectStore('cashEntries').put(row);
  1371→  for (const row of svcRows) void tx.objectStore('serviceTransactions').put(row);
  1372→  for (const row of billRows) void tx.objectStore('bills').put(row);
  1373→
  1374→  if (billingSettings) {
  1375→    void tx.objectStore('billingSettings').put({
  1376→      ...DEFAULT_BILLING_SETTINGS,
  1377→      ...billingSettings,
  1378→      userId,
  1379→      updatedAt: nowIso,
  1380→    } as OfflineDBSchema['billingSettings']['value']);
  1381→  }
  1382→
  1383→  // Commits everything or nothing — any failed request rejects and rolls back.
  1384→  await tx.done;
  1385→
  1386→  // Local data now mirrors the restored snapshot exactly — any pending
  1387→  // auto-sync deltas are stale (they'd re-push identical rows at best).
  1388→  clearDirty(userId);
  1389→
  1390→  return { message: 'Backup imported successfully' };
  1391→}
