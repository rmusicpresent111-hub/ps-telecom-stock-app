    12→    12→} from '@/lib/bill-pdf';
    13→    13→import { BillingSettings, Bill, PaymentMethod } from '@/lib/types';
    14→    14→import { ArrowLeft, FileText, Loader2, Trash2, Download, Share2, MessageCircle, ExternalLink, BadgeIndianRupee } from 'lucide-react';
    15→    15→import { toast } from 'sonner';

    17→    17→
    18→    18→type Mode = 'create' | 'view' | 'success';
    19→    19→
    20→    20→export default function InvoiceScreen() {

    22→    22→  const language = useAppStore(s => s.language);
    23→    23→  const goBack = useAppStore(s => s.goBack);
    24→    24→  const pendingBillData = useAppStore(s => s.pendingBillData);
    25→    25→  const setPendingBillData = useAppStore(s => s.setPendingBillData);

    27→    27→  const setSelectedBillId = useAppStore(s => s.setSelectedBillId);
    28→    28→  const navigateToTab = useAppStore(s => s.navigateToTab);

    30→    30→

    32→    32→  const [settings, setSettings] = useState<BillingSettings | null>(null);
    33→    33→  const [bill, setBill] = useState<Bill | null>(null);
    34→    34→  const [loading, setLoading] = useState(true);
    35→    35→  const [saving, setSaving] = useState(false);
    36→    36→  const [generating, setGenerating] = useState(false);
    37→    37→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false);
    38→    38→
    39→    39→  // create-mode form
    40→    40→  const [customerName, setCustomerName] = useState('');
    41→    41→  const [customerMobile, setCustomerMobile] = useState('');
    42→    42→  const [discountType, setDiscountType] = useState<'amount' | 'percent'>('amount');
    43→    43→  const [discountValue, setDiscountValue] = useState('');
    44→    44→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash');
    45→    45→  const [paidAmount, setPaidAmount] = useState('');
    46→    46→  const [note, setNote] = useState('');
    47→    47→  const initializedDiscount = useRef(false);
    48→    48→

    60→    60→        if (selectedBillId) {
    61→    61→          const existing = await getBillByIdOffline(selectedBillId, user.id);
    62→    62→          if (cancelled) return;

    74→    74→          // nothing to show — leave

    76→    76→        }
    77→    77→      } catch {
    78→    78→        if (!cancelled) goBack();
    79→    79→      } finally {
    80→    80→        if (!cancelled) setLoading(false);
    81→    81→      }

    84→    84→     
    85→    85→  }, [user?.id]);
    86→    86→

    88→    88→  useEffect(() => {
    89→    89→    if (!settings || initializedDiscount.current) return;

    92→    92→      setDiscountType('percent');
    93→    93→      setDiscountValue(String(settings.defaultDiscountPercent));

    95→    95→  }, [settings]);

   100→   100→    const unitPrice = pendingBillData?.unitPrice ?? 0;
   101→   101→    const subtotal = Math.round(qty * unitPrice * 100) / 100;
   102→   102→    let discountAmount = 0;
   103→   103→    const dv = Math.max(0, Number(discountValue) || 0);
   104→   104→    if (discountType === 'percent') {
   105→   105→      discountAmount = Math.round(((subtotal * Math.min(dv, 100)) / 100) * 100) / 100;
   106→   106→    } else {
   107→   107→      discountAmount = Math.round(Math.min(dv, subtotal) * 100) / 100;
   108→   108→    }

   110→   110→    const gstAmount = gstEnabled ? Math.round((((subtotal - discountAmount) * (settings?.gstRate ?? 0)) / 100) * 100) / 100 : 0;
   111→   111→    const total = Math.round((subtotal - discountAmount + gstAmount) * 100) / 100;

   113→   113→    paid = Math.min(paid, total);
   114→   114→    const due = Math.round((total - paid) * 100) / 100;
   115→   115→    return { subtotal, discountAmount, gstEnabled, gstAmount, total, paid, due };
   116→   116→  }, [pendingBillData, discountValue, discountType, settings, paymentMethod, paidAmount]);

   121→   121→    try {
   122→   122→      const { bill: created } = await createBillOffline({
   123→   123→        userId: user.id,
   124→   124→        transactionId: pendingBillData.transactionId,
   125→   125→        productId: pendingBillData.productId,
   126→   126→        productName: pendingBillData.productName,
   127→   127→        quantity: pendingBillData.quantity,
   128→   128→        unitPrice: pendingBillData.unitPrice,
   129→   129→        customerName,
   130→   130→        customerMobile,
   131→   131→        discountType,
   132→   132→        discountValue: Number(discountValue) || 0,
   133→   133→        paymentMethod,
   134→   134→        paidAmount: paidAmount === '' ? undefined : Number(paidAmount) || 0,
   135→   135→        note,
   136→   136→      });
   137→   137→      setBill(created);
   138→   138→      setMode('success');
   139→   139→      setPendingBillData(null);
   140→   140→      toast.success(t('billSaved', language));
   141→   141→    } catch (error) {
   142→   142→      toast.error((error as Error).message || t('error', language));
   143→   143→    } finally {
   144→   144→      setSaving(false);
   145→   145→    }
   146→   146→  }, [user?.id, pendingBillData, settings, customerName, customerMobile, discountType, discountValue, paymentMethod, paidAmount, note, setPendingBillData, language]);
   147→   147→
   148→   148→  const makePdf = useCallback(async (targetBill: Bill): Promise<GeneratedPdf | null> => {
   149→   149→    setGenerating(true);
   150→   150→    try {
   151→   151→      return await generateBillPdf(targetBill, {
   152→   152→        signatureDataUrl: settings?.signatureDataUrl,
   153→   153→        qrCodeDataUrl: settings?.qrCodeDataUrl,
   154→   154→        upiId: settings?.upiId,
   155→   155→        thankYouNote: settings?.thankYouNote,
   156→   156→        termsText: settings?.termsText,
   157→   157→      });
   158→   158→    } catch (error) {
   159→   159→      toast.error((error as Error).message || t('error', language));
   160→   160→      return null;
   161→   161→    } finally {
   162→   162→      setGenerating(false);
   163→   163→    }
   164→   164→  }, [settings, language]);
   165→   165→
   166→   166→  const handleWhatsApp = useCallback(async (targetBill: Bill) => {
   167→   167→    const pdf = await makePdf(targetBill);
   168→   168→    if (!pdf) return;
   169→   169→    downloadBlob(pdf.blob, pdf.fileName);
   170→   170→    if (targetBill.customerMobile) {
   171→   171→      window.open(whatsappUrl(targetBill.customerMobile, buildBillMessage(targetBill)), '_blank');
   172→   172→      toast.success(t('shareWhatsappHint', language));
   173→   173→    } else {
   174→   174→      toast.error(t('customerPhone', language) + ' ' + t('error', language));
   175→   175→    }
   176→   176→  }, [makePdf, language]);
   177→   177→
   178→   178→  const handleShare = useCallback(async (targetBill: Bill) => {
   179→   179→    const pdf = await makePdf(targetBill);
   180→   180→    if (!pdf) return;
   181→   181→    const shared = await sharePdfFile(pdf.blob, pdf.fileName, buildBillMessage(targetBill));
   182→   182→    if (!shared) {
   183→   183→      downloadBlob(pdf.blob, pdf.fileName);
   184→   184→      if (targetBill.customerMobile) {
   185→   185→        window.open(whatsappUrl(targetBill.customerMobile, buildBillMessage(targetBill)), '_blank');
   186→   186→        toast.success(t('shareWhatsappHint', language));
   187→   187→      } else {
   188→   188→        toast.success(t('downloadPdf', language));
   189→   189→      }
   190→   190→    }
   191→   191→  }, [makePdf, language]);
   192→   192→
   193→   193→  const handleDownload = useCallback(async (targetBill: Bill) => {
   194→   194→    const pdf = await makePdf(targetBill);
   195→   195→    if (!pdf) return;
   196→   196→    downloadBlob(pdf.blob, pdf.fileName);
   197→   197→    toast.success(t('downloadPdf', language) + ' ✓');
   198→   198→  }, [makePdf, language]);
   199→   199→
   200→   200→  const handleViewPdf = useCallback(async (targetBill: Bill) => {
   201→   201→    const pdf = await makePdf(targetBill);
   202→   202→    if (!pdf) return;
   203→   203→    const url = URL.createObjectURL(pdf.blob);

   212→   212→      toast.success(t('billDeleted', language));
   213→   213→      setHistoryView('bills');

   215→   215→    } catch {
   216→   216→      toast.error(t('error', language));
   217→   217→    }
   218→   218→  }, [user?.id, bill, language, navigateToTab, setHistoryView]);
   219→   219→
   220→   220→  const handleDone = useCallback(() => {
   221→   221→    setHistoryView('bills');
   222→   222→    navigateToTab('history');
   223→   223→  }, [navigateToTab, setHistoryView]);
   224→   224→
   225→   225→  if (loading) {
   226→   226→    return (
   227→   227→      <div className="animated-bg min-h-screen pb-24">
   228→   228→        <div className="max-w-md mx-auto px-4 pt-4">
   229→   229→          <InvoiceHeader goBack={goBack} title={t('eBill', language)} />
   230→   230→          <div className="text-center py-12 text-white/40">{t('loading', language)}</div>
   231→   231→        </div>
   232→   232→      </div>
   233→   233→    );
   234→   234→  }
   235→   235→
   236→   236→  // ============ CREATE MODE ============
   237→   237→  if (mode === 'create' && pendingBillData) {
   238→   238→    const p = pendingBillData;
   239→   239→    return (
   240→   240→      <div className="animated-bg min-h-screen pb-32">
   241→   241→        <div className="max-w-md mx-auto px-4 pt-4">
   242→   242→          <InvoiceHeader goBack={goBack} title={t('createEBill', language)} />
   243→   243→
   244→   244→          {/* Item summary card */}
   245→   245→          <div className="glass-card-strong p-5 mb-4">
   246→   246→            <div className="flex items-start justify-between">
   247→   247→              <div>
   248→   248→                <p className="text-xs text-white/50 mb-1">{t('item', language)}</p>
   249→   249→                <p className="text-base font-bold text-white/90">{p.productName}</p>
   250→   250→              </div>
   251→   251→              <span className="text-[10px] px-2 py-1 rounded-full bg-amber-500/20 text-amber-300 font-semibold">{t('eBill', language)}</span>
   252→   252→            </div>
   253→   253→            <div className="flex items-center justify-between mt-3 pt-3 border-t border-white/10">
   254→   254→              <span className="text-xs text-white/50">{t('quantity', language)}: <span className="text-white/90 font-medium">{p.quantity}</span></span>
   255→   255→              <span className="text-xs text-white/50">{t('price', language)}: <span className="text-white/90 font-medium">₹{formatMoney(p.unitPrice)}</span></span>
   256→   256→              <span className="text-sm font-bold text-emerald-400">₹{formatMoney(p.totalAmount)}</span>
   257→   257→            </div>
   258→   258→          </div>
   259→   259→
   260→   260→          {/* Customer details */}
   261→   261→          <div className="glass-card-strong p-5 space-y-4 mb-4">
   262→   262→            <p className="text-xs font-semibold text-emerald-400 uppercase tracking-wide">{t('customerName', language)}</p>
   263→   263→            <input
   264→   264→              type="text"
   265→   265→              value={customerName}
   266→   266→              onChange={(e) => setCustomerName(e.target.value)}
   267→   267→              placeholder={t('customerName', language)}
   268→   268→              className="glass-input w-full px-4 py-3 text-sm"
   269→   269→            />
   270→   270→            <input
   271→   271→              type="tel"
   272→   272→              value={customerMobile}
   273→   273→              onChange={(e) => setCustomerMobile(e.target.value.replace(/[^\d+]/g, ''))}
   274→   274→              placeholder={t('customerPhone', language) + ' (WhatsApp)'}
   275→   275→              className="glass-input w-full px-4 py-3 text-sm"
   276→   276→            />
   277→   277→            <input
   278→   278→              type="text"
   279→   279→              value={note}
   280→   280→              onChange={(e) => setNote(e.target.value)}
   281→   281→              placeholder={t('note', language)}

   283→   283→            />

   291→   291→                onClick={() => setDiscountType('amount')}
   292→   292→                className={`flex-1 py-2 text-xs font-semibold rounded-xl transition-all ${discountType === 'amount' ? 'neon-btn-solid' : 'glass-card text-white/60'}`}
   293→   293→              >
   294→   294→                ₹ {t('discount', language)}
   295→   295→              </button>
   296→   296→              <button
   297→   297→                onClick={() => setDiscountType('percent')}
   298→   298→                className={`flex-1 py-2 text-xs font-semibold rounded-xl transition-all ${discountType === 'percent' ? 'neon-btn-solid' : 'glass-card text-white/60'}`}
   299→   299→              >
   300→   300→                % {t('discount', language)}
   301→   301→              </button>
   302→   302→            </div>
   303→   303→            <input
   304→   304→              type="number"
   305→   305→              value={discountValue}

   308→   308→              className="glass-input w-full px-4 py-3 text-sm"
   309→   309→            />
   310→   310→
   311→   311→            <p className="text-xs font-semibold text-emerald-400 uppercase tracking-wide pt-2">{t('paymentMethod', language)}</p>
   312→   312→            <div className="grid grid-cols-4 gap-2">
   313→   313→              {(['cash', 'upi', 'card', 'due'] as PaymentMethod[]).map((pm) => (
   314→   314→                <button
   315→   315→                  key={pm}
   316→   316→                  onClick={() => { setPaymentMethod(pm); if (pm === 'due') setPaidAmount('0'); }}
   317→   317→                  className={`py-2.5 text-xs font-semibold rounded-xl transition-all ${paymentMethod === pm ? 'neon-btn-solid' : 'glass-card text-white/60'}`}
   318→   318→                >
   319→   319→                  {t(pm, language)}
   320→   320→                </button>
   321→   321→              ))}
   322→   322→            </div>
   323→   323→            {paymentMethod !== 'due' && (
   324→   324→              <input
   325→   325→                type="number"
   326→   326→                value={paidAmount}
   327→   327→                onChange={(e) => setPaidAmount(e.target.value)}
   328→   328→                placeholder={`${t('paidAmount', language)} (₹${formatMoney(totals.total)})`}
   329→   329→                className="glass-input w-full px-4 py-3 text-sm"
   330→   330→              />
   331→   331→            )}
   332→   332→          </div>
   333→   333→
   334→   334→          {/* Totals preview */}
   335→   335→          <div className="glass-card-strong p-5 space-y-2.5">

   337→   337→            {totals.discountAmount > 0 && (
   338→   338→              <TotalRow label={`${t('discount', language)}${discountType === 'percent' ? ` (${discountValue}%)` : ''}`} value={`-₹${formatMoney(totals.discountAmount)}`} accent="text-green-400" />

   350→   350→            {settings?.shopName && (
   351→   351→              <p className="text-[10px] text-white/30 pt-1">↳ {settings.shopName}{settings.gstEnabled && settings.gstNumber ? ` • GSTIN: ${settings.gstNumber}` : ''}</p>

   366→   366→            </button>
   367→   367→          </div>
   368→   368→        </div>
   369→   369→      </div>

   372→   372→

   374→   374→  if (bill && (mode === 'success' || mode === 'view')) {
   375→   375→    return (
   376→   376→      <div className="animated-bg min-h-screen pb-32">
   377→   377→        <div className="max-w-md mx-auto px-4 pt-4">
   378→   378→          <InvoiceHeader goBack={goBack} title={mode === 'success' ? t('billSaved', language) : t('viewBill', language)} />
   379→   379→
   380→   380→          {/* Receipt-style preview */}
   381→   381→          <motion.div
   382→   382→            initial={{ opacity: 0, y: 12 }}
   383→   383→            animate={{ opacity: 1, y: 0 }}
   384→   384→            className="rounded-2xl overflow-hidden shadow-2xl mb-4 bg-[#12151f] border border-[#D4A853]/30"
   385→   385→          >
   386→   386→            {/* header */}
   387→   387→            <div className="bg-[#181c28] px-5 py-4 border-b-2 border-[#D4A853]">
   388→   388→              <div className="flex justify-between items-start">
   389→   389→                <div>

   395→   395→                  </div>
   396→   396→                </div>
   397→   397→                <div className="text-right">
   398→   398→                  <span className="text-[9px] font-bold bg-[#D4A853] text-black px-2 py-0.5 rounded">{bill.gstEnabled ? 'TAX INVOICE' : 'INVOICE'}</span>
   399→   399→                  <p className="text-[10px] text-white/60 mt-1.5">{bill.billNumber}</p>
