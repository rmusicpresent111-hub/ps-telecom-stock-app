     1→/**
     2→ * Local-First Service Layer — 100% offline, NO cloud database.
     3→ *
     4→ * READ STRATEGY:  Always served from local IndexedDB (instant, works with zero internet).
     5→ * WRITE STRATEGY: Written to local IndexedDB immediately (atomic where it matters).
     6→ * DELETE STRATEGY: Deleted from local IndexedDB (with cascades where required).
     7→ *
     8→ * Stock transactions are applied atomically (products + transactions stores in a
     9→ * single IndexedDB transaction) so rapid taps can never create negative stock.
    10→ * Profit/report math uses the purchase-price SNAPSHOT captured on each transaction,
    11→ * so later price edits never rewrite historical profit.
    12→ */

    30→// ============ DATE HELPERS (local timezone — never use UTC for business days) ============
    31→
    32→/**
    33→ * Returns YYYY-MM-DD for a Date in LOCAL time.
    34→ * toISOString() uses UTC — for IST (UTC+5:30) sales between 00:00–05:30 local
    35→ * would land on the previous day. This helper fixes that class of bugs.
    36→ */
    37→export function localDateStr(d: Date = new Date()): string {
    38→  const y = d.getFullYear();
    39→  const m = String(d.getMonth() + 1).padStart(2, '0');
    40→  const day = String(d.getDate()).padStart(2, '0');
    41→  return `${y}-${m}-${day}`;
    42→}
    43→
    44→/**
    45→ * Resolves a reporting period to a [from, to] date range (local time).
    46→ */
    47→function resolvePeriodRange(options?: { period?: string; date?: string; from?: string; to?: string }): { from: string; to: string } {
    48→  const now = new Date();
    49→  if (options?.from && options?.to) {
    50→    // Guard against inverted custom ranges (from > to) — silently swap.
    51→    return options.from <= options.to
    52→      ? { from: options.from, to: options.to }
    53→      : { from: options.to, to: options.from };
    54→  }
    55→  if (options?.date) return { from: options.date, to: options.date };
    56→
    57→  const todayStr = localDateStr(now);
    58→  const period = options?.period || 'today';
    59→  if (period === 'week') {
    60→    const weekStart = new Date(now);
    61→    const day = weekStart.getDay();
    62→    const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1); // Monday start
    63→    weekStart.setDate(diff);
    64→    return { from: localDateStr(weekStart), to: todayStr };
    65→  }
    66→  if (period === 'month') {
    67→    return { from: `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`, to: todayStr };
    68→  }
    69→  // today (default) or any unknown period
    70→  return { from: todayStr, to: todayStr };
    71→}

    75→function stripSync<T>(row: Record<string, unknown>): T {
    76→  const { _synced: _s, _dirty: _d, ...rest } = row;
    77→  return rest as T;
    78→}

    80→function sortTxnsDesc(rows: OfflineDBSchema['transactions']['value'][]): OfflineDBSchema['transactions']['value'][] {
    81→  return [...rows].sort((a, b) => {
    82→    if (a.date !== b.date) return b.date.localeCompare(a.date);
    83→    return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    84→  });
    85→}
    86→
    87→function localTxnsToTransactions(localTxns: OfflineDBSchema['transactions']['value'][]): Transaction[] {
    88→  return sortTxnsDesc(localTxns).map(t => stripSync<Transaction>(t));
    89→}

    91→/** Cost basis for a transaction: the SNAPSHOT taken at sale time, falling back to the current product. */
    92→function txnCostBasis(t: Transaction, products: Product[]): number {
    93→  return t.product?.purchasePrice ?? products.find(p => p.id === t.productId)?.purchasePrice ?? 0;
    94→}

   183→/** Only these keys may be changed by an update — never trust raw spreads. */
   184→function applyProductUpdates(
   185→  product: OfflineDBSchema['products']['value'],
   186→  updates: Partial<Product>
   187→): OfflineDBSchema['products']['value'] {
   188→  const next = { ...product };
   189→  if (updates.name !== undefined) {
   190→    const name = String(updates.name).trim();
   191→    if (name) next.name = name;
   192→  }
   193→  if (updates.categoryId !== undefined) next.categoryId = updates.categoryId;
   194→  if (updates.boxNumber !== undefined) next.boxNumber = updates.boxNumber;
   195→  if (updates.quantity !== undefined) {
   196→    next.quantity = Math.max(0, Math.floor(Number(updates.quantity)) || 0);
   197→  }
   198→  if (updates.purchasePrice !== undefined) {
   199→    next.purchasePrice = Math.max(0, Number(updates.purchasePrice)) || 0;
   200→  }
   201→  if (updates.sellingPrice !== undefined) {
   202→    next.sellingPrice = Math.max(0, Number(updates.sellingPrice)) || 0;
   203→  }
   204→  if (updates.lowStockThreshold !== undefined) {
   205→    next.lowStockThreshold = Math.max(0, Math.floor(Number(updates.lowStockThreshold)) || 0);
   206→  }
   207→  return next;
   208→}

   230→export async function deleteProductOffline(id: string, userId: string): Promise<{ message: string }> {
   231→  // Ownership check — a product must never be deleted by a different account
   232→  // on a shared device.
   233→  const db = await getOfflineDB();
   234→  const product = await db.get('products', id);
   235→  if (!product || (userId && product.userId !== userId)) throw new Error('Not found');
   236→
   237→  // Delete related transactions first (cascade)
   238→  const localTxns = await offlineTransactions.getAll(userId);
   239→  const relatedTxns = localTxns.filter(t => t.productId === id);
   240→  for (const txn of relatedTxns) {
   241→    await offlineTransactions.delete(txn.id);
   242→  }
   243→
   244→  await offlineProducts.delete(id);
   245→  return { message: 'Product deleted successfully' };
   246→}

   248→// ============ TRANSACTIONS (Local, ATOMIC stock updates) ============
   249→
   250→/**
   251→ * Creates a transaction AND applies the stock change in ONE atomic IndexedDB
   252→ * transaction. Two rapid sells can never both pass the same stock check.
   253→ *
   254→ * NOTE: idb v8's db.transaction() has NO callback parameter — the correct
   255→ * atomic pattern is: open tx → await requests → `await tx.done`.
   256→ */
   257→export async function createTransactionOffline(transactionData: {
   258→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   259→  productId: string;
   260→  quantity: number;
   261→  unitPrice?: number;
   262→  totalAmount?: number;
   263→  date?: string;
   264→  userId: string;
   265→}): Promise<{ transaction: Transaction }> {
   266→  const quantity = Math.floor(Number(transactionData.quantity));
   267→  if (!Number.isFinite(quantity) || quantity <= 0) throw new Error('Quantity must be a positive whole number');
   268→
   269→  const unitPrice = Number(transactionData.unitPrice ?? 0);
   270→  const totalAmount = Number(transactionData.totalAmount ?? 0);
   271→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   272→  if (!Number.isFinite(totalAmount) || totalAmount < 0) throw new Error('Total amount cannot be negative');
   273→
   274→  const db = await getOfflineDB();
   275→  const now = new Date().toISOString();
   276→  const id = generateRecordId();
   277→
   278→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   279→  const productsStore = tx.objectStore('products');
   280→  const transactionsStore = tx.objectStore('transactions');
   281→
   282→  // Validate BEFORE writing anything — a readwrite tx with no writes commits empty.
   283→  const product = await productsStore.get(transactionData.productId);
   284→  if (!product) {
   285→    throw new Error('Product not found');
   286→  }
   287→
   288→  const isOut = transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL';
   289→  if (isOut && product.quantity < quantity) {
   290→    throw new Error(`Insufficient stock — only ${product.quantity} left`);
   291→  }
   292→
   293→  const quantityChange = transactionData.type === 'STOCK_IN' ? quantity : -quantity;
   294→
   295→  const row: OfflineDBSchema['transactions']['value'] = {
   296→    id,
   297→    productId: transactionData.productId,
   298→    type: transactionData.type,
   299→    quantity,
   300→    unitPrice,
   301→    totalAmount,
   302→    date: transactionData.date || localDateStr(),
   303→    userId: transactionData.userId,
   304→    createdAt: now,
   305→    // Snapshot the product prices at transaction time — historical profit must
   306→    // never change when the user later edits the product's prices.
   307→    product: {
   308→      id: product.id,
   309→      name: product.name,
   310→      purchasePrice: product.purchasePrice,
   311→      sellingPrice: product.sellingPrice,
   312→    },
   313→    _synced: Date.now(),
   314→    _dirty: 0,
   315→  };
   316→
   317→  await transactionsStore.put(row);
   318→  await productsStore.put({
   319→    ...product,
   320→    quantity: product.quantity + quantityChange,
   321→    updatedAt: now,
   322→  });
   323→
   324→  // Commit — rejects (and rolls back both stores) if anything failed.
   325→  await tx.done;
   326→
   327→  return { transaction: stripSync<Transaction>(row) };
   328→}

   343→export async function getDashboardOffline(userId: string) {
   344→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   345→    offlineProducts.getAll(userId),
   346→    offlineCategories.getAll(userId),
   347→    offlineTransactions.getAll(userId),
   348→  ]);
   349→
   350→  const products = localProducts.map(p => stripSync<Product>(p));
   351→  const categories = localCategories.map(c => stripSync<Category>(c));
   352→  const transactions = localTxnsToTransactions(localTransactions);
   353→
   354→  const today = localDateStr();
   355→  const todayTxns = transactions.filter(t => t.date === today);
   356→
   357→  const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   358→  const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   359→  const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   360→  const recentTransactions = transactions.slice(0, 5);
   361→
   362→  const productCountMap: Record<string, number> = {};
   363→  for (const p of products) {
   364→    productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   365→  }
   366→  const catsWithCount = categories.map(c => ({
   367→    ...c,
   368→    _count: { products: productCountMap[c.id] || 0 },
   369→  }));
   370→
   371→  const sevenDaysAgo = new Date();
   372→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   373→  const sevenDaysAgoStr = localDateStr(sevenDaysAgo);
   374→
   375→  const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   376→  const saleMap = new Map<string, { sales: number; quantity: number }>();
   377→  for (const t of saleTxns) {
   378→    const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   379→    existing.sales += t.totalAmount;
   380→    existing.quantity += t.quantity;
   381→    saleMap.set(t.date, existing);
   382→  }
   383→
   384→  const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   385→  for (let i = 6; i >= 0; i--) {
   386→    const d = new Date();
   387→    d.setDate(d.getDate() - i);
   388→    const dateStr = localDateStr(d);
   389→    const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   390→    const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   391→    saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   392→  }
   393→
   394→  const stockByCategory = new Map<string, { quantity: number; value: number }>();
   395→  for (const p of products) {
   396→    const cat = catsWithCount.find(c => c.id === p.categoryId);
   397→    const catName = cat?.name || 'Unknown';
   398→    const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   399→    existing.quantity += p.quantity;
   400→    existing.value += p.quantity * p.purchasePrice;
   401→    stockByCategory.set(catName, existing);
   402→  }
   403→
   404→  const stockOverview = Array.from(stockByCategory.entries())
   405→    .filter(([_, data]) => data.quantity > 0)
   406→    .map(([category, data]) => ({ category, ...data }));
   407→
   408→  return {
   409→    stats: {
   410→      totalItems: products.length,
   411→      lowItems,
   412→      todayTransactions: todayTxns.length,
   413→      stockValue,
   414→    },
   415→    lowStockProducts,
   416→    recentTransactions,
   417→    categories: catsWithCount,
   418→    saleOverview,
   419→    stockOverview,
   420→  };
   421→}

   428→export async function getExpensesOffline(userId: string, options?: { period?: string; date?: string; from?: string; to?: string; category?: string }) {
   429→  let localExpenses = await offlineExpenses.getAll(userId);
   430→
   431→  // Honor the reporting period exactly — otherwise "Today" would show all-time totals.
   432→  const { from, to } = resolvePeriodRange(options);
   433→  localExpenses = localExpenses.filter(e => e.date >= from && e.date <= to);
   434→  if (options?.category) localExpenses = localExpenses.filter(e => e.category === options.category);
   435→
   436→  const expenses = localExpenses
   437→    .map(e => stripSync<Expense>(e))
   438→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   439→  const totalExpense = expenses.reduce((sum, e) => sum + (e.amount || 0), 0);
   440→  const byCategory = expenses.reduce((acc, e) => {
   441→    const cat = e.category || 'other';
   442→    acc[cat] = (acc[cat] || 0) + (e.amount || 0);
   443→    return acc;
   444→  }, {} as Record<string, number>);
   445→
   446→  return {
   447→    expenses,
   448→    summary: { totalExpense, byCategory, count: expenses.length },
   449→  };
   450→}

   521→export async function getCashEntriesOffline(userId: string, options?: { period?: string; date?: string; from?: string; to?: string }) {
   522→  let localEntries = await offlineCashEntries.getAll(userId);
   523→
   524→  // Honor the reporting period.
   525→  const { from, to } = resolvePeriodRange(options);
   526→  localEntries = localEntries.filter(e => e.date >= from && e.date <= to);
   527→
   528→  const entries = localEntries
   529→    .map(e => stripSync<CashEntry>(e))
   530→    .sort((a, b) => b.date.localeCompare(a.date));
   531→  // Balance semantics: the LATEST entry in the filtered period holds the running balance.
   532→  const latest = entries[0];
   533→  const totalHandCash = latest?.handCash || 0;
   534→  const totalLiquidCash = latest?.liquidCash || 0;
   535→
   536→  return {
   537→    entries,
   538→    summary: { totalHandCash, totalLiquidCash, totalCash: totalHandCash + totalLiquidCash },
   539→  };

   961→/** Generates the next sequential bill number: PREFIX-YYMM-0001 */
   962→async function nextBillNumber(userId: string, prefix: string): Promise<string> {
   963→  const bills = await offlineBills.getAll(userId);
   964→  const now = new Date();
   965→  const yy = String(now.getFullYear()).slice(-2);
   966→  const mm = String(now.getMonth() + 1).padStart(2, '0');
   967→  const monthPrefix = `${prefix}-${yy}${mm}-`;
   968→  let maxSeq = 0;
   969→  for (const b of bills) {
   970→    if (typeof b.billNumber === 'string' && b.billNumber.startsWith(monthPrefix)) {
   971→      const seq = parseInt(b.billNumber.slice(monthPrefix.length), 10);
   972→      if (Number.isFinite(seq) && seq > maxSeq) maxSeq = seq;
   973→    }
   974→  }
   975→  return `${monthPrefix}${String(maxSeq + 1).padStart(4, '0')}`;

   978→export async function createBillOffline(data: {
   979→  userId: string;
   980→  transactionId: string;
   981→  productId: string;
   982→  productName: string;
   983→  quantity: number;
   984→  unitPrice: number;
   985→  customerName?: string;
   986→  customerMobile?: string;
   987→  discountType?: 'amount' | 'percent';
   988→  discountValue?: number;
   989→  paymentMethod?: PaymentMethod;
   990→  paidAmount?: number;
   991→  note?: string;
   992→  date?: string;
   993→}): Promise<{ bill: Bill }> {
   994→  const qty = Math.floor(Number(data.quantity));
   995→  if (!Number.isFinite(qty) || qty <= 0) throw new Error('Quantity must be a positive whole number');
   996→  const unitPrice = Number(data.unitPrice);
   997→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   998→
   999→  const settings = await getBillingSettingsOffline(data.userId);
  1000→
  1001→  const discountType = data.discountType === 'percent' ? 'percent' : 'amount';
  1002→  let discountValue = Math.max(0, Number(data.discountValue ?? 0) || 0);
  1003→  const subtotal = round2(qty * unitPrice);
  1004→
  1005→  let discountAmount = 0;
  1006→  if (discountType === 'percent') {
  1007→    discountValue = clampNumber(discountValue, 0, 100);
  1008→    discountAmount = round2((subtotal * discountValue) / 100);
  1009→  } else {
  1010→    discountAmount = round2(Math.min(discountValue, subtotal));
  1011→  }
  1012→
  1013→  const gstEnabled = !!settings.gstEnabled && !!settings.gstNumber.trim() && settings.gstRate > 0;
  1014→  const gstAmount = gstEnabled ? round2(((subtotal - discountAmount) * settings.gstRate) / 100) : 0;
  1015→  const total = round2(subtotal - discountAmount + gstAmount);
  1016→
  1017→  const paymentMethod: PaymentMethod = data.paymentMethod ?? 'cash';
  1018→  let paidAmount = data.paidAmount === undefined ? total : Math.max(0, Number(data.paidAmount) || 0);
  1019→  if (paymentMethod === 'due') paidAmount = 0;
  1020→  paidAmount = round2(Math.min(paidAmount, total));
  1021→  const dueAmount = round2(total - paidAmount);
  1022→
  1023→  // Guard: one bill per stock-out/sell transaction — prevents accidental duplicates
  1024→  const existingBills = await offlineBills.getAll(data.userId);
  1025→  const dupe = existingBills.find(b => b.transactionId === data.transactionId);
  1026→  if (dupe) throw new Error('A bill already exists for this transaction');
  1027→
  1028→  const now = new Date().toISOString();
  1029→  const billNumber = await nextBillNumber(data.userId, settings.billPrefix.trim() || 'PS');
  1030→
  1031→  const bill: OfflineDBSchema['bills']['value'] = {
  1032→    id: generateRecordId(),
  1033→    userId: data.userId,
  1034→    billNumber,
  1035→    customerName: (data.customerName ?? '').trim() || 'Walk-in Customer',
  1036→    customerMobile: (data.customerMobile ?? '').replace(/[^\d+]/g, ''),
  1037→    items: [
  1038→      {
  1039→        productId: data.productId,

  1079→export async function getBillsOffline(userId: string, options?: { from?: string; to?: string; search?: string }): Promise<Bill[]> {
  1080→  let rows = await offlineBills.getAll(userId);
  1081→  if (options?.from) rows = rows.filter(b => b.date >= options.from!);
  1082→  if (options?.to) rows = rows.filter(b => b.date <= options.to!);
  1083→  if (options?.search) {
  1084→    const q = options.search.toLowerCase();
  1085→    rows = rows.filter(b =>
  1086→      b.customerName.toLowerCase().includes(q) ||
  1087→      b.customerMobile.includes(q) ||
  1088→      b.billNumber.toLowerCase().includes(q) ||
  1089→      b.items.some(i => i.name.toLowerCase().includes(q))
  1090→    );
  1091→  }
  1092→  return rows
  1093→    .sort((a, b) => {
  1094→      if (a.date !== b.date) return b.date.localeCompare(a.date);
  1095→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
  1096→    })
  1097→    .map(b => b as unknown as Bill);
  1098→}

  1145→export async function importBackupOffline(userId: string, backupData: {
  1146→  categories: unknown[];
  1147→  products: unknown[];
  1148→  transactions: unknown[];
  1149→  expenses?: unknown[];
  1150→  cashEntries?: unknown[];
  1151→  serviceTransactions?: unknown[];
  1152→  bills?: unknown[];
  1153→  billingSettings?: unknown;
  1154→}) {
  1155→  const now = Date.now();
  1156→
  1157→  // ---- Numeric coercion helpers (same clamps the create* functions use) ----
  1158→  const toNum = (v: unknown): number => {
  1159→    const n = Number(v);
  1160→    return Number.isFinite(n) ? n : 0;
  1161→  };
  1162→  /** Non-negative number — mirrors Math.max(0, Number(x)) || 0 in create*. */
  1163→  const numPos = (v: unknown): number => Math.max(0, toNum(v));
  1164→  /** Non-negative whole number — mirrors Math.max(0, Math.floor(Number(x))) || 0. */
  1165→  const intPos = (v: unknown): number => Math.max(0, Math.floor(toNum(v)) || 0);
  1166→  /** Coerces the listed numeric fields (only when present) on a raw backup row. */
  1167→  const coerce = (row: Record<string, unknown>, fields: Record<string, 'int' | 'num'>): Record<string, unknown> => {
  1168→    const out: Record<string, unknown> = { ...row };
  1169→    for (const [key, kind] of Object.entries(fields)) {
  1170→      if (key in out) out[key] = kind === 'int' ? intPos(out[key]) : numPos(out[key]);
  1171→    }
  1172→    return out;
  1173→  };
  1174→  const hasName = (row: Record<string, unknown>): boolean =>
  1175→    typeof row.name === 'string' && row.name.trim() !== '';
  1176→
  1177→  // Validate rows: every row MUST have an id — malformed backups must never
  1178→  // be written into IndexedDB.
  1179→  const asRows = (rows: unknown[]): Record<string, unknown>[] => {
  1180→    if (!Array.isArray(rows)) return [];
  1181→    return rows.filter((r): r is Record<string, unknown> =>
  1182→      !!r && typeof r === 'object' && typeof (r as Record<string, unknown>).id === 'string' && (r as Record<string, unknown>).id !== ''
  1183→    );
  1184→  };
  1185→
  1186→  const categories = asRows(backupData.categories || []).filter(hasName);
  1187→  const products = asRows(backupData.products || []).filter(hasName);
  1188→  const transactions = asRows(backupData.transactions || []);
  1189→  const expenses = asRows(backupData.expenses || []);
  1190→  const cashEntries = asRows(backupData.cashEntries || []);
  1191→  const serviceTransactions = asRows(backupData.serviceTransactions || []);
  1192→  const bills = asRows(backupData.bills || []);
  1193→  const billingSettings = (backupData.billingSettings && typeof backupData.billingSettings === 'object')
  1194→    ? backupData.billingSettings as Record<string, unknown>
  1195→    : null;
  1196→
  1197→  if (categories.length === 0 && products.length === 0 && transactions.length === 0 && expenses.length === 0 && cashEntries.length === 0 && serviceTransactions.length === 0 && bills.length === 0) {
  1198→    throw new Error('Backup file contains no valid data');
  1199→  }
  1200→
  1201→  // REPLACE semantics: clear existing stores first so stale rows can't survive a restore.
  1202→  const [allLocalCats, allLocalProducts, allLocalTxns, allLocalExpenses, allLocalCash, allLocalSvc, allLocalBills] = await Promise.all([
  1203→    offlineCategories.getAll(userId),
  1204→    offlineProducts.getAll(userId),
  1205→    offlineTransactions.getAll(userId),
  1206→    offlineExpenses.getAll(userId),
  1207→    offlineCashEntries.getAll(userId),
  1208→    offlineServiceTransactions.getAll(userId),
  1209→    offlineBills.getAll(userId),
  1210→  ]);
  1211→  await Promise.all([
  1212→    ...allLocalCats.map(c => offlineCategories.delete(c.id)),
  1213→    ...allLocalProducts.map(p => offlineProducts.delete(p.id)),
  1214→    ...allLocalTxns.map(t => offlineTransactions.delete(t.id)),
  1215→    ...allLocalExpenses.map(e => offlineExpenses.delete(e.id)),
  1216→    ...allLocalCash.map(e => offlineCashEntries.delete(e.id)),
  1217→    ...allLocalSvc.map(s => offlineServiceTransactions.delete(s.id)),
  1218→    ...allLocalBills.map(b => offlineBills.delete(b.id)),
  1219→  ]);
  1220→
  1221→  if (categories.length) {
  1222→    await offlineCategories.putBulk(
  1223→      categories.map((cat) => ({ ...cat, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['categories']['value'][]
  1224→    );
  1225→  }
  1226→  if (products.length) {
  1227→    await offlineProducts.putBulk(
  1228→      products.map((prod) => coerce({ ...prod, userId, _synced: now, _dirty: 0 }, {
  1229→        quantity: 'int',
  1230→        purchasePrice: 'num',
  1231→        sellingPrice: 'num',
  1232→        lowStockThreshold: 'int',
  1233→      })) as OfflineDBSchema['products']['value'][]
  1234→    );
  1235→  }
  1236→  if (transactions.length) {
  1237→    await offlineTransactions.putBulk(
  1238→      transactions.map((txn) => coerce({ ...txn, userId, _synced: now, _dirty: 0 }, {
  1239→        quantity: 'int',
  1240→        unitPrice: 'num',
  1241→        totalAmount: 'num',
  1242→      })) as OfflineDBSchema['transactions']['value'][]
  1243→    );
  1244→  }
  1245→  if (expenses.length) {
  1246→    await offlineExpenses.putBulk(
  1247→      expenses.map((exp) => coerce({ ...exp, userId, _synced: now, _dirty: 0 }, { amount: 'num' })) as OfflineDBSchema['expenses']['value'][]
  1248→    );
  1249→  }
  1250→  if (cashEntries.length) {
  1251→    await offlineCashEntries.putBulk(
  1252→      cashEntries.map((entry) => coerce({ ...entry, userId, _synced: now, _dirty: 0 }, {
  1253→        handCash: 'num',
  1254→        liquidCash: 'num',
  1255→      })) as OfflineDBSchema['cashEntries']['value'][]
  1256→    );
  1257→  }
  1258→  if (serviceTransactions.length) {
  1259→    await offlineServiceTransactions.putBulk(
  1260→      serviceTransactions.map((svc) => coerce({ ...svc, userId, _synced: now, _dirty: 0 }, { amount: 'num' })) as OfflineDBSchema['serviceTransactions']['value'][]
  1261→    );
  1262→  }
  1263→  if (bills.length) {
  1264→    await offlineBills.putBulk(
  1265→      bills.map((bill) => coerce({ ...bill, userId, createdAt: bill.createdAt || new Date().toISOString(), updatedAt: bill.updatedAt || new Date().toISOString() }, {
  1266→        subtotal: 'num',
  1267→        discountValue: 'num',
  1268→        discountAmount: 'num',
  1269→        gstRate: 'num',
  1270→        gstAmount: 'num',
  1271→        total: 'num',
  1272→        paidAmount: 'num',
  1273→        dueAmount: 'num',
  1274→      })) as OfflineDBSchema['bills']['value'][]
  1275→    );
  1276→  }
  1277→  if (billingSettings) {
  1278→    const db = await getOfflineDB();
  1279→    await db.put('billingSettings', {
  1280→      ...DEFAULT_BILLING_SETTINGS,
  1281→      ...billingSettings,
  1282→      userId,
  1283→      updatedAt: new Date().toISOString(),
  1284→    } as OfflineDBSchema['billingSettings']['value']);
  1285→  }
  1286→
  1287→  return { message: 'Backup imported successfully' };
  1288→}
