     1→     1→/**
     2→     2→ * Offline-First Service Layer
     3→     3→ * 
     4→     4→ * READ STRATEGY: Local IndexedDB first → instant response. Then fetch from Supabase in background.
     5→     5→ * WRITE STRATEGY: Save to local IndexedDB immediately (mark dirty) → upload to Supabase in background.
     6→     6→ * DELETE STRATEGY: Delete from local IndexedDB + add to pendingDeletes queue → delete from Supabase when online.
     7→     7→ * 
     8→     8→ * This wraps the existing supabase-service functions with offline support.
     9→     9→ * The app always works - even with zero internet.
    10→    10→ */

   131→   131→export async function deleteCategoryOffline(id: string, userId: string): Promise<{ message: string }> {
   132→   132→  // Delete all products in this category (cascading delete)
   133→   133→  const allProducts = await offlineProducts.getAll(userId);
   134→   134→  const catProducts = allProducts.filter(p => p.categoryId === id);
   135→   135→  for (const prod of catProducts) {
   136→   136→    await deleteProductOffline(prod.id, userId);
   137→   137→  }
   138→   138→
   139→   139→  // Delete the category from local IndexedDB
   140→   140→  await offlineCategories.delete(id);
   141→   141→
   142→   142→  // Add to pending deletes for Supabase sync
   143→   143→  await offlinePendingDeletes.add('categories', id, userId);
   144→   144→
   145→   145→  // Try to delete from Supabase immediately if online
   146→   146→  if (isOnline()) {
   147→   147→    try {
   148→   148→      const { deleteCategory } = await import('./supabase-service');
   149→   149→      await deleteCategory(id);
   150→   150→      await offlinePendingDeletes.removeByItemId('categories', id);
   151→   151→    } catch {
   152→   152→      // Will retry on next sync
   153→   153→    }
   154→   154→  }
   155→   155→
   156→   156→  return { message: 'Category deleted successfully' };
   157→   157→}

   288→   288→export async function deleteProductOffline(id: string, userId: string): Promise<{ message: string }> {
   289→   289→  // Delete from local IndexedDB
   290→   290→  await offlineProducts.delete(id);
   291→   291→  
   292→   292→  // Also delete related transactions locally
   293→   293→  const localTxns = await offlineTransactions.getAll(userId);
   294→   294→  const relatedTxns = localTxns.filter(t => t.productId === id);
   295→   295→  for (const txn of relatedTxns) {
   296→   296→    await offlineTransactions.delete(txn.id);
   297→   297→    await offlinePendingDeletes.removeByItemId('transactions', txn.id);
   298→   298→  }
   299→   299→  
   300→   300→  // Add to pending deletes for Supabase sync
   301→   301→  await offlinePendingDeletes.add('products', id, userId);
   302→   302→
   303→   303→  // Try to delete from Supabase immediately if online
   304→   304→  if (isOnline()) {
   305→   305→    try {
   306→   306→      const { deleteProduct } = await import('./supabase-service');
   307→   307→      await deleteProduct(id);
   308→   308→      await offlinePendingDeletes.removeByItemId('products', id);
   309→   309→    } catch {
   310→   310→      // Will retry on next sync
   311→   311→    }
   312→   312→  }
   313→   313→

   319→   319→export async function createTransactionOffline(transactionData: {
   320→   320→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   321→   321→  productId: string;
   322→   322→  quantity: number;
   323→   323→  unitPrice?: number;
   324→   324→  totalAmount?: number;
   325→   325→  date?: string;
   326→   326→  userId: string;
   327→   327→}): Promise<{ transaction: Transaction }> {
   328→   328→  const id = `local_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
   329→   329→  const now = new Date().toISOString();
   330→   330→
   331→   331→  const products = await offlineProducts.getAll(transactionData.userId);
   332→   332→  const product = products.find(p => p.id === transactionData.productId);
   333→   333→
   334→   334→  if (!product) {
   335→   335→    throw new Error('Product not found');
   336→   336→  }
   337→   337→
   338→   338→  if ((transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL') && product.quantity < transactionData.quantity) {
   339→   339→    throw new Error('Insufficient stock');
   340→   340→  }
   341→   341→
   342→   342→  const quantityChange = transactionData.type === 'STOCK_IN' ? transactionData.quantity : -transactionData.quantity;
   343→   343→
   344→   344→  const transaction = {
   345→   345→    id,
   346→   346→    productId: transactionData.productId,
   347→   347→    type: transactionData.type,
   348→   348→    quantity: transactionData.quantity,
   349→   349→    unitPrice: transactionData.unitPrice ?? 0,
   350→   350→    totalAmount: transactionData.totalAmount ?? 0,
   351→   351→    date: transactionData.date || now.split('T')[0],
   352→   352→    userId: transactionData.userId,
   353→   353→    createdAt: now,
   354→   354→    product: {
   355→   355→      id: product.id,
   356→   356→      name: product.name,
   357→   357→      purchasePrice: product.purchasePrice,
   358→   358→      sellingPrice: product.sellingPrice,
   359→   359→    },
   360→   360→  };
   361→   361→
   362→   362→  await offlineTransactions.put({
   363→   363→    ...transaction,
   364→   364→    _synced: 0,
   365→   365→    _dirty: Date.now(),
   366→   366→  });
   367→   367→
   368→   368→  const updatedProduct = {
   369→   369→    ...product,
   370→   370→    quantity: product.quantity + quantityChange,
   371→   371→    updatedAt: now,
   372→   372→  };
   373→   373→  await offlineProducts.put({
   374→   374→    ...updatedProduct,
   375→   375→    _synced: product._synced,
   376→   376→    _dirty: Date.now(),
   377→   377→  });
   378→   378→
   379→   379→  if (isOnline()) {
   380→   380→    try {
   381→   381→      const { createTransaction } = await import('./supabase-service');
   382→   382→      const result = await createTransaction(transactionData);
   383→   383→      await offlineTransactions.put({
   384→   384→        ...result.transaction,
   385→   385→        _synced: Date.now(),
   386→   386→        _dirty: 0,
   387→   387→      } as OfflineDBSchema['transactions']['value']);
   388→   388→      await offlineTransactions.delete(id);
   389→   389→      const serverProducts = await offlineProducts.getAll(transactionData.userId);
   390→   390→      const serverProduct = serverProducts.find(p => p.id === transactionData.productId);
   391→   391→      if (serverProduct) {
   392→   392→        await offlineProducts.put({
   393→   393→          ...serverProduct,
   394→   394→          quantity: product.quantity + quantityChange,
   395→   395→          _synced: Date.now(),
   396→   396→          _dirty: 0,
   397→   397→        });
   398→   398→      }
   399→   399→      return { transaction: result.transaction as unknown as Transaction };
   400→   400→    } catch {
   401→   401→      // Keep local - will sync later
   402→   402→    }
   403→   403→  }
   404→   404→
   405→   405→  return { transaction: transaction as unknown as Transaction };
   406→   406→}

   408→   408→export async function deleteTransactionOffline(id: string, userId: string): Promise<{ success: boolean }> {
   409→   409→  const allTxns = await offlineTransactions.getAll(userId);
   410→   410→  const txn = allTxns.find(t => t.id === id);
   411→   411→
   412→   412→  if (txn) {
   413→   413→    // Reverse the stock change caused by this transaction
   414→   414→    const allProducts = await offlineProducts.getAll(userId);
   415→   415→    const product = allProducts.find(p => p.id === txn.productId);
   416→   416→    if (product) {
   417→   417→      const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   418→   418→      await offlineProducts.put({
   419→   419→        ...product,
   420→   420→        quantity: Math.max(0, product.quantity + quantityChange),
   421→   421→        _synced: product._synced,
   422→   422→        _dirty: Date.now(),
   423→   423→      } as OfflineDBSchema['products']['value']);
   424→   424→    }
   425→   425→  }
   426→   426→
   427→   427→  // Delete from local IndexedDB
   428→   428→  await offlineTransactions.delete(id);
   429→   429→
   430→   430→  // Add to pending deletes for Supabase sync
   431→   431→  await offlinePendingDeletes.add('transactions', id, userId);
   432→   432→
   433→   433→  // Try to delete from Supabase immediately if online
   434→   434→  if (isOnline()) {
   435→   435→    try {
   436→   436→      const { deleteTransaction } = await import('./supabase-service');
   437→   437→      await deleteTransaction(id);
   438→   438→      await offlinePendingDeletes.removeByItemId('transactions', id);
   439→   439→    } catch {
   440→   440→      // Will retry on next sync
   441→   441→    }
   442→   442→  }
   443→   443→
   444→   444→  return { success: true };
   445→   445→}

   498→   498→
   499→   499→    const today = new Date().toISOString().split('T')[0];
   500→   500→    const todayTxns = transactions.filter(t => t.date === today);
   501→   501→    
   502→   502→    const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   503→   503→    const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   504→   504→    const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   505→   505→    const recentTransactions = transactions.slice(0, 5);
   506→   506→
   507→   507→    const productCountMap: Record<string, number> = {};
   508→   508→    for (const p of products) {
   509→   509→      productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   510→   510→    }
   511→   511→    const catsWithCount = categories.map(c => ({
   512→   512→      ...c,
   513→   513→      _count: { products: productCountMap[c.id] || 0 },
   514→   514→    }));
   515→   515→
   516→   516→    const sevenDaysAgo = new Date();
   517→   517→    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   518→   518→    const sevenDaysAgoStr = sevenDaysAgo.toISOString().split('T')[0];
   519→   519→    
   520→   520→    const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   521→   521→    const saleMap = new Map<string, { sales: number; quantity: number }>();
   522→   522→    for (const t of saleTxns) {
   523→   523→      const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   524→   524→      existing.sales += t.totalAmount;
   525→   525→      existing.quantity += t.quantity;
   526→   526→      saleMap.set(t.date, existing);
   527→   527→    }
   528→   528→
   529→   529→    const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   530→   530→    for (let i = 6; i >= 0; i--) {
   531→   531→      const d = new Date();
   532→   532→      d.setDate(d.getDate() - i);
   533→   533→      const dateStr = d.toISOString().split('T')[0];
   534→   534→      const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   535→   535→      const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   536→   536→      saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   537→   537→    }
   538→   538→
   539→   539→    const stockByCategory = new Map<string, { quantity: number; value: number }>();
   540→   540→    for (const p of products) {
   541→   541→      const cat = catsWithCount.find(c => c.id === p.categoryId);
   542→   542→      const catName = cat?.name || 'Unknown';
   543→   543→      const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   544→   544→      existing.quantity += p.quantity;
   545→   545→      existing.value += p.quantity * p.purchasePrice;
   546→   546→      stockByCategory.set(catName, existing);
   547→   547→    }
   548→   548→
   549→   549→    const stockOverview = Array.from(stockByCategory.entries())
   550→   550→      .filter(([_, data]) => data.quantity > 0)
   551→   551→      .map(([category, data]) => ({ category, ...data }));
   552→   552→
   553→   553→    const result = {
   554→   554→      stats: {
   555→   555→        totalItems: products.length,
   556→   556→        lowItems,
   557→   557→        todayTransactions: todayTxns.length,
   558→   558→        stockValue,
   559→   559→      },
   560→   560→      lowStockProducts,
   561→   561→      recentTransactions,
   562→   562→      categories: catsWithCount,
   563→   563→      saleOverview,
   564→   564→      stockOverview,
   565→   565→    };
   566→   566→
   567→   567→    if (isOnline()) syncFromSupabase(userId).catch(() => {});
   568→   568→
   569→   569→    return result;
   570→   570→  } catch {
   571→   571→    if (isOnline()) {
   572→   572→      const { getDashboard } = await import('./supabase-service');
   573→   573→      return getDashboard(userId);
   574→   574→    }
   575→   575→    return {
   576→   576→      stats: { totalItems: 0, lowItems: 0, todayTransactions: 0, stockValue: 0 },
   577→   577→      lowStockProducts: [],
   578→   578→      recentTransactions: [],
   579→   579→      categories: [],
   580→   580→      saleOverview: [],
   581→   581→      stockOverview: [],
   582→   582→    };
   583→   583→  }
   584→   584→}

   925→   925→export async function getReportsOffline(userId: string, type: string, options?: { from?: string; to?: string }) {
   926→   926→  try {
   927→   927→    // Reports are computed from local data - always works offline!
   928→   928→    const [localProducts, localCategories, localTransactions] = await Promise.all([
   929→   929→      offlineProducts.getAll(userId),
   930→   930→      offlineCategories.getAll(userId),
   931→   931→      offlineTransactions.getAll(userId),
   932→   932→    ]);
   933→   933→
   934→   934→    const products = localProducts.map(({ _synced, _dirty, ...p }) => p as unknown as Product);
   935→   935→    const categories = localCategories.map(({ _synced, _dirty, ...c }) => c as unknown as Category);
   936→   936→    const transactions = localTxnsToTransactions(localTransactions);
   937→   937→
   938→   938→    // Apply date filters
   939→   939→    let filteredTxns = transactions;
   940→   940→    if (options?.from) filteredTxns = filteredTxns.filter(t => t.date >= options.from!);
   941→   941→    if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   942→   942→
   943→   943→    // If no local data and online, fall back to Supabase
   944→   944→    if (products.length === 0 && transactions.length === 0 && isOnline()) {
   945→   945→      const { getReports } = await import('./supabase-service');
   946→   946→      return getReports(userId, type, options);
   947→   947→    }
   948→   948→
   949→   949→    if (type === 'stock-value') {
   950→   950→      // Stock value report from local data
   951→   951→      const productCountMap: Record<string, number> = {};
   952→   952→      for (const p of products) {
   953→   953→        productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
