     1→     1→/**
     2→     2→ * Local-First Service Layer — 100% offline, NO cloud database.
     3→     3→ *
     4→     4→ * READ STRATEGY:  Always served from local IndexedDB (instant, works with zero internet).
     5→     5→ * WRITE STRATEGY: Written to local IndexedDB immediately (atomic where it matters).
     6→     6→ * DELETE STRATEGY: Deleted from local IndexedDB (with cascades where required).
     7→     7→ *

    11→    11→ * so later price edits never rewrite historical profit.

    30→    30→// ============ DATE HELPERS (local timezone — never use UTC for business days) ============
    31→    31→
    32→    32→/**
    33→    33→ * Returns YYYY-MM-DD for a Date in LOCAL time.
    34→    34→ * toISOString() uses UTC — for IST (UTC+5:30) sales between 00:00–05:30 local
    35→    35→ * would land on the previous day. This helper fixes that class of bugs.
    36→    36→ */
    37→    37→export function localDateStr(d: Date = new Date()): string {
    38→    38→  const y = d.getFullYear();
    39→    39→  const m = String(d.getMonth() + 1).padStart(2, '0');
    40→    40→  const day = String(d.getDate()).padStart(2, '0');
    41→    41→  return `${y}-${m}-${day}`;
    42→    42→}

    44→    44→/**
    45→    45→ * Resolves a reporting period to a [from, to] date range (local time).
    46→    46→ */
    47→    47→export function resolvePeriodRange(options?: { period?: string; date?: string; from?: string; to?: string }): { from: string; to: string } {
    48→    48→  const now = new Date();
    49→    49→  if (options?.from && options?.to) {
    50→    50→    // Guard against inverted custom ranges (from > to) — silently swap.
    51→    51→    return options.from <= options.to
    52→    52→      ? { from: options.from, to: options.to }
    53→    53→      : { from: options.to, to: options.from };
    54→    54→  }
    55→    55→  if (options?.date) return { from: options.date, to: options.date };
    56→    56→
    57→    57→  const todayStr = localDateStr(now);
    58→    58→  const period = options?.period || 'today';
    59→    59→  if (period === 'week') {
    60→    60→    const weekStart = new Date(now);
    61→    61→    const day = weekStart.getDay();
    62→    62→    const diff = weekStart.getDate() - day + (day === 0 ? -6 : 1); // Monday start
    63→    63→    weekStart.setDate(diff);
    64→    64→    return { from: localDateStr(weekStart), to: todayStr };
    65→    65→  }
    66→    66→  if (period === 'month') {
    67→    67→    return { from: `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`, to: todayStr };
    68→    68→  }
    69→    69→  // today (default) or any unknown period
    70→    70→  return { from: todayStr, to: todayStr };
    71→    71→}

    75→    75→function stripSync<T>(row: Record<string, unknown>): T {
    76→    76→  const { _synced: _s, _dirty: _d, ...rest } = row;
    77→    77→  return rest as T;
    78→    78→}

    80→    80→function sortTxnsDesc(rows: OfflineDBSchema['transactions']['value'][]): OfflineDBSchema['transactions']['value'][] {
    81→    81→  return [...rows].sort((a, b) => {
    82→    82→    if (a.date !== b.date) return b.date.localeCompare(a.date);
    83→    83→    return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    84→    84→  });
    85→    85→}

    87→    87→function localTxnsToTransactions(localTxns: OfflineDBSchema['transactions']['value'][]): Transaction[] {
    88→    88→  return sortTxnsDesc(localTxns).map(t => stripSync<Transaction>(t));
    89→    89→}

    91→    91→/** Cost basis for a transaction: the SNAPSHOT taken at sale time, falling back to the current product. */
    92→    92→function txnCostBasis(t: Transaction, products: Product[]): number {
    93→    93→  return t.product?.purchasePrice ?? products.find(p => p.id === t.productId)?.purchasePrice ?? 0;
    94→    94→}

   249→   249→// ============ TRANSACTIONS (Local, ATOMIC stock updates) ============
   250→   250→
   251→   251→/**
   252→   252→ * Creates a transaction AND applies the stock change in ONE atomic IndexedDB
   253→   253→ * transaction. Two rapid sells can never both pass the same stock check.
   254→   254→ *
   255→   255→ * NOTE: idb v8's db.transaction() has NO callback parameter — the correct
   256→   256→ * atomic pattern is: open tx → await requests → `await tx.done`.
   257→   257→ */
   258→   258→export async function createTransactionOffline(transactionData: {
   259→   259→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   260→   260→  productId: string;
   261→   261→  quantity: number;
   262→   262→  unitPrice?: number;
   263→   263→  totalAmount?: number;
   264→   264→  date?: string;
   265→   265→  userId: string;
   266→   266→}): Promise<{ transaction: Transaction }> {
   267→   267→  const quantity = Math.floor(Number(transactionData.quantity));
   268→   268→  if (!Number.isFinite(quantity) || quantity <= 0) throw new Error('Quantity must be a positive whole number');
   269→   269→
   270→   270→  const unitPrice = Number(transactionData.unitPrice ?? 0);
   271→   271→  const totalAmount = Number(transactionData.totalAmount ?? 0);
   272→   272→  if (!Number.isFinite(unitPrice) || unitPrice < 0) throw new Error('Price cannot be negative');
   273→   273→  if (!Number.isFinite(totalAmount) || totalAmount < 0) throw new Error('Total amount cannot be negative');
   274→   274→
   275→   275→  const db = await getOfflineDB();
   276→   276→  const now = new Date().toISOString();
   277→   277→  const id = generateRecordId();
   278→   278→
   279→   279→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   280→   280→  const productsStore = tx.objectStore('products');
   281→   281→  const transactionsStore = tx.objectStore('transactions');
   282→   282→
   283→   283→  // Validate BEFORE writing anything — a readwrite tx with no writes commits empty.
   284→   284→  const product = await productsStore.get(transactionData.productId);
   285→   285→  if (!product) {
   286→   286→    throw new Error('Product not found');
   287→   287→  }
   288→   288→
   289→   289→  const isOut = transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL';
   290→   290→  if (isOut && product.quantity < quantity) {
   291→   291→    throw new Error(`Insufficient stock — only ${product.quantity} left`);
   292→   292→  }
   293→   293→
   294→   294→  const quantityChange = transactionData.type === 'STOCK_IN' ? quantity : -quantity;
   295→   295→
   296→   296→  const row: OfflineDBSchema['transactions']['value'] = {
   297→   297→    id,
   298→   298→    productId: transactionData.productId,
   299→   299→    type: transactionData.type,
   300→   300→    quantity,
   301→   301→    unitPrice,
   302→   302→    totalAmount,
   303→   303→    date: transactionData.date || localDateStr(),
   304→   304→    userId: transactionData.userId,
   305→   305→    createdAt: now,
   306→   306→    // Snapshot the product prices at transaction time — historical profit must
   307→   307→    // never change when the user later edits the product's prices.
   308→   308→    product: {
   309→   309→      id: product.id,
   310→   310→      name: product.name,
   311→   311→      purchasePrice: product.purchasePrice,
   312→   312→      sellingPrice: product.sellingPrice,
   313→   313→    },
   314→   314→    _synced: Date.now(),
   315→   315→    _dirty: 0,
   316→   316→  };
   317→   317→
   318→   318→  await transactionsStore.put(row);
   319→   319→  await productsStore.put({
   320→   320→    ...product,
   321→   321→    quantity: product.quantity + quantityChange,
   322→   322→    updatedAt: now,
   323→   323→  });
   324→   324→
   325→   325→  // Commit — rejects (and rolls back both stores) if anything failed.
   326→   326→  await tx.done;
   327→   327→
   328→   328→  return { transaction: stripSync<Transaction>(row) };
   329→   329→}

   331→   331→export async function deleteTransactionOffline(id: string, userId: string): Promise<{ success: boolean }> {
   332→   332→  const db = await getOfflineDB();
   333→   333→  const now = new Date().toISOString();
   334→   334→
   335→   335→  const tx = db.transaction(['products', 'transactions'], 'readwrite');
   336→   336→  const productsStore = tx.objectStore('products');
   337→   337→  const transactionsStore = tx.objectStore('transactions');
   338→   338→
   339→   339→  const txn = await transactionsStore.get(id);
   340→   340→  if (!txn || txn.userId !== userId) {
   341→   341→    // Nothing to delete — no writes were queued, committing empty is harmless.
   342→   342→    await tx.done;
   343→   343→    return { success: true };
   344→   344→  }
   345→   345→
   346→   346→  // Reverse the stock change caused by this transaction (same atomic tx)
   347→   347→  const product = await productsStore.get(txn.productId);
   348→   348→  if (product) {
   349→   349→    const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   350→   350→    await productsStore.put({
   351→   351→      ...product,
   352→   352→      quantity: Math.max(0, product.quantity + quantityChange),
   353→   353→      updatedAt: now,
   354→   354→    });
   355→   355→  }
   356→   356→
   357→   357→  await transactionsStore.delete(id);
   358→   358→  await tx.done;
   359→   359→
   360→   360→  return { success: true };
   361→   361→}

   376→   376→export async function getDashboardOffline(userId: string) {
   377→   377→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   378→   378→    offlineProducts.getAll(userId),
   379→   379→    offlineCategories.getAll(userId),
   380→   380→    offlineTransactions.getAll(userId),
   381→   381→  ]);
   382→   382→
   383→   383→  const products = localProducts.map(p => stripSync<Product>(p));
   384→   384→  const categories = localCategories.map(c => stripSync<Category>(c));
   385→   385→  const transactions = localTxnsToTransactions(localTransactions);
   386→   386→
   387→   387→  const today = localDateStr();
   388→   388→  const todayTxns = transactions.filter(t => t.date === today);
   389→   389→
   390→   390→  const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   391→   391→  const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   392→   392→  const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   393→   393→  const recentTransactions = transactions.slice(0, 5);
   394→   394→
   395→   395→  const productCountMap: Record<string, number> = {};
   396→   396→  for (const p of products) {
   397→   397→    productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   398→   398→  }
   399→   399→  const catsWithCount = categories.map(c => ({
   400→   400→    ...c,
   401→   401→    _count: { products: productCountMap[c.id] || 0 },
   402→   402→  }));
   403→   403→
   404→   404→  const sevenDaysAgo = new Date();
   405→   405→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   406→   406→  const sevenDaysAgoStr = localDateStr(sevenDaysAgo);
   407→   407→
   408→   408→  const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   409→   409→  const saleMap = new Map<string, { sales: number; quantity: number }>();
   410→   410→  for (const t of saleTxns) {
   411→   411→    const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   412→   412→    existing.sales += t.totalAmount;
   413→   413→    existing.quantity += t.quantity;
   414→   414→    saleMap.set(t.date, existing);
   415→   415→  }
   416→   416→
   417→   417→  const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   418→   418→  for (let i = 6; i >= 0; i--) {
   419→   419→    const d = new Date();
   420→   420→    d.setDate(d.getDate() - i);
   421→   421→    const dateStr = localDateStr(d);
   422→   422→    const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   423→   423→    const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   424→   424→    saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   425→   425→  }
   426→   426→
   427→   427→  const stockByCategory = new Map<string, { quantity: number; value: number }>();
   428→   428→  for (const p of products) {
   429→   429→    const cat = catsWithCount.find(c => c.id === p.categoryId);
   430→   430→    const catName = cat?.name || 'Unknown';
   431→   431→    const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   432→   432→    existing.quantity += p.quantity;
   433→   433→    existing.value += p.quantity * p.purchasePrice;
   434→   434→    stockByCategory.set(catName, existing);
   435→   435→  }
   436→   436→
   437→   437→  const stockOverview = Array.from(stockByCategory.entries())
   438→   438→    .filter(([_, data]) => data.quantity > 0)
   439→   439→    .map(([category, data]) => ({ category, ...data }));
   440→   440→
   441→   441→  return {
   442→   442→    stats: {
   443→   443→      totalItems: products.length,
   444→   444→      lowItems,
   445→   445→      todayTransactions: todayTxns.length,
   446→   446→      stockValue,
   447→   447→    },
   448→   448→    lowStockProducts,
   449→   449→    recentTransactions,
   450→   450→    categories: catsWithCount,
   451→   451→    saleOverview,
   452→   452→    stockOverview,
   453→   453→  };
   454→   454→}

   665→   665→export async function getReportsOffline(userId: string, type: string, options?: { from?: string; to?: string }) {
   666→   666→  const [localProducts, localCategories, localTransactions] = await Promise.all([
   667→   667→    offlineProducts.getAll(userId),
   668→   668→    offlineCategories.getAll(userId),
   669→   669→    offlineTransactions.getAll(userId),
   670→   670→  ]);
   671→   671→
   672→   672→  const products = localProducts.map(p => stripSync<Product>(p));
   673→   673→  const categories = localCategories.map(c => stripSync<Category>(c));
   674→   674→  const transactions = localTxnsToTransactions(localTransactions);
   675→   675→
   676→   676→  // Apply date filters
   677→   677→  let filteredTxns = transactions;
   678→   678→  if (options?.from) filteredTxns = filteredTxns.filter(t => t.date >= options.from!);
   679→   679→  if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   680→   680→
   681→   681→  if (type === 'stock-value') {
   682→   682→    const stockValueData = categories.map(cat => {
   683→   683→      const catProducts = products.filter(p => p.categoryId === cat.id);
   684→   684→      const totalQty = catProducts.reduce((sum, p) => sum + p.quantity, 0);
   685→   685→      const totalPurchaseValue = catProducts.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   686→   686→      const totalSellingValue = catProducts.reduce((sum, p) => sum + p.quantity * p.sellingPrice, 0);
   687→   687→      const totalProfit = totalSellingValue - totalPurchaseValue;
   688→   688→      const lowStockCount = catProducts.filter(p => p.quantity <= p.lowStockThreshold).length;
   689→   689→
   690→   690→      return {
   691→   691→        categoryId: cat.id,
   692→   692→        categoryName: cat.name,
   693→   693→        categoryImage: cat.image,
   694→   694→        productCount: catProducts.length,
   695→   695→        totalQty,
   696→   696→        totalPurchaseValue,
   697→   697→        totalSellingValue,
   698→   698→        totalProfit,
   699→   699→        lowStockCount,
   700→   700→        products: catProducts.map(p => ({
   701→   701→          id: p.id,
   702→   702→          name: p.name,
   703→   703→          quantity: p.quantity,
   704→   704→          purchasePrice: p.purchasePrice,
   705→   705→          sellingPrice: p.sellingPrice,
   706→   706→          stockValue: p.quantity * p.purchasePrice,
   707→   707→          purchaseValue: p.quantity * p.purchasePrice,
   708→   708→          lowStock: p.quantity <= p.lowStockThreshold,
   709→   709→        })),
   710→   710→      };
   711→   711→    }).filter(item => item.productCount > 0);
   712→   712→
   713→   713→    const grandTotal = {
   714→   714→      totalProducts: stockValueData.reduce((s, d) => s + d.productCount, 0),
   715→   715→      totalQty: stockValueData.reduce((s, d) => s + d.totalQty, 0),
   716→   716→      totalPurchaseValue: stockValueData.reduce((s, d) => s + d.totalPurchaseValue, 0),
   717→   717→      totalSellingValue: stockValueData.reduce((s, d) => s + d.totalSellingValue, 0),
   718→   718→      totalProfit: stockValueData.reduce((s, d) => s + d.totalProfit, 0),
   719→   719→      totalLowStock: stockValueData.reduce((s, d) => s + d.lowStockCount, 0),
   720→   720→    };
   721→   721→
   722→   722→    return { type: 'stock-value', data: stockValueData, grandTotal };
   723→   723→  }
   724→   724→
   725→   725→  // Daily report
   726→   726→  if (type === 'daily') {
   727→   727→    const dailyMap = new Map<string, {
   728→   728→      date: string; revenue: number; cost: number; profit: number;
   729→   729→      stockIn: number; stockOut: number; sell: number;
   730→   730→    }>();

   742→   742→        entry.profit += (t.totalAmount || 0) - (t.quantity || 0) * costBasis;
   743→   743→        entry.sell += t.quantity || 0;
   744→   744→      } else if (t.type === 'STOCK_IN') {
   745→   745→        entry.cost += (t.quantity || 0) * (t.unitPrice || 0);
   746→   746→        entry.stockIn += t.quantity || 0;
   747→   747→      } else if (t.type === 'STOCK_OUT') {

   883→   883→  if (options?.from) localTxns = localTxns.filter(t => t.date >= options.from!);
   884→   884→  if (options?.to) localTxns = localTxns.filter(t => t.date <= options.to!);
   885→   885→
   886→   886→  return localTxns
   887→   887→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   888→   888→    .map(t => stripSync<ServiceTransaction>(t));
   889→   889→}

   952→   952→export async function importBackupOffline(userId: string, backupData: {
   953→   953→  categories: unknown[];
   954→   954→  products: unknown[];
   955→   955→  transactions: unknown[];
   956→   956→  expenses?: unknown[];
   957→   957→  cashEntries?: unknown[];
   958→   958→  serviceTransactions?: unknown[];
   959→   959→}) {
   960→   960→  const now = Date.now();
   961→   961→
   962→   962→  // Validate rows: every row MUST have an id — malformed backups must never
   963→   963→  // be written into IndexedDB.
   964→   964→  const asRows = (rows: unknown[]): Record<string, unknown>[] => {
   965→   965→    if (!Array.isArray(rows)) return [];
   966→   966→    return rows.filter((r): r is Record<string, unknown> =>
   967→   967→      !!r && typeof r === 'object' && typeof (r as Record<string, unknown>).id === 'string' && (r as Record<string, unknown>).id !== ''
   968→   968→    );
   969→   969→  };
   970→   970→
   971→   971→  const categories = asRows(backupData.categories || []);
   972→   972→  const products = asRows(backupData.products || []);
   973→   973→  const transactions = asRows(backupData.transactions || []);
   974→   974→  const expenses = asRows(backupData.expenses || []);
   975→   975→  const cashEntries = asRows(backupData.cashEntries || []);
   976→   976→  const serviceTransactions = asRows(backupData.serviceTransactions || []);
   977→   977→
   978→   978→  if (categories.length === 0 && products.length === 0 && transactions.length === 0 && expenses.length === 0 && cashEntries.length === 0 && serviceTransactions.length === 0) {
   979→   979→    throw new Error('Backup file contains no valid data');
   980→   980→  }
   981→   981→
   982→   982→  // REPLACE semantics: clear existing stores first so stale rows can't survive a restore.
   983→   983→  const [allLocalCats, allLocalProducts, allLocalTxns, allLocalExpenses, allLocalCash, allLocalSvc] = await Promise.all([
   984→   984→    offlineCategories.getAll(userId),
   985→   985→    offlineProducts.getAll(userId),
   986→   986→    offlineTransactions.getAll(userId),
   987→   987→    offlineExpenses.getAll(userId),
   988→   988→    offlineCashEntries.getAll(userId),
   989→   989→    offlineServiceTransactions.getAll(userId),
   990→   990→  ]);
   991→   991→  await Promise.all([
   992→   992→    ...allLocalCats.map(c => offlineCategories.delete(c.id)),
   993→   993→    ...allLocalProducts.map(p => offlineProducts.delete(p.id)),
   994→   994→    ...allLocalTxns.map(t => offlineTransactions.delete(t.id)),
   995→   995→    ...allLocalExpenses.map(e => offlineExpenses.delete(e.id)),
   996→   996→    ...allLocalCash.map(e => offlineCashEntries.delete(e.id)),
   997→   997→    ...allLocalSvc.map(s => offlineServiceTransactions.delete(s.id)),
   998→   998→  ]);
   999→   999→
  1000→  1000→  if (categories.length) {
  1001→  1001→    await offlineCategories.putBulk(
  1002→  1002→      categories.map((cat) => ({ ...cat, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['categories']['value'][]
  1003→  1003→    );
  1004→  1004→  }
  1005→  1005→  if (products.length) {
  1006→  1006→    await offlineProducts.putBulk(
  1007→  1007→      products.map((prod) => ({ ...prod, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['products']['value'][]
  1008→  1008→    );
  1009→  1009→  }
  1010→  1010→  if (transactions.length) {
  1011→  1011→    await offlineTransactions.putBulk(
  1012→  1012→      transactions.map((txn) => ({ ...txn, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['transactions']['value'][]
  1013→  1013→    );
  1014→  1014→  }
  1015→  1015→  if (expenses.length) {
  1016→  1016→    await offlineExpenses.putBulk(
  1017→  1017→      expenses.map((exp) => ({ ...exp, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['expenses']['value'][]
  1018→  1018→    );
  1019→  1019→  }
  1020→  1020→  if (cashEntries.length) {
  1021→  1021→    await offlineCashEntries.putBulk(
  1022→  1022→      cashEntries.map((entry) => ({ ...entry, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['cashEntries']['value'][]
  1023→  1023→    );
  1024→  1024→  }
  1025→  1025→  if (serviceTransactions.length) {
  1026→  1026→    await offlineServiceTransactions.putBulk(
  1027→  1027→      serviceTransactions.map((svc) => ({ ...svc, userId, _synced: now, _dirty: 0 })) as OfflineDBSchema['serviceTransactions']['value'][]
  1028→  1028→    );
  1029→  1029→  }
  1030→  1030→
  1031→  1031→  return { message: 'Backup imported successfully' };
  1032→  1032→}
