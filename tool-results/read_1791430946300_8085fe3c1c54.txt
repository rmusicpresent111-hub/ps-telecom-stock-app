     1→/**
     2→ * Offline-First Service Layer
     3→ * 
     4→ * READ STRATEGY: Local IndexedDB first → instant response. Then fetch from Supabase in background.
     5→ * WRITE STRATEGY: Save to local IndexedDB immediately (mark dirty) → upload to Supabase in background.
     6→ * DELETE STRATEGY: Delete from local IndexedDB + add to pendingDeletes queue → delete from Supabase when online.
     7→ * 
     8→ * This wraps the existing supabase-service functions with offline support.
     9→ * The app always works - even with zero internet.
    10→ */
    11→

    47→      const { getCategories } = await import('./supabase-service');
    48→      const result = await getCategories(userId);
    49→      return result.categories as Category[];
    50→    }
    51→    return [];
    52→  }

    96→  const existing = allCats.find(c => c.id === id);

    99→  const updated = {
   100→    ...existing,
   101→    ...updates,

   106→    ...updated,
   107→    _synced: existing._synced,
   108→    _dirty: Date.now(),
   109→  } as OfflineDBSchema['categories']['value']);
   110→
   111→  // Try to sync to Supabase immediately if online
   112→  if (isOnline()) {
   113→    try {

   115→      const result = await updateCategory(id, updates);
   116→      await offlineCategories.put({
   117→        ...result.category,
   118→        _synced: Date.now(),
   119→        _dirty: 0,
   120→      } as OfflineDBSchema['categories']['value']);

   123→      // Keep local entry dirty - will sync later
   124→    }

   131→export async function deleteCategoryOffline(id: string, userId: string): Promise<{ message: string }> {
   132→  // Delete all products in this category (cascading delete)
   133→  const allProducts = await offlineProducts.getAll(userId);
   134→  const catProducts = allProducts.filter(p => p.categoryId === id);
   135→  for (const prod of catProducts) {
   136→    await deleteProductOffline(prod.id, userId);
   137→  }
   138→
   139→  // Delete the category from local IndexedDB
   140→  await offlineCategories.delete(id);
   141→
   142→  // Add to pending deletes for Supabase sync
   143→  await offlinePendingDeletes.add('categories', id, userId);
   144→
   145→  // Try to delete from Supabase immediately if online
   146→  if (isOnline()) {
   147→    try {
   148→      const { deleteCategory } = await import('./supabase-service');
   149→      await deleteCategory(id);
   150→      await offlinePendingDeletes.removeByItemId('categories', id);
   151→    } catch {
   152→      // Will retry on next sync
   153→    }
   154→  }
   155→
   156→  return { message: 'Category deleted successfully' };
   157→}

   248→export async function updateProductOffline(id: string, updates: Partial<Product> & { userId: string }): Promise<{ product: Product }> {
   249→  const existing = await offlineProducts.getAll(updates.userId);
   250→  const product = existing.find(p => p.id === id);
   251→  
   252→  if (!product) {
   253→    throw new Error('Product not found');
   254→  }
   255→
   256→  const updatedProduct = {
   257→    ...product,
   258→    ...updates,
   259→    updatedAt: new Date().toISOString(),
   260→  };
   261→
   262→  await offlineProducts.put({
   263→    ...updatedProduct,
   264→    _synced: product._synced,
   265→    _dirty: Date.now(),
   266→  } as OfflineDBSchema['products']['value']);
   267→
   268→  if (isOnline()) {
   269→    try {
   270→      const { updateProduct } = await import('./supabase-service');
   271→      const { userId: _, ...updateFields } = updates;
   272→      const result = await updateProduct(id, updateFields);
   273→      await offlineProducts.put({
   274→        ...result.product,
   275→        _synced: Date.now(),
   276→        _dirty: 0,
   277→      } as OfflineDBSchema['products']['value']);
   278→      return { product: result.product as unknown as Product };
   279→    } catch {
   280→      // Keep local changes - will sync later
   281→    }
   282→  }
   283→
   284→  const { _synced, _dirty, category, ...prodData } = updatedProduct;
   285→  return { product: prodData as unknown as Product };
   286→}

   288→export async function deleteProductOffline(id: string, userId: string): Promise<{ message: string }> {
   289→  // Delete from local IndexedDB
   290→  await offlineProducts.delete(id);
   291→  
   292→  // Also delete related transactions locally
   293→  const localTxns = await offlineTransactions.getAll(userId);
   294→  const relatedTxns = localTxns.filter(t => t.productId === id);
   295→  for (const txn of relatedTxns) {
   296→    await offlineTransactions.delete(txn.id);
   297→    await offlinePendingDeletes.removeByItemId('transactions', txn.id);
   298→  }
   299→  
   300→  // Add to pending deletes for Supabase sync
   301→  await offlinePendingDeletes.add('products', id, userId);
   302→
   303→  // Try to delete from Supabase immediately if online
   304→  if (isOnline()) {
   305→    try {
   306→      const { deleteProduct } = await import('./supabase-service');
   307→      await deleteProduct(id);
   308→      await offlinePendingDeletes.removeByItemId('products', id);
   309→    } catch {
   310→      // Will retry on next sync
   311→    }
   312→  }
   313→
   314→  return { message: 'Product deleted successfully' };
   315→}

   319→export async function createTransactionOffline(transactionData: {
   320→  type: 'STOCK_IN' | 'STOCK_OUT' | 'SELL';
   321→  productId: string;
   322→  quantity: number;
   323→  unitPrice?: number;
   324→  totalAmount?: number;
   325→  date?: string;
   326→  userId: string;
   327→}): Promise<{ transaction: Transaction }> {
   328→  const id = `local_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
   329→  const now = new Date().toISOString();
   330→
   331→  const products = await offlineProducts.getAll(transactionData.userId);
   332→  const product = products.find(p => p.id === transactionData.productId);
   333→
   334→  if (!product) {
   335→    throw new Error('Product not found');
   336→  }
   337→
   338→  if ((transactionData.type === 'STOCK_OUT' || transactionData.type === 'SELL') && product.quantity < transactionData.quantity) {
   339→    throw new Error('Insufficient stock');
   340→  }
   341→
   342→  const quantityChange = transactionData.type === 'STOCK_IN' ? transactionData.quantity : -transactionData.quantity;
   343→
   344→  const transaction = {
   345→    id,
   346→    productId: transactionData.productId,
   347→    type: transactionData.type,
   348→    quantity: transactionData.quantity,
   349→    unitPrice: transactionData.unitPrice ?? 0,
   350→    totalAmount: transactionData.totalAmount ?? 0,
   351→    date: transactionData.date || now.split('T')[0],
   352→    userId: transactionData.userId,
   353→    createdAt: now,
   354→    product: {
   355→      id: product.id,
   356→      name: product.name,
   357→      purchasePrice: product.purchasePrice,
   358→      sellingPrice: product.sellingPrice,
   359→    },
   360→  };
   361→
   362→  await offlineTransactions.put({
   363→    ...transaction,
   364→    _synced: 0,
   365→    _dirty: Date.now(),
   366→  });
   367→
   368→  const updatedProduct = {
   369→    ...product,
   370→    quantity: product.quantity + quantityChange,
   371→    updatedAt: now,
   372→  };
   373→  await offlineProducts.put({
   374→    ...updatedProduct,
   375→    _synced: product._synced,
   376→    _dirty: Date.now(),
   377→  });
   378→
   379→  if (isOnline()) {
   380→    try {
   381→      const { createTransaction } = await import('./supabase-service');
   382→      const result = await createTransaction(transactionData);
   383→      await offlineTransactions.put({
   384→        ...result.transaction,
   385→        _synced: Date.now(),
   386→        _dirty: 0,
   387→      } as OfflineDBSchema['transactions']['value']);
   388→      await offlineTransactions.delete(id);
   389→      const serverProducts = await offlineProducts.getAll(transactionData.userId);
   390→      const serverProduct = serverProducts.find(p => p.id === transactionData.productId);
   391→      if (serverProduct) {
   392→        await offlineProducts.put({
   393→          ...serverProduct,
   394→          quantity: product.quantity + quantityChange,
   395→          _synced: Date.now(),
   396→          _dirty: 0,
   397→        });
   398→      }
   399→      return { transaction: result.transaction as unknown as Transaction };
   400→    } catch {
   401→      // Keep local - will sync later
   402→    }
   403→  }
   404→
   405→  return { transaction: transaction as unknown as Transaction };
   406→}

   408→export async function deleteTransactionOffline(id: string, userId: string): Promise<{ success: boolean }> {
   409→  const allTxns = await offlineTransactions.getAll(userId);
   410→  const txn = allTxns.find(t => t.id === id);
   411→
   412→  if (txn) {
   413→    // Reverse the stock change caused by this transaction
   414→    const allProducts = await offlineProducts.getAll(userId);
   415→    const product = allProducts.find(p => p.id === txn.productId);
   416→    if (product) {
   417→      const quantityChange = txn.type === 'STOCK_IN' ? -txn.quantity : txn.quantity;
   418→      await offlineProducts.put({
   419→        ...product,
   420→        quantity: Math.max(0, product.quantity + quantityChange),
   421→        _synced: product._synced,
   422→        _dirty: Date.now(),
   423→      } as OfflineDBSchema['products']['value']);
   424→    }
   425→  }
   426→
   427→  // Delete from local IndexedDB
   428→  await offlineTransactions.delete(id);
   429→
   430→  // Add to pending deletes for Supabase sync
   431→  await offlinePendingDeletes.add('transactions', id, userId);
   432→
   433→  // Try to delete from Supabase immediately if online
   434→  if (isOnline()) {
   435→    try {
   436→      const { deleteTransaction } = await import('./supabase-service');
   437→      await deleteTransaction(id);
   438→      await offlinePendingDeletes.removeByItemId('transactions', id);
   439→    } catch {
   440→      // Will retry on next sync
   441→    }
   442→  }
   443→
   444→  return { success: true };
   445→}

   447→export async function getTransactionsOffline(userId: string, options?: { type?: string; from?: string; to?: string }): Promise<Transaction[]> {
   448→  try {
   449→    let localTxns = await offlineTransactions.getAll(userId);
   450→    
   451→    if (options?.type) localTxns = localTxns.filter(t => t.type === options.type);
   452→    if (options?.from) localTxns = localTxns.filter(t => t.date >= options.from!);
   453→    if (options?.to) localTxns = localTxns.filter(t => t.date <= options.to!);
   454→
   455→    if (localTxns.length > 0) {
   456→      const txns = localTxns.map(({ _synced, _dirty, ...txn }) => txn as unknown as Transaction);
   457→      if (isOnline()) syncFromSupabase(userId).catch(() => {});
   458→      return txns;
   459→    }
   460→
   461→    if (isOnline()) {
   462→      const { getTransactions } = await import('./supabase-service');
   463→      const result = await getTransactions(userId, options);
   464→      return result.transactions as Transaction[];
   465→    }
   466→
   467→    return [];
   468→  } catch {
   469→    if (isOnline()) {
   470→      const { getTransactions } = await import('./supabase-service');
   471→      const result = await getTransactions(userId, options);
   472→      return result.transactions as Transaction[];
   473→    }
   474→    return [];
   475→  }
   476→}

   480→export async function getDashboardOffline(userId: string) {
   481→  try {
   482→    const [localProducts, localCategories, localTransactions] = await Promise.all([
   483→      offlineProducts.getAll(userId),
   484→      offlineCategories.getAll(userId),
   485→      offlineTransactions.getAll(userId),
   486→    ]);
   487→
   488→    const products = localProducts.map(({ _synced, _dirty, ...p }) => p as unknown as Product);
   489→    const categories = localCategories.map(({ _synced, _dirty, ...c }) => c as unknown as Category);
   490→    const transactions = localTxnsToTransactions(localTransactions);
   491→
   492→    if (products.length === 0 && categories.length === 0 && isOnline()) {
   493→      const { getDashboard } = await import('./supabase-service');
   494→      const result = await getDashboard(userId);
   495→      syncFromSupabase(userId).catch(() => {});
   496→      return result;
   497→    }
   498→
   499→    const today = new Date().toISOString().split('T')[0];
   500→    const todayTxns = transactions.filter(t => t.date === today);
   501→    
   502→    const lowItems = products.filter(p => p.quantity <= p.lowStockThreshold).length;
   503→    const stockValue = products.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   504→    const lowStockProducts = products.filter(p => p.quantity <= p.lowStockThreshold).slice(0, 10);
   505→    const recentTransactions = transactions.slice(0, 5);
   506→
   507→    const productCountMap: Record<string, number> = {};
   508→    for (const p of products) {
   509→      productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   510→    }
   511→    const catsWithCount = categories.map(c => ({
   512→      ...c,
   513→      _count: { products: productCountMap[c.id] || 0 },
   514→    }));
   515→
   516→    const sevenDaysAgo = new Date();
   517→    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   518→    const sevenDaysAgoStr = sevenDaysAgo.toISOString().split('T')[0];
   519→    
   520→    const saleTxns = transactions.filter(t => t.type === 'SELL' && t.date >= sevenDaysAgoStr);
   521→    const saleMap = new Map<string, { sales: number; quantity: number }>();
   522→    for (const t of saleTxns) {
   523→      const existing = saleMap.get(t.date) || { sales: 0, quantity: 0 };
   524→      existing.sales += t.totalAmount;
   525→      existing.quantity += t.quantity;
   526→      saleMap.set(t.date, existing);
   527→    }
   528→
   529→    const saleOverview: { date: string; label: string; sales: number; quantity: number }[] = [];
   530→    for (let i = 6; i >= 0; i--) {
   531→      const d = new Date();
   532→      d.setDate(d.getDate() - i);
   533→      const dateStr = d.toISOString().split('T')[0];
   534→      const dayLabel = d.toLocaleDateString('en-US', { weekday: 'short' });
   535→      const dayData = saleMap.get(dateStr) || { sales: 0, quantity: 0 };
   536→      saleOverview.push({ date: dateStr, label: dayLabel, ...dayData });
   537→    }
   538→
   539→    const stockByCategory = new Map<string, { quantity: number; value: number }>();
   540→    for (const p of products) {
   541→      const cat = catsWithCount.find(c => c.id === p.categoryId);
   542→      const catName = cat?.name || 'Unknown';
   543→      const existing = stockByCategory.get(catName) || { quantity: 0, value: 0 };
   544→      existing.quantity += p.quantity;
   545→      existing.value += p.quantity * p.purchasePrice;
   546→      stockByCategory.set(catName, existing);
   547→    }
   548→
   549→    const stockOverview = Array.from(stockByCategory.entries())
   550→      .filter(([_, data]) => data.quantity > 0)
   551→      .map(([category, data]) => ({ category, ...data }));
   552→
   553→    const result = {
   554→      stats: {
   555→        totalItems: products.length,
   556→        lowItems,
   557→        todayTransactions: todayTxns.length,
   558→        stockValue,
   559→      },
   560→      lowStockProducts,
   561→      recentTransactions,
   562→      categories: catsWithCount,
   563→      saleOverview,
   564→      stockOverview,
   565→    };
   566→
   567→    if (isOnline()) syncFromSupabase(userId).catch(() => {});
   568→
   569→    return result;
   570→  } catch {
   571→    if (isOnline()) {
   572→      const { getDashboard } = await import('./supabase-service');
   573→      return getDashboard(userId);
   574→    }
   575→    return {
   576→      stats: { totalItems: 0, lowItems: 0, todayTransactions: 0, stockValue: 0 },
   577→      lowStockProducts: [],
   578→      recentTransactions: [],
   579→      categories: [],
   580→      saleOverview: [],
   581→      stockOverview: [],
   582→    };
   583→  }
   584→}

   589→function localTxnsToTransactions(localTxns: { _synced: number; _dirty: number; [key: string]: unknown }[]): Transaction[] {
   590→  return localTxns
   591→    .sort((a, b) => new Date(b.createdAt as string).getTime() - new Date(a.createdAt as string).getTime())
   592→    .map(({ _synced, _dirty, ...txn }) => txn as unknown as Transaction);
   593→}

   780→export async function upsertCashEntryOffline(entryData: {
   781→  userId: string;
   782→  date: string;
   783→  handCash?: number;
   784→  liquidCash?: number;
   785→  note?: string;
   786→}): Promise<{ entry: CashEntry }> {
   787→  const now = new Date().toISOString();
   788→  
   789→  // Check if entry exists for this date
   790→  const allEntries = await offlineCashEntries.getAll(entryData.userId);
   791→  const existing = allEntries.find(e => e.date === entryData.date);
   792→  
   793→  let entry: CashEntry;
   794→  
   795→  if (existing) {
   796→    // Update existing
   797→    const updated = {
   798→      ...existing,
   799→      handCash: entryData.handCash !== undefined ? entryData.handCash : existing.handCash,
   800→      liquidCash: entryData.liquidCash !== undefined ? entryData.liquidCash : existing.liquidCash,
   801→      note: entryData.note !== undefined ? entryData.note : existing.note,
   802→      updatedAt: now,
   803→    };
   804→    
   805→    await offlineCashEntries.put({
   806→      ...updated,
   807→      _synced: existing._synced,
   808→      _dirty: Date.now(),
   809→    });
   810→    
   811→    const { _synced, _dirty, ...entryData2 } = updated;
   812→    entry = entryData2 as unknown as CashEntry;
   813→  } else {
   814→    // Create new
   815→    const id = `local_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
   816→    const newEntry = {
   817→      id,
   818→      userId: entryData.userId,
   819→      date: entryData.date,
   820→      handCash: entryData.handCash || 0,
   821→      liquidCash: entryData.liquidCash || 0,
   822→      note: entryData.note || '',
   823→      createdAt: now,
   824→      updatedAt: now,
   825→    };
   826→    
   827→    await offlineCashEntries.put({
   828→      ...newEntry,
   829→      _synced: 0,
   830→      _dirty: Date.now(),
   831→    });
   832→    
   833→    entry = newEntry as CashEntry;
   834→  }
   835→
   836→  // Try to sync to Supabase if online
   837→  if (isOnline()) {
   838→    try {
   839→      const { upsertCashEntry } = await import('./supabase-service');
   840→      const result = await upsertCashEntry(entryData);
   841→      // Update local with server version
   842→      if (existing) {
   843→        await offlineCashEntries.put({
   844→          ...result.entry,
   845→          _synced: Date.now(),
   846→          _dirty: 0,
   847→        });
   848→      } else {
   849→        await offlineCashEntries.put({
   850→          ...result.entry,
   851→          _synced: Date.now(),
   852→          _dirty: 0,
   853→        });
   854→        await offlineCashEntries.delete((entry as { id: string }).id);
   855→      }
   856→      return result;
   857→    } catch {
   858→      // Keep local
   859→    }
   860→  }
   861→
   862→  return { entry };
   863→}

   925→export async function getReportsOffline(userId: string, type: string, options?: { from?: string; to?: string }) {
   926→  try {
   927→    // Reports are computed from local data - always works offline!
   928→    const [localProducts, localCategories, localTransactions] = await Promise.all([
   929→      offlineProducts.getAll(userId),
   930→      offlineCategories.getAll(userId),
   931→      offlineTransactions.getAll(userId),
   932→    ]);
   933→
   934→    const products = localProducts.map(({ _synced, _dirty, ...p }) => p as unknown as Product);
   935→    const categories = localCategories.map(({ _synced, _dirty, ...c }) => c as unknown as Category);
   936→    const transactions = localTxnsToTransactions(localTransactions);
   937→
   938→    // Apply date filters
   939→    let filteredTxns = transactions;
   940→    if (options?.from) filteredTxns = filteredTxns.filter(t => t.date >= options.from!);
   941→    if (options?.to) filteredTxns = filteredTxns.filter(t => t.date <= options.to!);
   942→
   943→    // If no local data and online, fall back to Supabase
   944→    if (products.length === 0 && transactions.length === 0 && isOnline()) {
   945→      const { getReports } = await import('./supabase-service');
   946→      return getReports(userId, type, options);
   947→    }
   948→
   949→    if (type === 'stock-value') {
   950→      // Stock value report from local data
   951→      const productCountMap: Record<string, number> = {};
   952→      for (const p of products) {
   953→        productCountMap[p.categoryId] = (productCountMap[p.categoryId] || 0) + 1;
   954→      }
   955→
   956→      const stockValueData = categories.map(cat => {
   957→        const catProducts = products.filter(p => p.categoryId === cat.id);
   958→        const totalQty = catProducts.reduce((sum, p) => sum + p.quantity, 0);
   959→        const totalPurchaseValue = catProducts.reduce((sum, p) => sum + p.quantity * p.purchasePrice, 0);
   960→        const totalSellingValue = catProducts.reduce((sum, p) => sum + p.quantity * p.sellingPrice, 0);
   961→        const totalProfit = totalSellingValue - totalPurchaseValue;
   962→        const lowStockCount = catProducts.filter(p => p.quantity <= p.lowStockThreshold).length;
   963→
   964→        return {
   965→          categoryId: cat.id,
   966→          categoryName: cat.name,
   967→          categoryImage: cat.image,
   968→          productCount: catProducts.length,
   969→          totalQty,
   970→          totalPurchaseValue,
   971→          totalSellingValue,
   972→          totalProfit,
   973→          lowStockCount,
   974→          products: catProducts.map(p => ({
   975→            id: p.id,
   976→            name: p.name,
   977→            quantity: p.quantity,
   978→            purchasePrice: p.purchasePrice,
   979→            sellingPrice: p.sellingPrice,
   980→            stockValue: p.quantity * p.purchasePrice,
   981→            purchaseValue: p.quantity * p.purchasePrice,
   982→            lowStock: p.quantity <= p.lowStockThreshold,
   983→          })),
   984→        };
   985→      }).filter(item => item.productCount > 0);
   986→
   987→      const grandTotal = {
   988→        totalProducts: stockValueData.reduce((s, d) => s + d.productCount, 0),
   989→        totalQty: stockValueData.reduce((s, d) => s + d.totalQty, 0),
   990→        totalPurchaseValue: stockValueData.reduce((s, d) => s + d.totalPurchaseValue, 0),
   991→        totalSellingValue: stockValueData.reduce((s, d) => s + d.totalSellingValue, 0),
   992→        totalProfit: stockValueData.reduce((s, d) => s + d.totalProfit, 0),
   993→        totalLowStock: stockValueData.reduce((s, d) => s + d.lowStockCount, 0),
   994→      };
   995→
   996→      // Background sync
   997→      if (isOnline()) syncFromSupabase(userId).catch(() => {});
   998→
   999→      return { type: 'stock-value', data: stockValueData, grandTotal };
  1000→    }
  1001→

  1003→    if (type === 'daily') {

  1007→      }>();

  1010→        if (!dailyMap.has(t.date)) {
  1011→          dailyMap.set(t.date, { date: t.date, revenue: 0, cost: 0, profit: 0, stockIn: 0, stockOut: 0, sell: 0 });

  1015→

  1017→          entry.revenue += t.totalAmount || 0;

  1117→          entry.cost += (t.quantity || 0) * (t.unitPrice || 0);
  1118→          entry.stockIn += t.quantity || 0;
  1119→          entry.quantity += t.quantity || 0;
  1120→        } else if (t.type === 'STOCK_OUT') {
  1121→          entry.stockOut += t.quantity || 0;
  1122→          entry.quantity += t.quantity || 0;
  1123→        }
  1124→      }
  1125→
  1126→      const result = {
  1127→        type: 'category',
  1128→        data: Array.from(categoryMap.values()).sort((a, b) => b.revenue - a.revenue),
  1129→        grandTotal: {
  1130→          totalRevenue: Array.from(categoryMap.values()).reduce((s, d) => s + d.revenue, 0),
  1131→          totalCost: Array.from(categoryMap.values()).reduce((s, d) => s + d.cost, 0),
  1132→          totalProfit: Array.from(categoryMap.values()).reduce((s, d) => s + d.profit, 0),
  1133→          totalStockIn: Array.from(categoryMap.values()).reduce((s, d) => s + d.stockIn, 0),
  1134→          totalStockOut: Array.from(categoryMap.values()).reduce((s, d) => s + d.stockOut, 0),
  1135→          totalSell: Array.from(categoryMap.values()).reduce((s, d) => s + d.sell, 0),
  1136→          totalQuantity: Array.from(categoryMap.values()).reduce((s, d) => s + d.quantity, 0),
  1137→        },
  1138→      };
  1139→
  1140→      if (isOnline()) syncFromSupabase(userId).catch(() => {});
  1141→      return result;
  1142→    }
  1143→
  1144→    // Fallback to online for unknown report types
  1145→    if (isOnline()) {
  1146→      const { getReports } = await import('./supabase-service');
  1147→      return getReports(userId, type, options);
  1148→    }
  1149→
  1150→    return { type, data: [], grandTotal: {} };
  1151→  } catch {
  1152→    if (isOnline()) {
  1153→      const { getReports } = await import('./supabase-service');
  1154→      return getReports(userId, type, options);
  1155→    }
  1156→    return { type, data: [], grandTotal: {} };
  1157→  }
  1158→}
